   164→const Snd = {
   165→  ctx: null, master: null, noiseBuf: null, muted: false, ready: false,
   166→
   167→  init() {
   168→    if (this.ready) return;
   169→    try {
   170→      const AC = window.AudioContext || window.webkitAudioContext;
   171→      if (!AC) return;
   172→      this.ctx = new AC();
   173→      this.master = this.ctx.createGain();
   174→      this.master.gain.value = this.muted ? 0 : 0.5;
   175→      this.master.connect(this.ctx.destination);
   176→      /* بافر نویز سفید یک‌ثانیه‌ای (برای شلیک/برخورد/باد/رعد) */
   177→      this.noiseBuf = this.ctx.createBuffer(1, this.ctx.sampleRate, this.ctx.sampleRate);
   178→      const d = this.noiseBuf.getChannelData(0);
   179→      for (let j = 0; j < d.length; j++) d[j] = Math.random() * 2 - 1;
   180→      this.ready = true;
   181→      this.startAmbient();
   182→    } catch (e) { /* مرورگر بدون صدا */ }
   183→  },
   184→  resume() { if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
   185→  toggleMute() {
   186→    this.muted = !this.muted;
   187→    if (this.master) this.master.gain.value = this.muted ? 0 : 0.5;
   188→    return this.muted;
   189→  },
   190→
   191→  /* یک نُت: {type,f0,f1,dur,vol,att,delay} */
   192→  tone(o) {
   193→    if (!this.ready || this.muted) return;
   194→    try {
   195→      const t0 = this.ctx.currentTime + (o.delay || 0);
   196→      const osc = this.ctx.createOscillator(), g = this.ctx.createGain();
   197→      osc.type = o.type || 'sine';
   198→      osc.frequency.setValueAtTime(Math.max(1, o.f0), t0);
   199→      if (o.f1) osc.frequency.exponentialRampToValueAtTime(Math.max(1, o.f1), t0 + o.dur);
   200→      g.gain.setValueAtTime(0.0001, t0);
   201→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.008));
   202→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   203→      osc.connect(g); g.connect(this.master);
   204→      osc.start(t0); osc.stop(t0 + o.dur + 0.05);
   205→    } catch (e) {}
   206→  },
   207→  /* نویز فیلترشده: {dur,vol,ftype,freq,q,att,delay} */
   208→  noise(o) {
   209→    if (!this.ready || this.muted) return;
   210→    try {
   211→      const t0 = this.ctx.currentTime + (o.delay || 0);
   212→      const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf; src.loop = true;
   213→      const f = this.ctx.createBiquadFilter();
   214→      f.type = o.ftype || 'lowpass'; f.frequency.value = o.freq || 1000; f.Q.value = o.q || 0.8;
   215→      const g = this.ctx.createGain();
   216→      g.gain.setValueAtTime(0.0001, t0);
   217→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.005));
   218→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   219→      src.connect(f); f.connect(g); g.connect(this.master);
   220→      src.start(t0); src.stop(t0 + o.dur + 0.05);
   221→    } catch (e) {}
   222→  },
   223→
   224→  /* موسیقی محیطیِ شب: درونِ بم + باد + نُت‌های پراکنده‌ی دل‌انگیز */
   225→  startAmbient() {
   226→    try {
   227→      const dg = this.ctx.createGain(); dg.gain.value = 0.045; dg.connect(this.master);
   228→      const o1 = this.ctx.createOscillator(); o1.type = 'triangle'; o1.frequency.value = 55;
   229→      const o2 = this.ctx.createOscillator(); o2.type = 'sine'; o2.frequency.value = 110;
   230→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.09;
   231→      const lg = this.ctx.createGain(); lg.gain.value = 0.02;
   232→      lfo.connect(lg); lg.connect(dg.gain);
   233→      o1.connect(dg); o2.connect(dg);
   234→      o1.start(); o2.start(); lfo.start();
   235→      /* صدای باد (نویز لوپِ بسیار بم) */
   236→      const wind = this.ctx.createBufferSource(); wind.buffer = this.noiseBuf; wind.loop = true;
   237→      const wf = this.ctx.createBiquadFilter(); wf.type = 'lowpass'; wf.frequency.value = 260;
   238→      const wg = this.ctx.createGain(); wg.gain.value = 0.02;
   239→      wind.connect(wf); wf.connect(wg); wg.connect(this.master); wind.start();
   240→      this.pluck();
   241→    } catch (e) {}
   242→  },

   540→function tryShoot() {
   541→  if (fireTimer > 0) return;
   542→  const w = WEAPONS[state.weapon];
   543→  /* قدرت «شلیک سریع» نرخ شلیک را تقریباً دوبرابر می‌کند */
   544→  fireTimer = w.rate * (state.powerTimers.rapid > 0 ? 0.55 : 1);
   545→  updateAim();                             // اطمینان از جهت صحیح گلوله
   546→  const m = gunMuzzle(w.barrel);
   547→  for (let i = 0; i < w.pellets; i++) {
   548→    const a = state.aimAngle + (Math.random() - 0.5) * 2 * w.spread;
   549→    const sp = w.spd * rand(0.92, 1.08);
   550→    state.bullets.push({
   551→      x: m.x, y: m.y,
   552→      vx: Math.cos(a) * sp, vy: Math.sin(a) * sp,
   553→      dmg: weaponDmg(w), r: w.pellets > 1 ? 3 : 4, life: w.ttl,
   554→      knock: w.knock,                      // پس‌زنی سلاح (۳٫۰)
   555→      dead: false, pellet: w.pellets > 1,  // ساچمه‌های شاتگان ریزترند
   556→    });
   557→  }
   558→  state.stats.shots++;
   559→  state.fireFlash = 0.07;
   560→  state.muzzleGlow = 0.07;                 // نور دهانه روی محیط (۳٫۰)
   561→  state.recoil = w.pellets > 1 ? 1.6 : 1;  // لگد شاتکن محسوس‌تر است
   562→  SFX[w.sfx]();
   563→  spawnParticles(m.x, m.y, 3, '#ffe9a8', { speed: 90, life: 0.18, size: 2, grav: 0,
   564→    dir: state.aimAngle, spread: Math.PI / 3 });
   565→  spawnShell(state.aimAngle);              // پوکه‌ی پرتاب‌شده
   566→  if (w.pellets > 1) state.shake = Math.min(8, state.shake + 2);
   567→}

   569→/* پوکه‌ی براقی که بعد از هر شلیک به عقب پرتاب می‌شود و روی بام می‌غلتد */
   570→function spawnShell(ang) {
   571→  const gx = TOWER.x + Math.cos(ang) * 8, gy = TOWER_TOP_Y - 17 + Math.sin(ang) * 8;
   572→  const a = ang + rand(1.6, 2.2);          // زاویه‌ی پرتاب به عقب/بالا
   573→  state.particles.push({
   574→    x: gx, y: gy,
   575→    vx: Math.cos(a) * rand(60, 130), vy: Math.sin(a) * rand(60, 130) - 70,
   576→    age: 0, life: rand(0.5, 0.8), size: 2.6, color: '#d9a441', grav: 900,
   577→    shell: true, rot: rand(0, 6.28), vr: rand(-9, 9),
   578→  });
   579→}
   580→
   581→/* برخورد گلوله با زامبی‌ها + خروج گلوله از صحنه
   582→   (۳٫۰: اول دایره‌ی سر بررسی می‌شود — برخورد سر = ضربه‌ی بحرانی ×۲) */
   583→function updateBullets(dt) {
   584→  for (const b of state.bullets) {
   585→    if (b.dead) continue;
   586→    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
   587→    if (b.y > GROUND_Y + 4) {              // گلوله به زمین خورد
   588→      b.dead = true;
   589→      spawnParticles(b.x, GROUND_Y, 4, '#6a5a43', { speed: 70, life: 0.3, size: 2, grav: 300, up: true });
   590→      continue;
   591→    }
   592→    if (b.x < -40 || b.x > W + 40 || b.y < -40 || b.life <= 0) { b.dead = true; continue; }
   593→    const dir = Math.sign(b.vx) || 1;      // جهت پس‌زنی = جهت حرکت گلوله
   594→    for (const z of state.zombies) {
   595→      if (z.dead) continue;
   596→      const s = (z.radius / 15) * (z.depth || 1);   // مرکز بدن بر اساس اندازه‌ی نوع و عمق
   597→      /* --- سر: ضربه‌ی بحرانی --- */
   598→      const hx = z.x + z.dir * 3 * s, hy = z.y - 36 * s, hr = 7 * s + b.r;
   599→      if ((b.x - hx) * (b.x - hx) + (b.y - hy) * (b.y - hy) < hr * hr) {
   600→        b.dead = true;
   601→        hitZombie(z, b.dmg, b.x, b.y, dir, true, b.knock);
   602→        break;
   603→      }
   604→      /* --- بدن --- */
   605→      const dx = z.x - b.x, dy = (z.y - 20 * s) - b.y;
   606→      const rr2 = z.radius * (z.depth || 1) + b.r + 3;
   607→      if (dx * dx + dy * dy < rr2 * rr2) {
   608→        b.dead = true;
   609→        hitZombie(z, b.dmg, b.x, b.y, dir, false, b.knock);
   610→        break;
   611→      }
   612→    }
   613→  }
   614→  state.bullets = state.bullets.filter(b => !b.dead);
   615→}
   616→
   617→/* ---------------------- ۸) زامبی‌ها: محاصره‌ی دسته‌جمعی برج ----------------------
   618→   در ۳٫۰ زامبی‌ها دیگر در یک خط نمی‌ایستند: هر طرفِ برج تا ۵ ردیف
   619→   «دیوارِ زامبی» می‌سازند؛ نزدیک‌ترین مهاجم ردیفِ چسبیده به دیوار را می‌گیرد و
   620→   بقیه پشت سرش ردیف می‌شوند. ردیف‌های عقب بالاتر و کوچک‌تر رسم می‌شوند و
   621→   با کشته‌شدنِ ردیف جلو، بقیه فشرده شده و جلو می‌خزند. */
   622→function updateZombies(dt) {
   623→  /* رتبه‌بندی مهاجم‌های فعلی بر اساس نزدیکی به دیوار (هر طرف جداگانه):
   624→     نزدیک‌ترین = ردیف ۰؛ با مرگ ردیف جلو، رتبه‌ها فشرده می‌شوند */
   625→  const rank = { '-1': 0, '1': 0 };
   626→  const attackers = state.zombies.filter(z => !z.dead && z.attacking);
   627→  attackers.sort((a, b) => Math.abs(TOWER.x - a.x) - Math.abs(TOWER.x - b.x));
   628→  for (const z of attackers) {
   629→    z.attackSide = z.x < TOWER.x ? -1 : 1;
   630→    z.lane = Math.min(rank[z.attackSide]++, 4);
   631→  }
   632→
   633→  for (const z of state.zombies) {
   634→    if (z.dead) { z.deathT += dt; continue; }   // انیمیشن مرگ ادامه پیدا کند
   635→    z.hitFlash = Math.max(0, z.hitFlash - dt);
   636→    z.stagger = Math.max(0, z.stagger - dt);    // سرخوردن بعد از ضربه
   637→
   638→    const dx = TOWER.x - z.x;
   639→    const side = z.x < TOWER.x ? -1 : 1;        // کدام دیوار برج؟
   640→    const lane = z.attacking ? z.lane : Math.min(rank[side], 4);
   641→    const stop = TOWER.baseHalf + z.radius * 0.55 + lane * 15;
   642→
   643→    /* انجماد: همه کُند می‌شوند (پادشاه مقاوم‌تر است) */
   644→    const slow = state.powerTimers.freeze > 0 ? (z.type === 'boss' ? 0.6 : 0.32) : 1;
   645→    const spd = z.speed * slow * (z.stagger > 0 ? 0.25 : 1);
   646→
   647→    if (Math.abs(dx) > stop) {
   648→      /* راه رفتن به سمت برج (مهاجمِ پس‌زده‌شده هم برمی‌گردد) */
   649→      z.attacking = false;
   650→      z.lane = Math.min(rank[side], 4);
   651→      z.x += Math.sign(dx) * spd * dt;
   652→      z.walkPhase += dt * (4 + spd * 0.06);
   653→    } else {
   654→      /* رسید به دیوار برج: کوبیدن پایه با فاصله‌ی زمانیِ نوعِ خودش */
   655→      z.attacking = true;
   656→      z.attackSide = side;
   657→      z.attackAnim += dt * 5;
   658→      z.biteTimer -= dt;
   659→      if (z.biteTimer <= 0) {
   660→        z.biteTimer = z.biteInt;
   661→        damageTower(z.dmg, z.x);
   662→      }
   663→    }
   664→
   665→    /* عمق تصویری: ردیف‌های عقب‌تر کمی بالاتر ایستاده و کوچک‌تر دیده می‌شوند */
   666→    const yTarget = GROUND_Y - z.lane * 9 + z.yJit;
   667→    z.y += (yTarget - z.y) * Math.min(1, dt * 6);
   668→    z.depth = 1 - z.lane * 0.07;
   669→  }
   670→  state.zombies = state.zombies.filter(z => !z.dead || z.deathT < 0.55);
   671→}
   672→
   673→/* آسیب گلوله به زامبی (۳٫۰: کریت ×۲ + پس‌زنی + سرخوردن + نشانگر برخورد) */
   674→function hitZombie(z, dmg, bx, by, dir, crit, knock) {
   675→  if (z.dead) return;
   676→  const total = crit ? dmg * 2 : dmg;
   677→  z.hp -= total;
   678→  z.hitFlash = 0.15;
   679→  z.stagger = 0.15;                         // چند لحظه از می‌افتد
   680→  z.x += dir * (knock || 0) * (crit ? 1.7 : 1);   // پس‌زنی فیزیکی
   681→  if (crit) {
   682→    addText(z.x, z.y - 52, faNum(total), '#ffb347', 16);
   683→    spawnParticles(bx, by, 5, '#ffd873', { speed: 150, life: 0.35, size: 2.2, grav: 300 });
   684→  } else {
   685→    addText(z.x, z.y - 46, faNum(total), '#ffffff', 12);
   686→  }
   687→  spawnParticles(bx, by, 3, '#8fd06a', { speed: 110, life: 0.3, size: 2, grav: 400 });
   688→  spawnParticles(bx, by, 3, '#7a2f2f', { speed: 90,  life: 0.35, size: 2, grav: 500 });
   689→  if (Math.random() < 0.3) addDecal(z.x, false);   // چکه‌ی خون روی زمین
   690→  state.hitMark = 0.14; state.hitMarkCrit = !!crit; // نشانگر برخورد روی نشان
   691→  if (crit) SFX.crit(); else SFX.hit();
   692→  if (z.hp <= 0) killZombie(z);
   693→}

   695→/* مرگ زامبی: انیمیشن افتادن + خون + سکه + روح + جعبه‌ی قدرت + کمبو (۳٫۰) */
   696→function killZombie(z) {
   697→  if (z.dead) return;
   698→  z.dead = true; z.deathT = 0;
   699→  state.stats.kills++;
   700→  spawnParticles(z.x, z.y - 18, 12, '#5f8f3e', { speed: 170, life: 0.55, size: 3, grav: 500 });
   701→  spawnParticles(z.x, z.y - 18, 8, '#7a2f2f', { speed: 130, life: 0.5, size: 2.6, grav: 500 });
   702→  addDecal(z.x, true);                     // لکه‌ی خون بزرگ زیر جسد
   703→  /* روحِ سبزِ کوچکی که آرام به آسمان می‌رود */
   704→  state.particles.push({
   705→    x: z.x, y: z.y - 26, vx: rand(-6, 6), vy: -38,
   706→    age: 0, life: 1.3, size: 3, color: 'rgba(190,255,200,0.5)', grav: -26, glow: true,
   707→  });
   708→  SFX.death();
   709→  dropCoins(z);
   710→  maybeDropPowerup(z);
   711→  bumpCombo();
   712→  updateHUD();
   713→  if (z.type === 'boss') {
   714→    state.shake = 16; state.flash = 0.8;
   715→    state.slowmo = 0.9;                    // لحظه‌ی سینمایی مرگ پادشاه
   716→    SFX.slowmo();
   717→    spawnParticles(z.x, z.y - 60, 40, '#ffd873', { speed: 300, life: 1.0, size: 3.5, grav: 400 });
   718→    spawnParticles(z.x, z.y - 60, 24, '#e8563f', { speed: 240, life: 0.8, size: 3, grav: 400 });
   719→    setAnnounce('👑 پادشاه کشته شد!', 'برج پابرجایی!', 2.2);
   720→    SFX.bossDie();
   721→  }
   722→}

   724→/* سیستم کمبو (۳٫۰): هر کشتنِ تازه، پنجره را باز نگه می‌دارد؛
   725→   در رکوردهای مشخص پیام + سکه‌ی جایزه می‌گیری. */
   726→function bumpCombo() {

   728→  state.comboT = COMBO_WINDOW;

   737→    SFX.combo(state.combo);
   738→    pulseCoins();
   739→  }

   743→function damageTower(amount, srcX) {

   745→  state.tower.hp = Math.max(0, state.tower.hp - amount);

   750→  /* تراشه‌های سنگ از پایه‌ی برج می‌پارد (۳٫۰: بیشتر و پررنگ‌تر) */

   754→    { speed: 100, life: 0.35, size: 2, grav: 500 });

   756→  updateHUD();

   760→/* تله پایه برج: آسیب خودکار و پیوسته به زامبی‌های نزدیک */
   761→function updateTrap(dt) {
   762→  const lvl = state.upgrades.trap;
   763→  if (lvl <= 0) return;
   764→  const dps = UPGRADES.trap.dps + UPGRADES.trap.dpsPerLvl * (lvl - 1);
   765→  for (const z of state.zombies) {
   766→    if (z.dead) continue;
   767→    if (Math.abs(z.x - TOWER.x) < UPGRADES.trap.radius) {
   768→      z.hp -= dps * dt;
   769→      z.hitFlash = Math.max(z.hitFlash, 0.06);
   770→      if (Math.random() < dt * 6) { addZap(z.x, z.y - 18); SFX.trap(); }  // قوس برق نمایشی
   771→      if (z.hp <= 0) killZombie(z);
   772→    }
   773→  }
   774→}

   777→/* با مرگ هر زامبی چند سکه می‌پرد بیرون (تعدادش به نوع دشمن بستگی دارد) */
   778→function dropCoins(z) {
   779→  const range = ZTYPES[z.type].coins;
   780→  const n = Math.round(rand(range[0], range[1]));
   781→  for (let i = 0; i < n; i++) {
   782→    state.coinsArr.push({
   783→      x: z.x, y: z.y - 18, vx: rand(-100, 100), vy: rand(-280, -140),
   784→      phase: 'drop', t: 0, bounces: 0, flySpeed: 420, dead: false,
   785→    });
   786→  }
   787→}

   789→/* محل شمارنده سکه در HUD (تبدیل مختصات صفحه به مختصات بوم تا سکه‌ها دقیقاً به آن پرو کنند) */
   790→function coinTarget() {
   791→  try {
   792→    const r = canvas.getBoundingClientRect();
   793→    const c = $('coinItem').getBoundingClientRect();
   794→    return {
   795→      x: (c.left + c.width / 2 - r.left) / r.width * W,
   796→      y: (c.top + c.height / 2 - r.top) / r.height * H,
   797→    };
   798→  } catch (e) { return { x: W - 150, y: 26 }; }
   799→}

   801→function updateCoins(dt) {
   802→  const target = coinTarget(); // یک بار در هر فریم محاسبه می‌شود
   803→  for (const c of state.coinsArr) {
   804→    if (c.dead) continue;
   805→    c.t += dt;
   806→    if (c.phase === 'drop') {
   807→      /* پرتاب اولیه + جاذبه + وایراستن روی زمین */
   808→      c.vy += 900 * dt;
   809→      c.x += c.vx * dt; c.y += c.vy * dt;
   810→      if (c.y > GROUND_Y - 6) {
   811→        c.y = GROUND_Y - 6;
   812→        c.vy *= -0.45; c.vx *= 0.7; c.bounces++;
   813→        if (Math.abs(c.vy) < 40 || c.bounces >= 2 || c.t > 0.9) c.phase = 'fly';
   814→      }
   815→    } else {
   816→      /* پرو شتاب‌دار به سمت شمارنده سکه در HUD */
   817→      const dx = target.x - c.x, dy = target.y - c.y;
   818→      const d = Math.hypot(dx, dy) || 1;
   819→      c.flySpeed += 1400 * dt;
   820→      const step = Math.min(d, c.flySpeed * dt);
   821→      c.x += dx / d * step; c.y += dy / d * step;
   822→      if (d < 18) { collectCoin(c); continue; }
   823→    }
   824→    /* برق کوچکی که گاهی از سکه می‌پرد (۳٫۰) */
   825→    if (Math.random() < dt * 2.2) {
   826→      state.particles.push({ x: c.x + rand(-5, 5), y: c.y + rand(-5, 5), vx: 0, vy: -14,
   827→        age: 0, life: 0.35, size: 1.2, color: '#ffe98a', grav: 0 });
   828→    }
   829→    if (c.t > 4) collectCoin(c); // بیمه: سکه گم‌نشده جمع شود
   830→  }
   831→  state.coinsArr = state.coinsArr.filter(c => !c.dead);
   832→}

   834→function collectCoin(c) {
   835→  if (c.dead) return;
   836→  c.dead = true;
   837→  state.coins++;
   838→  state.stats.earned++;
   839→  updateHUD();
   840→  pulseCoins();
   841→}

   868→    spawnPowerupObj(pickPowerType(), z.x - 24, z.y - 40);

   875→  if (state.killsSinceDrop > 24 || Math.random() < 0.085) {

   881→function updatePowerups(dt) {
   882→  const tx = TOWER.x, ty = TOWER_TOP_Y - 24;   // مقصد: بازیکن روی بام
   883→  for (const p of state.powerups) {
   884→    if (p.dead) continue;
   885→    p.t += dt;
   886→    if (p.phase === 'drop') {
   887→      p.vy += 800 * dt;
   888→      p.x += p.vx * dt; p.y += p.vy * dt;
   889→      if (p.y > GROUND_Y - 10) {
   890→        p.y = GROUND_Y - 10;
   891→        p.vy *= -0.42; p.vx *= 0.65;
   892→        if (Math.abs(p.vy) < 30 || p.t > 0.8) p.phase = 'wait';
   893→      }
   894→    } else if (p.phase === 'wait') {
   895→      if (p.t > 1.15) p.phase = 'fly';    // مکث کوتاه روی زمین، بعد پرو به بازیکن
   896→    } else {
   897→      const dx = tx - p.x, dy = ty - p.y, d = Math.hypot(dx, dy) || 1;
   898→      p.fs += 900 * dt;
   899→      const step = Math.min(d, p.fs * dt);
   900→      p.x += dx / d * step; p.y += dy / d * step;
   901→      if (d < 18) { applyPowerup(p); p.dead = true; continue; }
   902→    }
   903→    if (p.t > POWER_LIFE) p.dead = true;  // اگر کسی نگرفت محو شود
   904→  }
   905→  state.powerups = state.powerups.filter(p => !p.dead);
   906→}

   908→/* اعمال اثر جعبه‌ی قدرت */
   909→function applyPowerup(p) {
   910→  const def = PTYPES[p.type];
   911→  SFX.power();
   912→  switch (p.type) {
   913→    case 'medkit': {
   914→      const heal = Math.min(def.heal, state.tower.maxHp - state.tower.hp);
   915→      state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + def.heal);
   916→      addText(TOWER.x, TOWER_TOP_Y - 44, '+' + faNum(Math.round(heal)) + ' ❤️', '#7be389', 15);
   917→      spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#7be389', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
   918→      break;
   919→    }
   920→    case 'rapid':
   921→      state.powerTimers.rapid = def.dur;
   922→      addText(TOWER.x, TOWER_TOP_Y - 44, 'شلیک سریع ⚡', '#ffd873', 15);
   923→      spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#ffd873', { speed: 160, life: 0.5, size: 2.4, grav: 250, up: true });
   924→      break;
   925→    case 'freeze':
   926→      state.powerTimers.freeze = def.dur;
   927→      SFX.freeze();
   928→      addText(TOWER.x, TOWER_TOP_Y - 44, 'انجماد ❄', '#8fd6ff', 15);
   929→      spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 18, '#bfe6ff', { speed: 180, life: 0.7, size: 2.4, grav: 120 });
   930→      break;
   931→    case 'coin': {
   932→      state.coins += def.bonus;
   933→      state.stats.earned += def.bonus;
   934→      for (const c of state.coinsArr) c.phase = 'fly';   // همه‌ی سکه‌های زمین هم جمع شود
   935→      addText(TOWER.x, TOWER_TOP_Y - 44, '+' + faNum(def.bonus) + ' 💰', '#f5c542', 15);
   936→      pulseCoins();
   937→      break;
   938→    }
   939→  }
   940→  updateHUD();
   941→}

   971→/* خرید ارتقا — در ۳٫۰ وسط مبارزه هم مجاز است (پنل زنده‌ی کنار نقشه) */
   972→function attemptBuy(kind) {
   973→  if (!canPlay()) return;                  // فقط وقتی بازی جریان دارد
   974→  const u = UPGRADES[kind];
   975→
   976→  /* --- سلاح‌های یک‌بار خرید (شاتگان/مسلسل) --- */
   977→  if (kind === 'shotgun' || kind === 'mg') {
   978→    if (WEAPONS[kind].owned) return;
   979→    if (state.coins < u.price) { denyBPItem(kind); return; }
   980→    state.coins -= u.price;
   981→    WEAPONS[kind].owned = true;
   982→    selectWeapon(kind);                    // بلافاصله دستِ کاربر می‌افتد
   983→    SFX.buy();
   984→    spawnParticles(TOWER.x, TOWER_TOP_Y - 20, 18, '#ffd873', { speed: 170, life: 0.6, size: 2.6, grav: 300, up: true });
   985→    updateHUD(); refreshBuildPanel();
   986→    return;
   987→  }
   988→
   989→  /* --- تعمیر سریع --- */
   990→  if (kind === 'repair') {
   991→    if (state.tower.hp >= state.tower.maxHp - 0.5) return;
   992→    if (state.coins < u.price) { denyBPItem(kind); return; }
   993→    state.coins -= u.price;
   994→    const heal = Math.round(state.tower.maxHp * u.frac);
   995→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + heal);
   996→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(heal), '#6fce62', 16);
   997→    spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#6fce62', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
   998→    SFX.buy();
   999→    updateHUD(); refreshBuildPanel();
  1000→    return;
  1001→  }
  1002→
  1003→  /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله) --- */
  1004→  const lvl = state.upgrades[kind];
  1005→  if (lvl >= u.max) return;
  1006→  const price = priceOf(kind);
  1007→  if (state.coins < price) { denyBPItem(kind); return; }
  1008→  state.coins -= price;
  1009→  state.upgrades[kind] = lvl + 1;
  1010→  if (kind === 'health') {
  1011→    /* هم حداکثر جون زیاد می‌شود هم همین الان ترمیم می‌شود */
  1012→    state.tower.maxHp += u.bonus;
  1013→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + u.bonus);
  1014→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(u.bonus), '#6fce62', 15);
  1015→  }
  1016→  if (kind === 'trap' && lvl === 0) {
  1017→    /* نصب اول تله: جلوه بصری نصب */
  1018→    spawnParticles(TOWER.x, GROUND_Y - 8, 16, '#ffcf6e', { speed: 150, life: 0.5, size: 2.5, grav: 300, up: true });
  1019→    state.shake = Math.min(8, state.shake + 4);
  1020→  }
  1021→  SFX.buy();
  1022→  updateHUD(); refreshBuildPanel();
  1023→}
