   203→/* ---------------------- ۳) موتور صدا (Web Audio) ---------------------- */
   204→/* همه‌ی صداها همین‌جا سینتز می‌شوند — هیچ فایل صوتی خارجی وجود ندارد.
   205→   AudioContext فقط بعد از اولین تعامل کاربر ساخته می‌شود (سیاست مرورگرها). */
   206→const Snd = {
   207→  ctx: null, master: null, noiseBuf: null, muted: false, ready: false,
   208→
   209→  init() {
   210→    if (this.ready) return;
   211→    try {
   212→      const AC = window.AudioContext || window.webkitAudioContext;
   213→      if (!AC) return;
   214→      this.ctx = new AC();
   215→      this.master = this.ctx.createGain();
   216→      this.master.gain.value = this.muted ? 0 : 0.5;
   217→      this.master.connect(this.ctx.destination);
   218→      /* بافر نویز سفید یک‌ثانیه‌ای (برای شلیک/برخورد/باد/رعد) */
   219→      this.noiseBuf = this.ctx.createBuffer(1, this.ctx.sampleRate, this.ctx.sampleRate);
   220→      const d = this.noiseBuf.getChannelData(0);
   221→      for (let j = 0; j < d.length; j++) d[j] = Math.random() * 2 - 1;
   222→      this.ready = true;
   223→      this.startAmbient();
   224→    } catch (e) { /* مرورگر بدون صدا */ }
   225→  },
   226→  resume() { if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
   227→  toggleMute() {
   228→    this.muted = !this.muted;
   229→    if (this.master) this.master.gain.value = this.muted ? 0 : 0.5;
   230→    return this.muted;
   231→  },
   232→
   233→  /* یک نُت: {type,f0,f1,dur,vol,att,delay} */
   234→  tone(o) {
   235→    if (!this.ready || this.muted) return;
   236→    try {
   237→      const t0 = this.ctx.currentTime + (o.delay || 0);
   238→      const osc = this.ctx.createOscillator(), g = this.ctx.createGain();
   239→      osc.type = o.type || 'sine';
   240→      osc.frequency.setValueAtTime(Math.max(1, o.f0), t0);
   241→      if (o.f1) osc.frequency.exponentialRampToValueAtTime(Math.max(1, o.f1), t0 + o.dur);
   242→      g.gain.setValueAtTime(0.0001, t0);
   243→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.008));
   244→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   245→      osc.connect(g); g.connect(this.master);
   246→      osc.start(t0); osc.stop(t0 + o.dur + 0.05);
   247→    } catch (e) {}
   248→  },
   249→  /* نویز فیلترشده: {dur,vol,ftype,freq,q,att,delay} */
   250→  noise(o) {
   251→    if (!this.ready || this.muted) return;
   252→    try {
   253→      const t0 = this.ctx.currentTime + (o.delay || 0);
   254→      const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf; src.loop = true;
   255→      const f = this.ctx.createBiquadFilter();
   256→      f.type = o.ftype || 'lowpass'; f.frequency.value = o.freq || 1000; f.Q.value = o.q || 0.8;
   257→      const g = this.ctx.createGain();
   258→      g.gain.setValueAtTime(0.0001, t0);
   259→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.005));
   260→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   261→      src.connect(f); f.connect(g); g.connect(this.master);
   262→      src.start(t0); src.stop(t0 + o.dur + 0.05);
   263→    } catch (e) {}
   264→  },
   265→
   266→  /* موسیقی محیطیِ شب: درونِ بم + باد + نُت‌های پراکنده‌ی دل‌انگیز */
   267→  startAmbient() {
   268→    try {
   269→      const dg = this.ctx.createGain(); dg.gain.value = 0.045; dg.connect(this.master);
   270→      const o1 = this.ctx.createOscillator(); o1.type = 'triangle'; o1.frequency.value = 55;
   271→      const o2 = this.ctx.createOscillator(); o2.type = 'sine'; o2.frequency.value = 110;
   272→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.09;
   273→      const lg = this.ctx.createGain(); lg.gain.value = 0.02;
   274→      lfo.connect(lg); lg.connect(dg.gain);
   275→      o1.connect(dg); o2.connect(dg);
   276→      o1.start(); o2.start(); lfo.start();
   277→      /* صدای باد (نویز لوپِ بسیار بم) */
   278→      const wind = this.ctx.createBufferSource(); wind.buffer = this.noiseBuf; wind.loop = true;

   549→function startWave(n) {
   550→  state.wave = n;
   551→  state.mode = 'playing';
   552→  const c = WAVES[n - 1];
   553→  const q = [];
   554→  for (let i = 0; i < c.n; i++) q.push('normal');
   555→  for (let i = 0; i < c.f; i++) q.push('fast');
   556→  for (let i = 0; i < c.t; i++) q.push('tank');
   557→  for (let i = q.length - 1; i > 0; i--) {          // شافل فیشر–ییتس
   558→    const j = Math.floor(rand(0, i + 1));
   559→    const tmp = q[i]; q[i] = q[j]; q[j] = tmp;
   560→  }
   561→  if (c.b) q.push('boss');                           // پادشاه همیشه آخرِ صف می‌ایستد
   562→  state.spawn = {
   563→    queue: q,
   564→    timer: 1.2,
   565→    interval: Math.max(2.0 - n * 0.14, 0.5),        // فاصله بین تولدها (کم‌تر هر موج)
   566→    bossDelay: false,

   663→function tryShoot() {
   664→  if (fireTimer > 0) return;
   665→  const w = WEAPONS[state.weapon];
   666→  /* قدرت «شلیک سریع» نرخ شلیک را تقریباً دوبرابر می‌کند */
   667→  fireTimer = w.rate * (state.powerTimers.rapid > 0 ? 0.55 : 1);
   668→  updateAim();                             // اطمینان از جهت صحیح گلوله
   669→  const m = gunMuzzle(w.barrel);
   670→  for (let i = 0; i < w.pellets; i++) {
   671→    const a = state.aimAngle + (Math.random() - 0.5) * 2 * w.spread;
   672→    const sp = w.spd * rand(0.92, 1.08);
   673→    state.bullets.push({
   674→      x: m.x, y: m.y,
   675→      vx: Math.cos(a) * sp, vy: Math.sin(a) * sp,
   676→      dmg: weaponDmg(w), r: w.pellets > 1 ? 3 : 4, life: w.ttl,
   677→      knock: w.knock,                      // پس‌زنی سلاح (۳٫۰)
   678→      dead: false, pellet: w.pellets > 1,  // ساچمه‌های شاتگان ریزترند
   679→    });
   680→  }
   681→  state.stats.shots++;
   682→  state.fireFlash = 0.07;
   683→  state.muzzleGlow = 0.07;                 // نور دهانه روی محیط (۳٫۰)
   684→  state.recoil = w.pellets > 1 ? 1.6 : 1;  // لگد شاتکن محسوس‌تر است
   685→  SFX[w.sfx]();
   686→  spawnParticles(m.x, m.y, 3, '#ffe9a8', { speed: 90, life: 0.18, size: 2, grav: 0,
   687→    dir: state.aimAngle, spread: Math.PI / 3 });
   688→  spawnShell(state.aimAngle);              // پوکه‌ی پرتاب‌شده
   689→  if (w.pellets > 1) state.shake = Math.min(8, state.shake + 2);
   690→}

   692→/* پوکه‌ی براقی که بعد از هر شلیک به عقب پرتاب می‌شود و روی بام می‌غلتد */
   693→function spawnShell(ang) {
   694→  const gx = TOWER.x + Math.cos(ang) * 8, gy = TOWER_TOP_Y - 17 + Math.sin(ang) * 8;
   695→  const a = ang + rand(1.6, 2.2);          // زاویه‌ی پرتاب به عقب/بالا
   696→  state.particles.push({
   697→    x: gx, y: gy,
   698→    vx: Math.cos(a) * rand(60, 130), vy: Math.sin(a) * rand(60, 130) - 70,
   699→    age: 0, life: rand(0.5, 0.8), size: 2.6, color: '#d9a441', grav: 900,
   700→    shell: true, rot: rand(0, 6.28), vr: rand(-9, 9),
   701→  });
   702→}

   704→/* برخورد گلوله با زامبی‌ها + خروج گلوله از صحنه
   705→   (۳٫۰: اول دایره‌ی سر بررسی می‌شود — برخورد سر = ضربه‌ی بحرانی ×۲) */
   706→function updateBullets(dt) {
   707→  for (const b of state.bullets) {
   708→    if (b.dead) continue;
   709→    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
   710→    /* رد برفیِ گلوله‌ی یخی برجک (۴٫۰) */
   711→    if (b.frost && Math.random() < dt * 34) {
   712→      state.particles.push({ x: b.x, y: b.y, vx: rand(-12, 12), vy: rand(-12, 12),
   713→        age: 0, life: 0.3, size: 1.3, color: 'rgba(205,240,255,0.9)', grav: 30 });
   714→    }
   715→    if (b.y > GROUND_Y + 4) {              // گلوله به زمین خورد
   716→      b.dead = true;
   717→      spawnParticles(b.x, GROUND_Y, 4, '#6a5a43', { speed: 70, life: 0.3, size: 2, grav: 300, up: true });
   718→      continue;
   719→    }
   720→    if (b.x < -40 || b.x > W + 40 || b.y < -40 || b.life <= 0) { b.dead = true; continue; }
   721→    const dir = Math.sign(b.vx) || 1;      // جهت پس‌زنی = جهت حرکت گلوله
   722→    const src = b.tur || 'player';         // منبع گلوله: بازیکن یا برجک (۴٫۰)
   723→    for (const z of state.zombies) {
   724→      if (z.dead) continue;
   725→      const s = (z.radius / 15) * (z.depth || 1);   // مرکز بدن بر اساس اندازه‌ی نوع و عمق
   726→      /* --- سر: ضربه‌ی بحرانی --- */
   727→      const hx = z.x + z.dir * 3 * s, hy = z.y - 36 * s, hr = 7 * s + b.r;
   728→      if ((b.x - hx) * (b.x - hx) + (b.y - hy) * (b.y - hy) < hr * hr) {
   729→        b.dead = true;
   730→        hitZombie(z, b.dmg, b.x, b.y, dir, true, b.knock, src);
   731→        if (b.frost) chillZombie(z);
   732→        break;
   733→      }
   734→      /* --- بدن --- */
   735→      const dx = z.x - b.x, dy = (z.y - 20 * s) - b.y;
   736→      const rr2 = z.radius * (z.depth || 1) + b.r + 3;
   737→      if (dx * dx + dy * dy < rr2 * rr2) {
   738→        b.dead = true;
   739→        hitZombie(z, b.dmg, b.x, b.y, dir, !!b.crit, b.knock, src);
   740→        if (b.frost) chillZombie(z);
   741→        break;
   742→      }
   743→    }
   744→  }
   745→  state.bullets = state.bullets.filter(b => !b.dead);
   746→}
   747→
   748→/* ---------------------- ۸) زامبی‌ها: محاصره‌ی دسته‌جمعی برج ----------------------
   749→   در ۳٫۰ زامبی‌ها دیگر در یک خط نمی‌ایستند: هر طرفِ برج تا ۵ ردیف
   750→   «دیوارِ زامبی» می‌سازند؛ نزدیک‌ترین مهاجم ردیفِ چسبیده به دیوار را می‌گیرد و
   751→   بقیه پشت سرش ردیف می‌شوند. ردیف‌های عقب بالاتر و کوچک‌تر رسم می‌شوند و
   752→   با کشته‌شدنِ ردیف جلو، بقیه فشرده شده و جلو می‌خزند. */
   753→function updateZombies(dt) {
   754→  /* رتبه‌بندی مهاجم‌های فعلی بر اساس نزدیکی به دیوار (هر طرف جداگانه):
   755→     نزدیک‌ترین = ردیف ۰؛ با مرگ ردیف جلو، رتبه‌ها فشرده می‌شوند */
   756→  const rank = { '-1': 0, '1': 0 };
   757→  const attackers = state.zombies.filter(z => !z.dead && z.attacking);
   758→  attackers.sort((a, b) => Math.abs(TOWER.x - a.x) - Math.abs(TOWER.x - b.x));
   759→  for (const z of attackers) {
   760→    z.attackSide = z.x < TOWER.x ? -1 : 1;
   761→    z.lane = Math.min(rank[z.attackSide]++, 4);
   762→  }
   763→
   764→  for (const z of state.zombies) {
   765→    if (z.dead) { z.deathT += dt; continue; }   // انیمیشن مرگ ادامه پیدا کند
   766→    z.hitFlash = Math.max(0, z.hitFlash - dt);
   767→    z.stagger = Math.max(0, z.stagger - dt);    // سرخوردن بعد از ضربه
   768→    z.chill = Math.max(0, (z.chill || 0) - dt); // یخ‌زدگی برجک یخ‌زن (۴٫۰)
   769→
   770→    const dx = TOWER.x - z.x;
   771→    const side = z.x < TOWER.x ? -1 : 1;        // کدام دیوار برج؟
   772→    const lane = z.attacking ? z.lane : Math.min(rank[side], 4);
   773→    const stop = TOWER.baseHalf + z.radius * 0.55 + lane * 15;
   774→
   775→    /* انجماد: همه کُند می‌شوند (پادشاه مقاوم‌تر است) — یخ‌زدگی برجک هم جداگانه اثر می‌کند */
   776→    const slow = state.powerTimers.freeze > 0 ? (z.type === 'boss' ? 0.6 : 0.32) : 1;
   777→    const spd = z.speed * slow * (z.chill > 0 ? 0.55 : 1) * (z.stagger > 0 ? 0.25 : 1);
   778→
   779→    if (Math.abs(dx) > stop) {
   780→      /* راه رفتن به سمت برج (مهاجمِ پس‌زده‌شده هم برمی‌گردد) */
   781→      z.attacking = false;
   782→      z.lane = Math.min(rank[side], 4);
   783→      z.x += Math.sign(dx) * spd * dt;
   784→      z.walkPhase += dt * (4 + spd * 0.06);
   785→    } else {
   786→      /* رسید به دیوار برج: کوبیدن پایه با فاصله‌ی زمانیِ نوعِ خودش */
   787→      z.attacking = true;
   788→      z.attackSide = side;
   789→      z.attackAnim += dt * 5;
   790→      z.biteTimer -= dt;
   791→      if (z.biteTimer <= 0) {
   792→        z.biteTimer = z.biteInt;
   793→        damageTower(z.dmg, z.x);
   794→      }
   795→    }
   796→
   797→    /* عمق تصویری: ردیف‌های عقب‌تر کمی بالاتر ایستاده و کوچک‌تر دیده می‌شوند */
   798→    const yTarget = GROUND_Y - z.lane * 9 + z.yJit;
   799→    z.y += (yTarget - z.y) * Math.min(1, dt * 6);
   800→    z.depth = 1 - z.lane * 0.07;
   801→
   802→    /* غبار گام‌ها (۶٫۰): ذره‌ی ریز پشت پای زامبیِ در حال راه رفتن */
   803→    if (!z.dead && Math.abs(dx) > stop && Math.random() < dt * 2.2) {
   804→      state.particles.push({
   805→        x: z.x - z.dir * 7, y: z.y + 1,
   806→        vx: -z.dir * rand(5, 15), vy: rand(-20, -7),
   807→        age: 0, life: rand(0.25, 0.42), size: rand(1.2, 1.9),
   808→        color: 'rgba(96,84,58,0.45)', grav: 60,
   809→      });
   810→    }
   811→  }
   812→  state.zombies = state.zombies.filter(z => !z.dead || z.deathT < 0.55);
   813→}
   814→
   815→/* آسیب گلوله به زامبی (۳٫۰: کریت ×۲ + پس‌زنی + سرخوردن + نشانگر برخورد) */
   816→function hitZombie(z, dmg, bx, by, dir, crit, knock, src) {
   817→  if (z.dead) return;
   818→  if (src) z.killedBy = src;                 // منبع ضربه — برای آمار کشتار برجک‌ها (۴٫۰)
   819→  const total = crit ? dmg * 2 : dmg;
   820→  z.hp -= total;
   821→  z.hitFlash = 0.15;
   822→  z.stagger = 0.15;                         // چند لحظه از می‌افتد
   823→  z.x += dir * (knock || 0) * (crit ? 1.7 : 1);   // پس‌زنی فیزیکی
   824→  if (crit) {
   825→    addText(z.x, z.y - 52, faNum(total), '#ffb347', 16);
   826→    spawnParticles(bx, by, 5, '#ffd873', { speed: 150, life: 0.35, size: 2.2, grav: 300 });
   827→  } else {
   828→    addText(z.x, z.y - 46, faNum(total), '#ffffff', 12);
   829→  }
   830→  spawnParticles(bx, by, 3, '#8fd06a', { speed: 110, life: 0.3, size: 2, grav: 400 });
   831→  spawnParticles(bx, by, 3, '#7a2f2f', { speed: 90,  life: 0.35, size: 2, grav: 500 });
   832→  if (Math.random() < 0.3) addDecal(z.x, false);   // چکه‌ی خون روی زمین
   833→  state.hitMark = 0.14; state.hitMarkCrit = !!crit; // نشانگر برخورد روی نشان
   834→  if (crit) SFX.crit(); else SFX.hit();
   835→  if (z.hp <= 0) killZombie(z);
   836→}

   838→/* مرگ زامبی: انیمیشن افتادن + خون + سکه + روح + جعبه‌ی قدرت + کمبو (۳٫۰) */
   839→function killZombie(z) {
   840→  if (z.dead) return;
   841→  z.dead = true; z.deathT = 0;
   842→  state.stats.kills++;
   843→  if (z.killedBy && z.killedBy !== 'player') state.stats.turKills++;   // کشتار دفاع‌های خودکار (۴٫۰)
   844→  spawnParticles(z.x, z.y - 18, 12, '#5f8f3e', { speed: 170, life: 0.55, size: 3, grav: 500 });
   845→  spawnParticles(z.x, z.y - 18, 8, '#7a2f2f', { speed: 130, life: 0.5, size: 2.6, grav: 500 });
   846→  addDecal(z.x, true);                     // لکه‌ی خون بزرگ زیر جسد
   847→  /* روحِ سبزِ کوچکی که آرام به آسمان می‌رود */
   848→  state.particles.push({
   849→    x: z.x, y: z.y - 26, vx: rand(-6, 6), vy: -38,
   850→    age: 0, life: 1.3, size: 3, color: 'rgba(190,255,200,0.5)', grav: -26, glow: true,
   851→  });
   852→  SFX.death();
   853→  dropCoins(z);
   854→  maybeDropPowerup(z);
   855→  bumpCombo();
   856→  updateHUD();
   857→  if (z.type === 'boss') {
   858→    state.shake = 16; state.flash = 0.8;
   859→    state.slowmo = 0.9;                    // لحظه‌ی سینمایی مرگ پادشاه
   860→    SFX.slowmo();
   861→    spawnParticles(z.x, z.y - 60, 40, '#ffd873', { speed: 300, life: 1.0, size: 3.5, grav: 400 });
   862→    spawnParticles(z.x, z.y - 60, 24, '#e8563f', { speed: 240, life: 0.8, size: 3, grav: 400 });
   863→    setAnnounce('👑 پادشاه کشته شد!', 'برج پابرجایی!', 2.2);
   864→    SFX.bossDie();
   865→  }
   866→}

   868→/* سیستم کمبو (۳٫۰): هر کشتنِ تازه، پنجره را باز نگه می‌دارد؛
   869→   در رکوردهای مشخص پیام + سکه‌ی جایزه می‌گیری. */
   870→function bumpCombo() {
   871→  state.combo++;
   872→  state.comboT = COMBO_WINDOW;
   873→  state.comboPop = 0.25;
   874→  state.stats.maxCombo = Math.max(state.stats.maxCombo, state.combo);
   875→  const m = COMBO_MILESTONES[state.combo];
   876→  if (m) {

   996→function updateTurrets(dt) {
   997→  for (const t of state.turrets) {
   998→    const def = TURRETS[t.type];
   999→    t.cd = Math.max(0, t.cd - dt);
  1000→    t.flash = Math.max(0, t.flash - dt);
  1001→    t.recoil = Math.max(0, t.recoil - dt * 5);
  1002→    t.hover = !state.placing && Math.abs(state.mouse.x - t.x) < 26 && Math.abs(state.mouse.y - (t.y - 14)) < 30;
  1003→
  1004→    /* نزدیک‌ترین زامبیِ زنده در شعاع برد */
  1005→    let best = null, bd = Infinity;
  1006→    for (const z of state.zombies) {
  1007→      if (z.dead) continue;
  1008→      const d = Math.abs(z.x - t.x);
  1009→      if (d <= def.range && d < bd) { bd = d; best = z; }
  1010→    }
  1011→    if (best) {
  1012→      const s = (best.radius / 15) * (best.depth || 1);
  1013→      const want = Math.atan2((best.y - 20 * s) - (t.y - def.h), best.x - t.x);
  1014→      let diff = want - t.aim;
  1015→      while (diff > Math.PI) diff -= Math.PI * 2;
  1016→      while (diff < -Math.PI) diff += Math.PI * 2;
  1017→      t.aim += diff * Math.min(1, dt * 8);   // چرخش نرم لوله به سمت هدف
  1018→      if (t.cd <= 0 && Math.abs(diff) < 0.35) turretFire(t);
  1019→    }
  1020→    if (t.type === 'gatling') t.spin += dt * (best ? 16 : 3);   // چرخش لوله‌های مسلسل
  1021→    /* برفک‌های محیطی برجک یخ‌زن */
  1022→    if (t.type === 'frost' && Math.random() < dt * 2.5) {
  1023→      state.particles.push({ x: t.x + rand(-13, 13), y: t.y - rand(14, 30),
  1024→        vx: rand(-6, 6), vy: rand(8, 20), age: 0, life: rand(0.6, 1.2),

  1227→      break;
  1228→    }
  1229→  }
  1230→  updateHUD();
  1231→}

  1233→/* ---------------------- ۱۰) اقتصاد و ارتقاها (ساخت‌وساز زنده) ---------------------- */
  1234→/* قیمت ارتقای kind در سطح فعلی (گردشده به مضرب ۵) */
  1235→function priceOf(kind) {
  1236→  const u = UPGRADES[kind], lvl = state.upgrades[kind];
  1237→  return Math.round(u.base * Math.pow(u.growth, lvl) / 5) * 5;
  1238→}

  1240→/* لرزش آیتم پنل + صدای «نه» وقتی سکه کافی نیست */
  1241→function denyBPItem(kind) {
  1242→  const item = $('bp-' + kind);
  1243→  if (item) { item.classList.remove('deny'); void item.offsetWidth; item.classList.add('deny'); }
  1244→  SFX.deny();

  1247→/* انتخاب سلاح (کلید ۱/۲/۳ یا کلیک روی نوار سلاح) */
  1248→function selectWeapon(id) {
  1249→  if (!WEAPONS[id].owned) {
  1250→    SFX.deny();
  1251→    const el = document.querySelector('#weaponBar .wslot[data-w="' + id + '"]');
  1252→    if (el) { el.classList.remove('deny'); void el.offsetWidth; el.classList.add('deny'); }
  1253→    return;
  1254→  }
  1255→  if (state.weapon === id) return;
  1256→  state.weapon = id;
  1257→  SFX.click();
  1258→  refreshWeaponBar();
  1259→}
  1260→
  1261→/* خرید ارتقا — در ۳٫۰ وسط مبارزه هم مجاز است (پنل زنده‌ی کنار نقشه) */
  1262→function attemptBuy(kind) {
  1263→  if (!canPlay()) return;                  // فقط وقتی بازی جریان دارد
  1264→  const u = UPGRADES[kind];
  1265→
  1266→  /* --- برجک‌های دفاعی: به حالت انتخاب محل نصب می‌رود (۴٫۰) --- */
  1267→  if (kind === 'gatling' || kind === 'sniper' || kind === 'frost') {
  1268→    startPlacement(kind);
  1269→    return;
  1270→  }
  1271→
  1272→  /* --- سلاح‌های یک‌بار خرید (شاتگان/مسلسل) --- */
  1273→  if (kind === 'shotgun' || kind === 'mg') {
  1274→    if (WEAPONS[kind].owned) return;
  1275→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1276→    state.coins -= u.price;
  1277→    WEAPONS[kind].owned = true;
  1278→    selectWeapon(kind);                    // بلافاصله دستِ کاربر می‌افتد
  1279→    SFX.buy();
  1280→    spawnParticles(TOWER.x, TOWER_TOP_Y - 20, 18, '#ffd873', { speed: 170, life: 0.6, size: 2.6, grav: 300, up: true });
  1281→    updateHUD(); refreshBuildPanel();
  1282→    return;
  1283→  }
  1284→
  1285→  /* --- تعمیر سریع --- */
  1286→  if (kind === 'repair') {
  1287→    if (state.tower.hp >= state.tower.maxHp - 0.5) return;
  1288→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1289→    state.coins -= u.price;
  1290→    const heal = Math.round(state.tower.maxHp * u.frac);
  1291→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + heal);
  1292→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(heal), '#6fce62', 16);
  1293→    spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#6fce62', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
  1294→    SFX.buy();
  1295→    updateHUD(); refreshBuildPanel();
  1296→    return;
  1297→  }
  1298→
  1299→  /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله) --- */
  1300→  const lvl = state.upgrades[kind];
  1301→  if (lvl >= u.max) return;
  1302→  const price = priceOf(kind);
  1303→  if (state.coins < price) { denyBPItem(kind); return; }
  1304→  state.coins -= price;
  1305→  state.upgrades[kind] = lvl + 1;
  1306→  if (kind === 'health') {
  1307→    /* هم حداکثر جون زیاد می‌شود هم همین الان ترمیم می‌شود */
  1308→    state.tower.maxHp += u.bonus;
  1309→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + u.bonus);
  1310→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(u.bonus), '#6fce62', 15);
  1311→  }
  1312→  if (kind === 'turretUp') {
  1313→    /* جلوه‌ی ارتقای برجک‌ها: جرقه‌ی طلایی روی همه‌ی برجک‌های نصب‌شده (۴٫۰) */
  1314→    for (const t of state.turrets) spawnParticles(t.x, t.y - 18, 10, '#ffd873',
  1315→      { speed: 130, life: 0.5, size: 2.2, grav: 250, up: true });
  1316→    if (state.turrets.length) addText(TOWER.x, TOWER_TOP_Y - 40, 'برجک‌ها قوی‌تر شدند! ⬆', '#ffd873', 14);
  1317→  }
  1318→  if (kind === 'trap' && lvl === 0) {
  1319→    /* نصب اول تله: جلوه بصری نصب */
  1320→    spawnParticles(TOWER.x, GROUND_Y - 8, 16, '#ffcf6e', { speed: 150, life: 0.5, size: 2.5, grav: 300, up: true });
  1321→    state.shake = Math.min(8, state.shake + 4);
  1322→  }
  1323→  SFX.buy();
  1324→  updateHUD(); refreshBuildPanel();
  1325→}
