   178→/* همه‌ی صداها همین‌جا سینتز می‌شوند — هیچ فایل صوتی خارجی وجود ندارد.
   179→   AudioContext فقط بعد از اولین تعامل کاربر ساخته می‌شود (سیاست مرورگرها). */
   180→const Snd = {
   181→  ctx: null, master: null, noiseBuf: null, muted: false, ready: false,
   182→
   183→  init() {
   184→    if (this.ready) return;
   185→    try {
   186→      const AC = window.AudioContext || window.webkitAudioContext;
   187→      if (!AC) return;
   188→      this.ctx = new AC();
   189→      this.master = this.ctx.createGain();
   190→      this.master.gain.value = this.muted ? 0 : 0.5;
   191→      this.master.connect(this.ctx.destination);
   192→      /* بافر نویز سفید یک‌ثانیه‌ای (برای شلیک/برخورد/باد/رعد) */
   193→      this.noiseBuf = this.ctx.createBuffer(1, this.ctx.sampleRate, this.ctx.sampleRate);
   194→      const d = this.noiseBuf.getChannelData(0);
   195→      for (let j = 0; j < d.length; j++) d[j] = Math.random() * 2 - 1;
   196→      this.ready = true;
   197→      this.startAmbient();
   198→    } catch (e) { /* مرورگر بدون صدا */ }
   199→  },
   200→  resume() { if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
   201→  toggleMute() {
   202→    this.muted = !this.muted;
   203→    if (this.master) this.master.gain.value = this.muted ? 0 : 0.5;
   204→    return this.muted;
   205→  },
   206→
   207→  /* یک نُت: {type,f0,f1,dur,vol,att,delay} */
   208→  tone(o) {
   209→    if (!this.ready || this.muted) return;
   210→    try {
   211→      const t0 = this.ctx.currentTime + (o.delay || 0);
   212→      const osc = this.ctx.createOscillator(), g = this.ctx.createGain();
   213→      osc.type = o.type || 'sine';
   214→      osc.frequency.setValueAtTime(Math.max(1, o.f0), t0);
   215→      if (o.f1) osc.frequency.exponentialRampToValueAtTime(Math.max(1, o.f1), t0 + o.dur);
   216→      g.gain.setValueAtTime(0.0001, t0);
   217→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.008));
   218→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   219→      osc.connect(g); g.connect(this.master);
   220→      osc.start(t0); osc.stop(t0 + o.dur + 0.05);
   221→    } catch (e) {}
   222→  },
   223→  /* نویز فیلترشده: {dur,vol,ftype,freq,q,att,delay} */
   224→  noise(o) {
   225→    if (!this.ready || this.muted) return;
   226→    try {
   227→      const t0 = this.ctx.currentTime + (o.delay || 0);
   228→      const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf; src.loop = true;
   229→      const f = this.ctx.createBiquadFilter();
   230→      f.type = o.ftype || 'lowpass'; f.frequency.value = o.freq || 1000; f.Q.value = o.q || 0.8;
   231→      const g = this.ctx.createGain();
   232→      g.gain.setValueAtTime(0.0001, t0);
   233→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.005));
   234→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   235→      src.connect(f); f.connect(g); g.connect(this.master);
   236→      src.start(t0); src.stop(t0 + o.dur + 0.05);
   237→    } catch (e) {}
   238→  },
   239→
   240→  /* موسیقی محیطیِ شب: درونِ بم + باد + نُت‌های پراکنده‌ی دل‌انگیز */
   241→  startAmbient() {
   242→    try {
   243→      const dg = this.ctx.createGain(); dg.gain.value = 0.045; dg.connect(this.master);
   244→      const o1 = this.ctx.createOscillator(); o1.type = 'triangle'; o1.frequency.value = 55;
   245→      const o2 = this.ctx.createOscillator(); o2.type = 'sine'; o2.frequency.value = 110;
   246→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.09;
   247→      const lg = this.ctx.createGain(); lg.gain.value = 0.02;
   248→      lfo.connect(lg); lg.connect(dg.gain);
   249→      o1.connect(dg); o2.connect(dg);
   250→      o1.start(); o2.start(); lfo.start();
   251→      /* صدای باد (نویز لوپِ بسیار بم) */
   252→      const wind = this.ctx.createBufferSource(); wind.buffer = this.noiseBuf; wind.loop = true;
   253→      const wf = this.ctx.createBiquadFilter(); wf.type = 'lowpass'; wf.frequency.value = 260;
   254→      const wg = this.ctx.createGain(); wg.gain.value = 0.02;
   255→      wind.connect(wf); wf.connect(wg); wg.connect(this.master); wind.start();
   256→      this.pluck();
   257→    } catch (e) {}
   258→  },

   521→/* آغاز موج n: صف تولد بر اساس ترکیب جدول WAVES ساخته و شافل می‌شود */
   522→function startWave(n) {
   523→  state.wave = n;
   524→  state.mode = 'playing';
   525→  const c = WAVES[n - 1];
   526→  const q = [];
   527→  for (let i = 0; i < c.n; i++) q.push('normal');
   528→  for (let i = 0; i < c.f; i++) q.push('fast');
   529→  for (let i = 0; i < c.t; i++) q.push('tank');
   530→  for (let i = q.length - 1; i > 0; i--) {          // شافل فیشر–ییتس
   531→    const j = Math.floor(rand(0, i + 1));
   532→    const tmp = q[i]; q[i] = q[j]; q[j] = tmp;
   533→  }
   534→  if (c.b) q.push('boss');                           // پادشاه همیشه آخرِ صف می‌ایستد
   535→  state.spawn = {
   536→    queue: q,
   537→    timer: 1.2,
   538→    interval: Math.max(2.0 - n * 0.14, 0.5),        // فاصله بین تولدها (کم‌تر هر موج)
   539→    bossDelay: false,
   540→  };
   541→  setAnnounce('موج ' + faNum(n),
   542→    n === TOTAL_WAVES ? 'موج آخر! پادشاه زامبی‌ها پشتِ کاره...' : 'دفاع کن! 🧟', 2);
   543→  SFX.wave();
   544→  $('bpHead').classList.remove('attention');
   545→  updateHUD();
   546→}

   550→function waveComplete() {
   551→  if (state.wave >= TOTAL_WAVES) { victory(); return; }
   552→  /* پادشایی موج: جایزه‌ی نقدی صبح (۱۲ + ۶×شماره موج) */

   554→  state.coins += bonus;
   555→  state.stats.earned += bonus;
   556→  addText(TOWER.x, TOWER_TOP_Y - 56, 'جایزه‌ی موج: +' + faNum(bonus) + ' 💰', '#ffd873', 16);
   557→  pulseCoins();
   558→  /* صحنه پاک شود و صبح طلوع کند */
   559→  state.zombies = []; state.bullets = [];
   560→  state.mode = 'day';
   561→  state.dayTarget = 1;                     // گذار نرم به روز
   562→  state.goingNight = false;

   565→  setAnnounce('☀️ صبح شد!', 'موج ' + faNum(state.wave) + ' پاک شد — ارتقا بخر و شب را شروع کن', 3);
   566→  SFX.clear();
   567→  SFX.morning();                           // چرپ‌چرپ صبحگاهی

   634→function tryShoot() {
   635→  if (fireTimer > 0) return;
   636→  const w = WEAPONS[state.weapon];
   637→  /* قدرت «شلیک سریع» نرخ شلیک را تقریباً دوبرابر می‌کند */
   638→  fireTimer = w.rate * (state.powerTimers.rapid > 0 ? 0.55 : 1);
   639→  updateAim();                             // اطمینان از جهت صحیح گلوله
   640→  const m = gunMuzzle(w.barrel);
   641→  for (let i = 0; i < w.pellets; i++) {
   642→    const a = state.aimAngle + (Math.random() - 0.5) * 2 * w.spread;
   643→    const sp = w.spd * rand(0.92, 1.08);
   644→    state.bullets.push({
   645→      x: m.x, y: m.y,
   646→      vx: Math.cos(a) * sp, vy: Math.sin(a) * sp,
   647→      dmg: weaponDmg(w), r: w.pellets > 1 ? 3 : 4, life: w.ttl,
   648→      knock: w.knock,                      // پس‌زنی سلاح (۳٫۰)
   649→      dead: false, pellet: w.pellets > 1,  // ساچمه‌های شاتگان ریزترند
   650→    });
   651→  }
   652→  state.stats.shots++;
   653→  state.fireFlash = 0.07;
   654→  state.muzzleGlow = 0.07;                 // نور دهانه روی محیط (۳٫۰)
   655→  state.recoil = w.pellets > 1 ? 1.6 : 1;  // لگد شاتکن محسوس‌تر است
   656→  SFX[w.sfx]();
   657→  spawnParticles(m.x, m.y, 3, '#ffe9a8', { speed: 90, life: 0.18, size: 2, grav: 0,
   658→    dir: state.aimAngle, spread: Math.PI / 3 });
   659→  spawnShell(state.aimAngle);              // پوکه‌ی پرتاب‌شده
   660→  if (w.pellets > 1) state.shake = Math.min(8, state.shake + 2);
   661→}

   663→/* پوکه‌ی براقی که بعد از هر شلیک به عقب پرتاب می‌شود و روی بام می‌غلتد */
   664→function spawnShell(ang) {
   665→  const gx = TOWER.x + Math.cos(ang) * 8, gy = TOWER_TOP_Y - 17 + Math.sin(ang) * 8;
   666→  const a = ang + rand(1.6, 2.2);          // زاویه‌ی پرتاب به عقب/بالا
   667→  state.particles.push({
   668→    x: gx, y: gy,
   669→    vx: Math.cos(a) * rand(60, 130), vy: Math.sin(a) * rand(60, 130) - 70,
   670→    age: 0, life: rand(0.5, 0.8), size: 2.6, color: '#d9a441', grav: 900,
   671→    shell: true, rot: rand(0, 6.28), vr: rand(-9, 9),
   672→  });

   675→/* برخورد گلوله با زامبی‌ها + خروج گلوله از صحنه
   676→   (۳٫۰: اول دایره‌ی سر بررسی می‌شود — برخورد سر = ضربه‌ی بحرانی ×۲) */
   677→function updateBullets(dt) {
   678→  for (const b of state.bullets) {
   679→    if (b.dead) continue;
   680→    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
   681→    /* رد برفیِ گلوله‌ی یخی برجک (۴٫۰) */
   682→    if (b.frost && Math.random() < dt * 34) {
   683→      state.particles.push({ x: b.x, y: b.y, vx: rand(-12, 12), vy: rand(-12, 12),
   684→        age: 0, life: 0.3, size: 1.3, color: 'rgba(205,240,255,0.9)', grav: 30 });
   685→    }
   686→    if (b.y > GROUND_Y + 4) {              // گلوله به زمین خورد
   687→      b.dead = true;
   688→      spawnParticles(b.x, GROUND_Y, 4, '#6a5a43', { speed: 70, life: 0.3, size: 2, grav: 300, up: true });
   689→      continue;
   690→    }
   691→    if (b.x < -40 || b.x > W + 40 || b.y < -40 || b.life <= 0) { b.dead = true; continue; }
   692→    const dir = Math.sign(b.vx) || 1;      // جهت پس‌زنی = جهت حرکت گلوله
   693→    const src = b.tur || 'player';         // منبع گلوله: بازیکن یا برجک (۴٫۰)
   694→    for (const z of state.zombies) {
   695→      if (z.dead) continue;
   696→      const s = (z.radius / 15) * (z.depth || 1);   // مرکز بدن بر اساس اندازه‌ی نوع و عمق
   697→      /* --- سر: ضربه‌ی بحرانی --- */
   698→      const hx = z.x + z.dir * 3 * s, hy = z.y - 36 * s, hr = 7 * s + b.r;
   699→      if ((b.x - hx) * (b.x - hx) + (b.y - hy) * (b.y - hy) < hr * hr) {
   700→        b.dead = true;
   701→        hitZombie(z, b.dmg, b.x, b.y, dir, true, b.knock, src);
   702→        if (b.frost) chillZombie(z);
   703→        break;
   704→      }
   705→      /* --- بدن --- */
   706→      const dx = z.x - b.x, dy = (z.y - 20 * s) - b.y;
   707→      const rr2 = z.radius * (z.depth || 1) + b.r + 3;
   708→      if (dx * dx + dy * dy < rr2 * rr2) {
   709→        b.dead = true;
   710→        hitZombie(z, b.dmg, b.x, b.y, dir, !!b.crit, b.knock, src);
   711→        if (b.frost) chillZombie(z);
   712→        break;
   713→      }
   714→    }
   715→  }
   716→  state.bullets = state.bullets.filter(b => !b.dead);
   717→}

   719→/* ---------------------- ۸) زامبی‌ها: محاصره‌ی دسته‌جمعی برج ----------------------
   720→   در ۳٫۰ زامبی‌ها دیگر در یک خط نمی‌ایستند: هر طرفِ برج تا ۵ ردیف
   721→   «دیوارِ زامبی» می‌سازند؛ نزدیک‌ترین مهاجم ردیفِ چسبیده به دیوار را می‌گیرد و
   722→   بقیه پشت سرش ردیف می‌شوند. ردیف‌های عقب بالاتر و کوچک‌تر رسم می‌شوند و
   723→   با کشته‌شدنِ ردیف جلو، بقیه فشرده شده و جلو می‌خزند. */
   724→function updateZombies(dt) {
   725→  /* رتبه‌بندی مهاجم‌های فعلی بر اساس نزدیکی به دیوار (هر طرف جداگانه):
   726→     نزدیک‌ترین = ردیف ۰؛ با مرگ ردیف جلو، رتبه‌ها فشرده می‌شوند */
   727→  const rank = { '-1': 0, '1': 0 };
   728→  const attackers = state.zombies.filter(z => !z.dead && z.attacking);
   729→  attackers.sort((a, b) => Math.abs(TOWER.x - a.x) - Math.abs(TOWER.x - b.x));
   730→  for (const z of attackers) {
   731→    z.attackSide = z.x < TOWER.x ? -1 : 1;
   732→    z.lane = Math.min(rank[z.attackSide]++, 4);
   733→  }
   734→
   735→  for (const z of state.zombies) {
   736→    if (z.dead) { z.deathT += dt; continue; }   // انیمیشن مرگ ادامه پیدا کند
   737→    z.hitFlash = Math.max(0, z.hitFlash - dt);
   738→    z.stagger = Math.max(0, z.stagger - dt);    // سرخوردن بعد از ضربه
   739→    z.chill = Math.max(0, (z.chill || 0) - dt); // یخ‌زدگی برجک یخ‌زن (۴٫۰)
   740→
   741→    const dx = TOWER.x - z.x;
   742→    const side = z.x < TOWER.x ? -1 : 1;        // کدام دیوار برج؟
   743→    const lane = z.attacking ? z.lane : Math.min(rank[side], 4);
   744→    const stop = TOWER.baseHalf + z.radius * 0.55 + lane * 15;
   745→
   746→    /* انجماد: همه کُند می‌شوند (پادشاه مقاوم‌تر است) — یخ‌زدگی برجک هم جداگانه اثر می‌کند */
   747→    const slow = state.powerTimers.freeze > 0 ? (z.type === 'boss' ? 0.6 : 0.32) : 1;
   748→    const spd = z.speed * slow * (z.chill > 0 ? 0.55 : 1) * (z.stagger > 0 ? 0.25 : 1);
   749→
   750→    if (Math.abs(dx) > stop) {
   751→      /* راه رفتن به سمت برج (مهاجمِ پس‌زده‌شده هم برمی‌گردد) */
   752→      z.attacking = false;
   753→      z.lane = Math.min(rank[side], 4);
   754→      z.x += Math.sign(dx) * spd * dt;
   755→      z.walkPhase += dt * (4 + spd * 0.06);
   756→    } else {
   757→      /* رسید به دیوار برج: کوبیدن پایه با فاصله‌ی زمانیِ نوعِ خودش */
   758→      z.attacking = true;
   759→      z.attackSide = side;
   760→      z.attackAnim += dt * 5;
   761→      z.biteTimer -= dt;
   762→      if (z.biteTimer <= 0) {
   763→        z.biteTimer = z.biteInt;
   764→        damageTower(z.dmg, z.x);
   765→      }
   766→    }
   767→
   768→    /* عمق تصویری: ردیف‌های عقب‌تر کمی بالاتر ایستاده و کوچک‌تر دیده می‌شوند */
   769→    const yTarget = GROUND_Y - z.lane * 9 + z.yJit;
   770→    z.y += (yTarget - z.y) * Math.min(1, dt * 6);
   771→    z.depth = 1 - z.lane * 0.07;
   772→  }
   773→  state.zombies = state.zombies.filter(z => !z.dead || z.deathT < 0.55);
   774→}

   776→/* آسیب گلوله به زامبی (۳٫۰: کریت ×۲ + پس‌زنی + سرخوردن + نشانگر برخورد) */
   777→function hitZombie(z, dmg, bx, by, dir, crit, knock, src) {
   778→  if (z.dead) return;
   779→  if (src) z.killedBy = src;                 // منبع ضربه — برای آمار کشتار برجک‌ها (۴٫۰)
   780→  const total = crit ? dmg * 2 : dmg;
   781→  z.hp -= total;
   782→  z.hitFlash = 0.15;
   783→  z.stagger = 0.15;                         // چند لحظه از می‌افتد
   784→  z.x += dir * (knock || 0) * (crit ? 1.7 : 1);   // پس‌زنی فیزیکی
   785→  if (crit) {
   786→    addText(z.x, z.y - 52, faNum(total), '#ffb347', 16);
   787→    spawnParticles(bx, by, 5, '#ffd873', { speed: 150, life: 0.35, size: 2.2, grav: 300 });
   788→  } else {
   789→    addText(z.x, z.y - 46, faNum(total), '#ffffff', 12);
   790→  }
   791→  spawnParticles(bx, by, 3, '#8fd06a', { speed: 110, life: 0.3, size: 2, grav: 400 });
   792→  spawnParticles(bx, by, 3, '#7a2f2f', { speed: 90,  life: 0.35, size: 2, grav: 500 });
   793→  if (Math.random() < 0.3) addDecal(z.x, false);   // چکه‌ی خون روی زمین
   794→  state.hitMark = 0.14; state.hitMarkCrit = !!crit; // نشانگر برخورد روی نشان
   795→  if (crit) SFX.crit(); else SFX.hit();
   796→  if (z.hp <= 0) killZombie(z);
   797→}

   865→/* تله پایه برج: آسیب خودکار و پیوسته به زامبی‌های نزدیک */
   866→function updateTrap(dt) {
   867→  const lvl = state.upgrades.trap;
   868→  if (lvl <= 0) return;
   869→  const dps = UPGRADES.trap.dps + UPGRADES.trap.dpsPerLvl * (lvl - 1);
   870→  for (const z of state.zombies) {
   871→    if (z.dead) continue;
   872→    if (Math.abs(z.x - TOWER.x) < UPGRADES.trap.radius) {
   873→      z.hp -= dps * dt;
   874→      z.hitFlash = Math.max(z.hitFlash, 0.06);
   875→      if (Math.random() < dt * 6) { addZap(z.x, z.y - 18); SFX.trap(); }  // قوس برق نمایشی
   876→      if (z.hp <= 0) { z.killedBy = 'trap'; killZombie(z); }
   877→    }
   878→  }
   879→}

   881→/* ---------------------- ۸٫۵) برجک‌های دفاعی خودکار (۴٫۰) ----------------------
   882→   بازیکن برجک می‌خرد → شبحِ نصب دنبال ماوس می‌آید → کلیک روی زمین = نصب.
   883→   هر برجک نزدیک‌ترین زامبیِ در برد را نشانه می‌رود و خودکار شلیک می‌کند. */
   884→
   885→/* آیا نقطه‌ی (x,y) جای نصب مناسبی است؟ (نوار زمین، دوری از برج/برجک‌ها/مانع‌های صحنه) */
   886→function placementValid(x, y) {
   887→  if (y < GROUND_Y - 34 || y > GROUND_Y + 16) return false;    // فقط نوار زمین
   888→  if (x < 26 || x > W - 200) return false;                      // داخل کادر و دور از پنل ساخت‌وساز
   889→  if (Math.abs(x - TOWER.x) < TOWER.baseHalf + 34) return false; // چسبیده به پایه‌ی برج نه
   890→  for (const t of state.turrets) if (Math.hypot(t.x - x, t.y - y) < 36) return false;
   891→  for (const lx of LAMPS) if (Math.abs(x - lx) < 18) return false;
   892→  for (const tx of [88, 874]) if (Math.abs(x - tx) < 22) return false;   // درخت‌های خشک
   893→  for (const p of PUDDLES) if (Math.abs(x - p.x) < p.w / 2 + 8) return false;
   894→  return true;
   895→}

   956→/* حلقه‌ی برجک‌ها: هدف‌گیری نزدیک‌ترین زامبی در برد + چرخش لوله + شلیک خودکار */
   957→function updateTurrets(dt) {
   958→  for (const t of state.turrets) {
   959→    const def = TURRETS[t.type];
   960→    t.cd = Math.max(0, t.cd - dt);
   961→    t.flash = Math.max(0, t.flash - dt);
   962→    t.recoil = Math.max(0, t.recoil - dt * 5);
   963→    t.hover = !state.placing && Math.abs(state.mouse.x - t.x) < 26 && Math.abs(state.mouse.y - (t.y - 14)) < 30;
   964→
   965→    /* نزدیک‌ترین زامبیِ زنده در شعاع برد */
   966→    let best = null, bd = Infinity;
   967→    for (const z of state.zombies) {
   968→      if (z.dead) continue;
   969→      const d = Math.abs(z.x - t.x);
   970→      if (d <= def.range && d < bd) { bd = d; best = z; }
   971→    }
   972→    if (best) {
   973→      const s = (best.radius / 15) * (best.depth || 1);
   974→      const want = Math.atan2((best.y - 20 * s) - (t.y - def.h), best.x - t.x);
   975→      let diff = want - t.aim;
   976→      while (diff > Math.PI) diff -= Math.PI * 2;
   977→      while (diff < -Math.PI) diff += Math.PI * 2;
   978→      t.aim += diff * Math.min(1, dt * 8);   // چرخش نرم لوله به سمت هدف
   979→      if (t.cd <= 0 && Math.abs(diff) < 0.35) turretFire(t);
   980→    }
   981→    if (t.type === 'gatling') t.spin += dt * (best ? 16 : 3);   // چرخش لوله‌های مسلسل
   982→    /* برفک‌های محیطی برجک یخ‌زن */
   983→    if (t.type === 'frost' && Math.random() < dt * 2.5) {
   984→      state.particles.push({ x: t.x + rand(-13, 13), y: t.y - rand(14, 30),
   985→        vx: rand(-6, 6), vy: rand(8, 20), age: 0, life: rand(0.6, 1.2),

  1052→function updateCoins(dt) {

  1054→  for (const c of state.coinsArr) {
  1055→    if (c.dead) continue;

  1066→    } else {
  1067→      /* پرو شتاب‌دار به سمت شمارنده سکه در HUD */

  1069→      const d = Math.hypot(dx, dy) || 1;

  1071→      const step = Math.min(d, c.flySpeed * dt);

  1077→      state.particles.push({ x: c.x + rand(-5, 5), y: c.y + rand(-5, 5), vx: 0, vy: -14,
  1078→        age: 0, life: 0.35, size: 1.2, color: '#ffe98a', grav: 0 });

  1080→    if (c.t > 4) collectCoin(c); // بیمه: سکه گم‌نشده جمع شود
  1081→  }

  1223→function attemptBuy(kind) {
  1224→  if (!canPlay()) return;                  // فقط وقتی بازی جریان دارد
  1225→  const u = UPGRADES[kind];
  1226→
  1227→  /* --- برجک‌های دفاعی: به حالت انتخاب محل نصب می‌رود (۴٫۰) --- */
  1228→  if (kind === 'gatling' || kind === 'sniper' || kind === 'frost') {
  1229→    startPlacement(kind);
  1230→    return;
  1231→  }
  1232→
  1233→  /* --- سلاح‌های یک‌بار خرید (شاتگان/مسلسل) --- */
  1234→  if (kind === 'shotgun' || kind === 'mg') {
  1235→    if (WEAPONS[kind].owned) return;
  1236→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1237→    state.coins -= u.price;
  1238→    WEAPONS[kind].owned = true;
  1239→    selectWeapon(kind);                    // بلافاصله دستِ کاربر می‌افتد
  1240→    SFX.buy();
  1241→    spawnParticles(TOWER.x, TOWER_TOP_Y - 20, 18, '#ffd873', { speed: 170, life: 0.6, size: 2.6, grav: 300, up: true });
  1242→    updateHUD(); refreshBuildPanel();
  1243→    return;
  1244→  }
  1245→
  1246→  /* --- تعمیر سریع --- */
  1247→  if (kind === 'repair') {
  1248→    if (state.tower.hp >= state.tower.maxHp - 0.5) return;
  1249→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1250→    state.coins -= u.price;
  1251→    const heal = Math.round(state.tower.maxHp * u.frac);
  1252→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + heal);

  1258→  }

  1260→  /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله) --- */
  1261→  const lvl = state.upgrades[kind];
  1262→  if (lvl >= u.max) return;
  1263→  const price = priceOf(kind);
  1264→  if (state.coins < price) { denyBPItem(kind); return; }
  1265→  state.coins -= price;
  1266→  state.upgrades[kind] = lvl + 1;

  1268→    /* هم حداکثر جون زیاد می‌شود هم همین الان ترمیم می‌شود */

  1271→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(u.bonus), '#6fce62', 15);

  1274→    /* جلوه‌ی ارتقای برجک‌ها: جرقه‌ی طلایی روی همه‌ی برجک‌های نصب‌شده (۴٫۰) */
  1275→    for (const t of state.turrets) spawnParticles(t.x, t.y - 18, 10, '#ffd873',
  1276→      { speed: 130, life: 0.5, size: 2.2, grav: 250, up: true });
  1277→    if (state.turrets.length) addText(TOWER.x, TOWER_TOP_Y - 40, 'برجک‌ها قوی‌تر شدند! ⬆', '#ffd873', 14);
  1278→  }
  1279→  if (kind === 'trap' && lvl === 0) {
  1280→    /* نصب اول تله: جلوه بصری نصب */
  1281→    spawnParticles(TOWER.x, GROUND_Y - 8, 16, '#ffcf6e', { speed: 150, life: 0.5, size: 2.5, grav: 300, up: true });
  1282→    state.shake = Math.min(8, state.shake + 4);
  1283→  }
  1284→  SFX.buy();
  1285→  updateHUD(); refreshBuildPanel();
  1286→}
