   153→function csWalkStep(dt) {
   154→  for (const z of CS.zombies) {
   155→    z.walkPhase += dt * 5;
   156→    const dx = TOWER.x - z.x;
   157→    if (Math.abs(dx) > 118) z.x += Math.sign(dx) * z.speed * dt;
   158→  }
   159→}
