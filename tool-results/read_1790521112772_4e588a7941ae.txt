   178→const Snd = {
   179→  ctx: null, master: null, noiseBuf: null, muted: false, ready: false,
   180→
   181→  init() {
   182→    if (this.ready) return;
   183→    try {
   184→      const AC = window.AudioContext || window.webkitAudioContext;
   185→      if (!AC) return;
   186→      this.ctx = new AC();
   187→      this.master = this.ctx.createGain();
   188→      this.master.gain.value = this.muted ? 0 : 0.5;
   189→      this.master.connect(this.ctx.destination);
   190→      /* بافر نویز سفید یک‌ثانیه‌ای (برای شلیک/برخورد/باد/رعد) */
   191→      this.noiseBuf = this.ctx.createBuffer(1, this.ctx.sampleRate, this.ctx.sampleRate);
   192→      const d = this.noiseBuf.getChannelData(0);
   193→      for (let j = 0; j < d.length; j++) d[j] = Math.random() * 2 - 1;
   194→      this.ready = true;
   195→      this.startAmbient();
   196→    } catch (e) { /* مرورگر بدون صدا */ }
   197→  },
   198→  resume() { if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
   199→  toggleMute() {
   200→    this.muted = !this.muted;
   201→    if (this.master) this.master.gain.value = this.muted ? 0 : 0.5;

   205→  /* یک نُت: {type,f0,f1,dur,vol,att,delay} */
   206→  tone(o) {
   207→    if (!this.ready || this.muted) return;
   208→    try {
   209→      const t0 = this.ctx.currentTime + (o.delay || 0);
   210→      const osc = this.ctx.createOscillator(), g = this.ctx.createGain();
   211→      osc.type = o.type || 'sine';
   212→      osc.frequency.setValueAtTime(Math.max(1, o.f0), t0);
   213→      if (o.f1) osc.frequency.exponentialRampToValueAtTime(Math.max(1, o.f1), t0 + o.dur);
   214→      g.gain.setValueAtTime(0.0001, t0);
   215→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.008));
   216→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   217→      osc.connect(g); g.connect(this.master);
   218→      osc.start(t0); osc.stop(t0 + o.dur + 0.05);
   219→    } catch (e) {}
   220→  },
   221→  /* نویز فیلترشده: {dur,vol,ftype,freq,q,att,delay} */
   222→  noise(o) {
   223→    if (!this.ready || this.muted) return;
   224→    try {
   225→      const t0 = this.ctx.currentTime + (o.delay || 0);
   226→      const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf; src.loop = true;
   227→      const f = this.ctx.createBiquadFilter();
   228→      f.type = o.ftype || 'lowpass'; f.frequency.value = o.freq || 1000; f.Q.value = o.q || 0.8;
   229→      const g = this.ctx.createGain();
   230→      g.gain.setValueAtTime(0.0001, t0);
   231→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.005));
   232→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   233→      src.connect(f); f.connect(g); g.connect(this.master);
   234→      src.start(t0); src.stop(t0 + o.dur + 0.05);
   235→    } catch (e) {}
   236→  },
   237→
   238→  /* موسیقی محیطیِ شب: درونِ بم + باد + نُت‌های پراکنده‌ی دل‌انگیز */
   239→  startAmbient() {
   240→    try {
   241→      const dg = this.ctx.createGain(); dg.gain.value = 0.045; dg.connect(this.master);
   242→      const o1 = this.ctx.createOscillator(); o1.type = 'triangle'; o1.frequency.value = 55;
   243→      const o2 = this.ctx.createOscillator(); o2.type = 'sine'; o2.frequency.value = 110;
   244→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.09;
   245→      const lg = this.ctx.createGain(); lg.gain.value = 0.02;
   246→      lfo.connect(lg); lg.connect(dg.gain);
   247→      o1.connect(dg); o2.connect(dg);
   248→      o1.start(); o2.start(); lfo.start();
   249→      /* صدای باد (نویز لوپِ بسیار بم) */
   250→      const wind = this.ctx.createBufferSource(); wind.buffer = this.noiseBuf; wind.loop = true;
   251→      const wf = this.ctx.createBiquadFilter(); wf.type = 'lowpass'; wf.frequency.value = 260;
   252→      const wg = this.ctx.createGain(); wg.gain.value = 0.02;
   253→      wind.connect(wf); wf.connect(wg); wg.connect(this.master); wind.start();
   254→      this.pluck();
   255→    } catch (e) {}

   507→function startWave(n) {
   508→  state.wave = n;
   509→  state.mode = 'playing';
   510→  const c = WAVES[n - 1];
   511→  const q = [];
   512→  for (let i = 0; i < c.n; i++) q.push('normal');
   513→  for (let i = 0; i < c.f; i++) q.push('fast');
   514→  for (let i = 0; i < c.t; i++) q.push('tank');
   515→  for (let i = q.length - 1; i > 0; i--) {          // شافل فیشر–ییتس
   516→    const j = Math.floor(rand(0, i + 1));
   517→    const tmp = q[i]; q[i] = q[j]; q[j] = tmp;
   518→  }
   519→  if (c.b) q.push('boss');                           // پادشاه همیشه آخرِ صف می‌ایستد
   520→  state.spawn = {
   521→    queue: q,
   522→    timer: 1.2,
   523→    interval: Math.max(2.0 - n * 0.14, 0.5),        // فاصله بین تولدها (کم‌تر هر موج)
   524→    bossDelay: false,
   525→  };
   526→  setAnnounce('موج ' + faNum(n),
   527→    n === TOTAL_WAVES ? 'موج آخر! پادشاه زامبی‌ها پشتِ کاره...' : 'دفاع کن! 🧟', 2);

   530→  updateHUD();

   589→function tryShoot() {

   591→  const w = WEAPONS[state.weapon];
   592→  /* قدرت «شلیک سریع» نرخ شلیک را تقریباً دوبرابر می‌کند */
   593→  fireTimer = w.rate * (state.powerTimers.rapid > 0 ? 0.55 : 1);
   594→  updateAim();                             // اطمینان از جهت صحیح گلوله
   595→  const m = gunMuzzle(w.barrel);
   596→  for (let i = 0; i < w.pellets; i++) {
   597→    const a = state.aimAngle + (Math.random() - 0.5) * 2 * w.spread;
   598→    const sp = w.spd * rand(0.92, 1.08);
   599→    state.bullets.push({
   600→      x: m.x, y: m.y,
   601→      vx: Math.cos(a) * sp, vy: Math.sin(a) * sp,
   602→      dmg: weaponDmg(w), r: w.pellets > 1 ? 3 : 4, life: w.ttl,
   603→      knock: w.knock,                      // پس‌زنی سلاح (۳٫۰)
   604→      dead: false, pellet: w.pellets > 1,  // ساچمه‌های شاتگان ریزترند
   605→    });
   606→  }
   607→  state.stats.shots++;
   608→  state.fireFlash = 0.07;
   609→  state.muzzleGlow = 0.07;                 // نور دهانه روی محیط (۳٫۰)
   610→  state.recoil = w.pellets > 1 ? 1.6 : 1;  // لگد شاتکن محسوس‌تر است
   611→  SFX[w.sfx]();
   612→  spawnParticles(m.x, m.y, 3, '#ffe9a8', { speed: 90, life: 0.18, size: 2, grav: 0,
   613→    dir: state.aimAngle, spread: Math.PI / 3 });
   614→  spawnShell(state.aimAngle);              // پوکه‌ی پرتاب‌شده
   615→  if (w.pellets > 1) state.shake = Math.min(8, state.shake + 2);
   616→}

   618→/* پوکه‌ی براقی که بعد از هر شلیک به عقب پرتاب می‌شود و روی بام می‌غلتد */
   619→function spawnShell(ang) {
   620→  const gx = TOWER.x + Math.cos(ang) * 8, gy = TOWER_TOP_Y - 17 + Math.sin(ang) * 8;
   621→  const a = ang + rand(1.6, 2.2);          // زاویه‌ی پرتاب به عقب/بالا
   622→  state.particles.push({
   623→    x: gx, y: gy,
   624→    vx: Math.cos(a) * rand(60, 130), vy: Math.sin(a) * rand(60, 130) - 70,
   625→    age: 0, life: rand(0.5, 0.8), size: 2.6, color: '#d9a441', grav: 900,
   626→    shell: true, rot: rand(0, 6.28), vr: rand(-9, 9),
   627→  });
   628→}

   630→/* برخورد گلوله با زامبی‌ها + خروج گلوله از صحنه
   631→   (۳٫۰: اول دایره‌ی سر بررسی می‌شود — برخورد سر = ضربه‌ی بحرانی ×۲) */
   632→function updateBullets(dt) {
   633→  for (const b of state.bullets) {
   634→    if (b.dead) continue;
   635→    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
   636→    /* رد برفیِ گلوله‌ی یخی برجک (۴٫۰) */
   637→    if (b.frost && Math.random() < dt * 34) {
   638→      state.particles.push({ x: b.x, y: b.y, vx: rand(-12, 12), vy: rand(-12, 12),
   639→        age: 0, life: 0.3, size: 1.3, color: 'rgba(205,240,255,0.9)', grav: 30 });
   640→    }
   641→    if (b.y > GROUND_Y + 4) {              // گلوله به زمین خورد
   642→      b.dead = true;
   643→      spawnParticles(b.x, GROUND_Y, 4, '#6a5a43', { speed: 70, life: 0.3, size: 2, grav: 300, up: true });
   644→      continue;
   645→    }
   646→    if (b.x < -40 || b.x > W + 40 || b.y < -40 || b.life <= 0) { b.dead = true; continue; }
   647→    const dir = Math.sign(b.vx) || 1;      // جهت پس‌زنی = جهت حرکت گلوله
   648→    const src = b.tur || 'player';         // منبع گلوله: بازیکن یا برجک (۴٫۰)
   649→    for (const z of state.zombies) {
   650→      if (z.dead) continue;
   651→      const s = (z.radius / 15) * (z.depth || 1);   // مرکز بدن بر اساس اندازه‌ی نوع و عمق
   652→      /* --- سر: ضربه‌ی بحرانی --- */
   653→      const hx = z.x + z.dir * 3 * s, hy = z.y - 36 * s, hr = 7 * s + b.r;
   654→      if ((b.x - hx) * (b.x - hx) + (b.y - hy) * (b.y - hy) < hr * hr) {
   655→        b.dead = true;
   656→        hitZombie(z, b.dmg, b.x, b.y, dir, true, b.knock, src);
   657→        if (b.frost) chillZombie(z);
   658→        break;
   659→      }
   660→      /* --- بدن --- */
   661→      const dx = z.x - b.x, dy = (z.y - 20 * s) - b.y;
   662→      const rr2 = z.radius * (z.depth || 1) + b.r + 3;
   663→      if (dx * dx + dy * dy < rr2 * rr2) {
   664→        b.dead = true;
   665→        hitZombie(z, b.dmg, b.x, b.y, dir, !!b.crit, b.knock, src);
   666→        if (b.frost) chillZombie(z);
   667→        break;
   668→      }
   669→    }
   670→  }
   671→  state.bullets = state.bullets.filter(b => !b.dead);
   672→}

   674→/* ---------------------- ۸) زامبی‌ها: محاصره‌ی دسته‌جمعی برج ----------------------
   675→   در ۳٫۰ زامبی‌ها دیگر در یک خط نمی‌ایستند: هر طرفِ برج تا ۵ ردیف
   676→   «دیوارِ زامبی» می‌سازند؛ نزدیک‌ترین مهاجم ردیفِ چسبیده به دیوار را می‌گیرد و
   677→   بقیه پشت سرش ردیف می‌شوند. ردیف‌های عقب بالاتر و کوچک‌تر رسم می‌شوند و
   678→   با کشته‌شدنِ ردیف جلو، بقیه فشرده شده و جلو می‌خزند. */
   679→function updateZombies(dt) {
   680→  /* رتبه‌بندی مهاجم‌های فعلی بر اساس نزدیکی به دیوار (هر طرف جداگانه):
   681→     نزدیک‌ترین = ردیف ۰؛ با مرگ ردیف جلو، رتبه‌ها فشرده می‌شوند */
   682→  const rank = { '-1': 0, '1': 0 };
   683→  const attackers = state.zombies.filter(z => !z.dead && z.attacking);
   684→  attackers.sort((a, b) => Math.abs(TOWER.x - a.x) - Math.abs(TOWER.x - b.x));
   685→  for (const z of attackers) {
   686→    z.attackSide = z.x < TOWER.x ? -1 : 1;
   687→    z.lane = Math.min(rank[z.attackSide]++, 4);
   688→  }
   689→
   690→  for (const z of state.zombies) {
   691→    if (z.dead) { z.deathT += dt; continue; }   // انیمیشن مرگ ادامه پیدا کند
   692→    z.hitFlash = Math.max(0, z.hitFlash - dt);
   693→    z.stagger = Math.max(0, z.stagger - dt);    // سرخوردن بعد از ضربه
   694→    z.chill = Math.max(0, (z.chill || 0) - dt); // یخ‌زدگی برجک یخ‌زن (۴٫۰)
   695→
   696→    const dx = TOWER.x - z.x;
   697→    const side = z.x < TOWER.x ? -1 : 1;        // کدام دیوار برج؟
   698→    const lane = z.attacking ? z.lane : Math.min(rank[side], 4);
   699→    const stop = TOWER.baseHalf + z.radius * 0.55 + lane * 15;
   700→
   701→    /* انجماد: همه کُند می‌شوند (پادشاه مقاوم‌تر است) — یخ‌زدگی برجک هم جداگانه اثر می‌کند */
   702→    const slow = state.powerTimers.freeze > 0 ? (z.type === 'boss' ? 0.6 : 0.32) : 1;
   703→    const spd = z.speed * slow * (z.chill > 0 ? 0.55 : 1) * (z.stagger > 0 ? 0.25 : 1);
   704→
   705→    if (Math.abs(dx) > stop) {
   706→      /* راه رفتن به سمت برج (مهاجمِ پس‌زده‌شده هم برمی‌گردد) */
   707→      z.attacking = false;
   708→      z.lane = Math.min(rank[side], 4);
   709→      z.x += Math.sign(dx) * spd * dt;
   710→      z.walkPhase += dt * (4 + spd * 0.06);
   711→    } else {
   712→      /* رسید به دیوار برج: کوبیدن پایه با فاصله‌ی زمانیِ نوعِ خودش */
   713→      z.attacking = true;
   714→      z.attackSide = side;
   715→      z.attackAnim += dt * 5;
   716→      z.biteTimer -= dt;
   717→      if (z.biteTimer <= 0) {
   718→        z.biteTimer = z.biteInt;
   719→        damageTower(z.dmg, z.x);
   720→      }
   721→    }
   722→
   723→    /* عمق تصویری: ردیف‌های عقب‌تر کمی بالاتر ایستاده و کوچک‌تر دیده می‌شوند */
   724→    const yTarget = GROUND_Y - z.lane * 9 + z.yJit;
   725→    z.y += (yTarget - z.y) * Math.min(1, dt * 6);
   726→    z.depth = 1 - z.lane * 0.07;
   727→  }
   728→  state.zombies = state.zombies.filter(z => !z.dead || z.deathT < 0.55);
   729→}

   731→/* آسیب گلوله به زامبی (۳٫۰: کریت ×۲ + پس‌زنی + سرخوردن + نشانگر برخورد) */
   732→function hitZombie(z, dmg, bx, by, dir, crit, knock, src) {
   733→  if (z.dead) return;
   734→  if (src) z.killedBy = src;                 // منبع ضربه — برای آمار کشتار برجک‌ها (۴٫۰)
   735→  const total = crit ? dmg * 2 : dmg;
   736→  z.hp -= total;
   737→  z.hitFlash = 0.15;
   738→  z.stagger = 0.15;                         // چند لحظه از می‌افتد
   739→  z.x += dir * (knock || 0) * (crit ? 1.7 : 1);   // پس‌زنی فیزیکی
   740→  if (crit) {
   741→    addText(z.x, z.y - 52, faNum(total), '#ffb347', 16);
   742→    spawnParticles(bx, by, 5, '#ffd873', { speed: 150, life: 0.35, size: 2.2, grav: 300 });
   743→  } else {
   744→    addText(z.x, z.y - 46, faNum(total), '#ffffff', 12);
   745→  }
   746→  spawnParticles(bx, by, 3, '#8fd06a', { speed: 110, life: 0.3, size: 2, grav: 400 });
   747→  spawnParticles(bx, by, 3, '#7a2f2f', { speed: 90,  life: 0.35, size: 2, grav: 500 });
   748→  if (Math.random() < 0.3) addDecal(z.x, false);   // چکه‌ی خون روی زمین
   749→  state.hitMark = 0.14; state.hitMarkCrit = !!crit; // نشانگر برخورد روی نشان
   750→  if (crit) SFX.crit(); else SFX.hit();
   751→  if (z.hp <= 0) killZombie(z);
   752→}

   754→/* مرگ زامبی: انیمیشن افتادن + خون + سکه + روح + جعبه‌ی قدرت + کمبو (۳٫۰) */
   755→function killZombie(z) {
   756→  if (z.dead) return;
   757→  z.dead = true; z.deathT = 0;
   758→  state.stats.kills++;
   759→  if (z.killedBy && z.killedBy !== 'player') state.stats.turKills++;   // کشتار دفاع‌های خودکار (۴٫۰)
   760→  spawnParticles(z.x, z.y - 18, 12, '#5f8f3e', { speed: 170, life: 0.55, size: 3, grav: 500 });
   761→  spawnParticles(z.x, z.y - 18, 8, '#7a2f2f', { speed: 130, life: 0.5, size: 2.6, grav: 500 });
   762→  addDecal(z.x, true);                     // لکه‌ی خون بزرگ زیر جسد
   763→  /* روحِ سبزِ کوچکی که آرام به آسمان می‌رود */
   764→  state.particles.push({
   765→    x: z.x, y: z.y - 26, vx: rand(-6, 6), vy: -38,
   766→    age: 0, life: 1.3, size: 3, color: 'rgba(190,255,200,0.5)', grav: -26, glow: true,
   767→  });
   768→  SFX.death();
   769→  dropCoins(z);
   770→  maybeDropPowerup(z);
   771→  bumpCombo();
   772→  updateHUD();
   773→  if (z.type === 'boss') {
   774→    state.shake = 16; state.flash = 0.8;
   775→    state.slowmo = 0.9;                    // لحظه‌ی سینمایی مرگ پادشاه
   776→    SFX.slowmo();
   777→    spawnParticles(z.x, z.y - 60, 40, '#ffd873', { speed: 300, life: 1.0, size: 3.5, grav: 400 });
   778→    spawnParticles(z.x, z.y - 60, 24, '#e8563f', { speed: 240, life: 0.8, size: 3, grav: 400 });
   779→    setAnnounce('👑 پادشاه کشته شد!', 'برج پابرجایی!', 2.2);
   780→    SFX.bossDie();
   781→  }
   782→}

   802→/* کاهش جون برج (کوبیدن پایه توسط زامبی‌ها) */
   803→function damageTower(amount, srcX) {
   804→  if (state.mode !== 'playing') return;
   805→  state.tower.hp = Math.max(0, state.tower.hp - amount);
   806→  state.flash = 1;
   807→  state.shake = Math.min(10, state.shake + 3);
   808→  const side = srcX < TOWER.x ? -1 : 1;
   809→  addText(TOWER.x + side * (TOWER.baseHalf + 6), GROUND_Y - 60, faNum(amount), '#ff6b5e', 16);
   810→  /* تراشه‌های سنگ از پایه‌ی برج می‌پارد (۳٫۰: بیشتر و پررنگ‌تر) */

   812→    { speed: 130, life: 0.45, size: 2.6, grav: 500 });
   813→  spawnParticles(TOWER.x + side * TOWER.baseHalf, GROUND_Y - rand(8, 40), 3, '#c9b38a',
   814→    { speed: 100, life: 0.35, size: 2, grav: 500 });

   820→/* تله پایه برج: آسیب خودکار و پیوسته به زامبی‌های نزدیک */
   821→function updateTrap(dt) {
   822→  const lvl = state.upgrades.trap;
   823→  if (lvl <= 0) return;
   824→  const dps = UPGRADES.trap.dps + UPGRADES.trap.dpsPerLvl * (lvl - 1);
   825→  for (const z of state.zombies) {
   826→    if (z.dead) continue;
   827→    if (Math.abs(z.x - TOWER.x) < UPGRADES.trap.radius) {
   828→      z.hp -= dps * dt;
   829→      z.hitFlash = Math.max(z.hitFlash, 0.06);
   830→      if (Math.random() < dt * 6) { addZap(z.x, z.y - 18); SFX.trap(); }  // قوس برق نمایشی
   831→      if (z.hp <= 0) { z.killedBy = 'trap'; killZombie(z); }
   832→    }
   833→  }
   834→}

   836→/* ---------------------- ۸٫۵) برجک‌های دفاعی خودکار (۴٫۰) ----------------------
   837→   بازیکن برجک می‌خرد → شبحِ نصب دنبال ماوس می‌آید → کلیک روی زمین = نصب.
   838→   هر برجک نزدیک‌ترین زامبیِ در برد را نشانه می‌رود و خودکار شلیک می‌کند. */
   839→
   840→/* آیا نقطه‌ی (x,y) جای نصب مناسبی است؟ (نوار زمین، دوری از برج/برجک‌ها/مانع‌های صحنه) */
   841→function placementValid(x, y) {
   842→  if (y < GROUND_Y - 34 || y > GROUND_Y + 16) return false;    // فقط نوار زمین
   843→  if (x < 26 || x > W - 200) return false;                      // داخل کادر و دور از پنل ساخت‌وساز
   844→  if (Math.abs(x - TOWER.x) < TOWER.baseHalf + 34) return false; // چسبیده به پایه‌ی برج نه
   845→  for (const t of state.turrets) if (Math.hypot(t.x - x, t.y - y) < 36) return false;
   846→  for (const lx of LAMPS) if (Math.abs(x - lx) < 18) return false;
   847→  for (const tx of [88, 874]) if (Math.abs(x - tx) < 22) return false;   // درخت‌های خشک
   848→  for (const p of PUDDLES) if (Math.abs(x - p.x) < p.w / 2 + 8) return false;
   849→  return true;
   850→}

   861→  state.placing = { type: kind };
   862→  SFX.click();
   863→  if (!state.tipTurret) {
   864→    state.tipTurret = true;
   865→    setAnnounce('🛠 نصب ' + TURRETS[kind].name, 'روی زمینِ سبز کلیک کن — راست‌کلیک: لغو', 2.4);
   866→  }
   867→  refreshBuildPanel();
   868→}

   878→/* کلیک چپ در حالت نصب: اگر جای‌ درست بود برجک ساخته می‌شود */
   879→function handlePlaceClick() {
   880→  const pl = state.placing;
   881→  if (!pl) return;
   882→  const def = TURRETS[pl.type];
   883→  const mx = state.mouse.x, my = state.mouse.y;
   884→  if (!placementValid(mx, my)) { SFX.deny(); return; }
   885→  if (state.turrets.length >= MAX_TURRETS) { cancelPlacement(); return; }
   886→  if (state.coins < def.price) { denyBPItem(pl.type); cancelPlacement(); return; }
   887→  state.coins -= def.price;
   888→  state.turrets.push({
   889→    type: pl.type, x: mx, y: my,
   890→    cd: 0.45,                                // کمی تأخیر اولیه (حسِ نصب)
   891→    aim: mx < TOWER.x ? 0 : Math.PI,         // اول رو به برج
   892→    spin: 0, flash: 0, recoil: 0, hover: false, bob: rand(0, 6.28),
   893→  });
   894→  /* جلوه‌ی نصب: گردوخاک + کوبیدن + متن */
   895→  spawnParticles(mx, my + 2, 14, '#7a6a4c', { speed: 110, life: 0.5, size: 2.6, grav: 380, up: true });
   896→  spawnParticles(mx, my - 4, 8, '#c9b38a', { speed: 90, life: 0.4, size: 2, grav: 300, up: true });
   897→  state.shake = Math.min(6, state.shake + 3);
   898→  addText(mx, my - 40, def.name, '#9fe08a', 13);
   899→  SFX.placeT();
   900→  state.placing = null;
   901→  updateHUD();
   902→}

   904→/* یخ‌زدگی زامبی با گلوله‌ی برجک یخ‌زن */
   905→function chillZombie(z) {
   906→  if (z.dead) return;
   907→  z.chill = Math.max(z.chill || 0, TURRETS.frost.chill);

   912→function updateTurrets(dt) {
   913→  for (const t of state.turrets) {
   914→    const def = TURRETS[t.type];
   915→    t.cd = Math.max(0, t.cd - dt);
   916→    t.flash = Math.max(0, t.flash - dt);
   917→    t.recoil = Math.max(0, t.recoil - dt * 5);
   918→    t.hover = !state.placing && Math.abs(state.mouse.x - t.x) < 26 && Math.abs(state.mouse.y - (t.y - 14)) < 30;
   919→
   920→    /* نزدیک‌ترین زامبیِ زنده در شعاع برد */
   921→    let best = null, bd = Infinity;
   922→    for (const z of state.zombies) {
   923→      if (z.dead) continue;
   924→      const d = Math.abs(z.x - t.x);
   925→      if (d <= def.range && d < bd) { bd = d; best = z; }
   926→    }
   927→    if (best) {
   928→      const s = (best.radius / 15) * (best.depth || 1);
   929→      const want = Math.atan2((best.y - 20 * s) - (t.y - def.h), best.x - t.x);
   930→      let diff = want - t.aim;
   931→      while (diff > Math.PI) diff -= Math.PI * 2;
   932→      while (diff < -Math.PI) diff += Math.PI * 2;
   933→      t.aim += diff * Math.min(1, dt * 8);   // چرخش نرم لوله به سمت هدف
   934→      if (t.cd <= 0 && Math.abs(diff) < 0.35) turretFire(t);
   935→    }
   936→    if (t.type === 'gatling') t.spin += dt * (best ? 16 : 3);   // چرخش لوله‌های مسلسل
   937→    /* برفک‌های محیطی برجک یخ‌زن */
   938→    if (t.type === 'frost' && Math.random() < dt * 2.5) {
   939→      state.particles.push({ x: t.x + rand(-13, 13), y: t.y - rand(14, 30),
   940→        vx: rand(-6, 6), vy: rand(8, 20), age: 0, life: rand(0.6, 1.2),
   941→        size: rand(1, 1.8), color: 'rgba(200,235,255,0.8)', grav: 10 });

  1007→function updateCoins(dt) {
  1008→  const target = coinTarget(); // یک بار در هر فریم محاسبه می‌شود
  1009→  for (const c of state.coinsArr) {
  1010→    if (c.dead) continue;

  1020→      }
  1021→    } else {

  1023→      const dx = target.x - c.x, dy = target.y - c.y;
  1024→      const d = Math.hypot(dx, dy) || 1;

  1033→        age: 0, life: 0.35, size: 1.2, color: '#ffe98a', grav: 0 });

  1037→  state.coinsArr = state.coinsArr.filter(c => !c.dead);

  1178→function attemptBuy(kind) {
  1179→  if (!canPlay()) return;                  // فقط وقتی بازی جریان دارد
  1180→  const u = UPGRADES[kind];
  1181→
  1182→  /* --- برجک‌های دفاعی: به حالت انتخاب محل نصب می‌رود (۴٫۰) --- */
  1183→  if (kind === 'gatling' || kind === 'sniper' || kind === 'frost') {
  1184→    startPlacement(kind);
  1185→    return;
  1186→  }
  1187→
  1188→  /* --- سلاح‌های یک‌بار خرید (شاتگان/مسلسل) --- */
  1189→  if (kind === 'shotgun' || kind === 'mg') {
  1190→    if (WEAPONS[kind].owned) return;
  1191→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1192→    state.coins -= u.price;
  1193→    WEAPONS[kind].owned = true;
  1194→    selectWeapon(kind);                    // بلافاصله دستِ کاربر می‌افتد
  1195→    SFX.buy();
  1196→    spawnParticles(TOWER.x, TOWER_TOP_Y - 20, 18, '#ffd873', { speed: 170, life: 0.6, size: 2.6, grav: 300, up: true });
  1197→    updateHUD(); refreshBuildPanel();
  1198→    return;
  1199→  }
  1200→
  1201→  /* --- تعمیر سریع --- */
  1202→  if (kind === 'repair') {
  1203→    if (state.tower.hp >= state.tower.maxHp - 0.5) return;
  1204→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1205→    state.coins -= u.price;
  1206→    const heal = Math.round(state.tower.maxHp * u.frac);
  1207→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + heal);
  1208→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(heal), '#6fce62', 16);
  1209→    spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#6fce62', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
  1210→    SFX.buy();
  1211→    updateHUD(); refreshBuildPanel();
  1212→    return;
  1213→  }
  1214→
  1215→  /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله) --- */
  1216→  const lvl = state.upgrades[kind];
  1217→  if (lvl >= u.max) return;
  1218→  const price = priceOf(kind);
  1219→  if (state.coins < price) { denyBPItem(kind); return; }
  1220→  state.coins -= price;
  1221→  state.upgrades[kind] = lvl + 1;
  1222→  if (kind === 'health') {
  1223→    /* هم حداکثر جون زیاد می‌شود هم همین الان ترمیم می‌شود */
  1224→    state.tower.maxHp += u.bonus;
  1225→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + u.bonus);
  1226→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(u.bonus), '#6fce62', 15);
  1227→  }
  1228→  if (kind === 'turretUp') {
  1229→    /* جلوه‌ی ارتقای برجک‌ها: جرقه‌ی طلایی روی همه‌ی برجک‌های نصب‌شده (۴٫۰) */
  1230→    for (const t of state.turrets) spawnParticles(t.x, t.y - 18, 10, '#ffd873',
  1231→      { speed: 130, life: 0.5, size: 2.2, grav: 250, up: true });
  1232→    if (state.turrets.length) addText(TOWER.x, TOWER_TOP_Y - 40, 'برجک‌ها قوی‌تر شدند! ⬆', '#ffd873', 14);
  1233→  }
  1234→  if (kind === 'trap' && lvl === 0) {
  1235→    /* نصب اول تله: جلوه بصری نصب */
  1236→    spawnParticles(TOWER.x, GROUND_Y - 8, 16, '#ffcf6e', { speed: 150, life: 0.5, size: 2.5, grav: 300, up: true });
  1237→    state.shake = Math.min(8, state.shake + 4);
  1238→  }
  1239→  SFX.buy();
  1240→  updateHUD(); refreshBuildPanel();
  1241→}
