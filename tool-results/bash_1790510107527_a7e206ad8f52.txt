   120→/* ---------------------- ۳) موتور صدا (Web Audio) ---------------------- */
   121→/* همه‌ی صداها همین‌جا سینتز می‌شوند — هیچ فایل صوتی خارجی وجود ندارد.
   122→   AudioContext فقط بعد از اولین تعامل کاربر ساخته می‌شود (سیاست مرورگرها). */
   123→const Snd = {
   124→  ctx: null, master: null, noiseBuf: null, muted: false, ready: false,
   125→
   126→  init() {
   127→    if (this.ready) return;
   128→    try {
   129→      const AC = window.AudioContext || window.webkitAudioContext;
   130→      if (!AC) return;
   131→      this.ctx = new AC();
   132→      this.master = this.ctx.createGain();
   133→      this.master.gain.value = this.muted ? 0 : 0.5;
   134→      this.master.connect(this.ctx.destination);
   135→      /* بافر نویز سفید یک‌ثانیه‌ای (برای شلیک/برخورد/باد) */
   136→      this.noiseBuf = this.ctx.createBuffer(1, this.ctx.sampleRate, this.ctx.sampleRate);
   137→      const d = this.noiseBuf.getChannelData(0);
   138→      for (let j = 0; j < d.length; j++) d[j] = Math.random() * 2 - 1;
   139→      this.ready = true;
   140→      this.startAmbient();
   141→    } catch (e) { /* مرورگر بدون صدا */ }
   142→  },
   143→  resume() { if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
   144→  toggleMute() {
   145→    this.muted = !this.muted;
   146→    if (this.master) this.master.gain.value = this.muted ? 0 : 0.5;
   147→    return this.muted;
   148→  },
   149→
   150→  /* یک نُت: {type,f0,f1,dur,vol,att,delay} */
   151→  tone(o) {
   152→    if (!this.ready || this.muted) return;
   153→    try {
   154→      const t0 = this.ctx.currentTime + (o.delay || 0);
   155→      const osc = this.ctx.createOscillator(), g = this.ctx.createGain();
   156→      osc.type = o.type || 'sine';
   157→      osc.frequency.setValueAtTime(Math.max(1, o.f0), t0);
   158→      if (o.f1) osc.frequency.exponentialRampToValueAtTime(Math.max(1, o.f1), t0 + o.dur);
   159→      g.gain.setValueAtTime(0.0001, t0);
   160→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.008));
   161→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   162→      osc.connect(g); g.connect(this.master);
   163→      osc.start(t0); osc.stop(t0 + o.dur + 0.05);
   164→    } catch (e) {}
   165→  },
   166→  /* نویز فیلترشده: {dur,vol,ftype,freq,q,att,delay} */
   167→  noise(o) {
   168→    if (!this.ready || this.muted) return;
   169→    try {
   170→      const t0 = this.ctx.currentTime + (o.delay || 0);
   171→      const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf; src.loop = true;
   172→      const f = this.ctx.createBiquadFilter();
   173→      f.type = o.ftype || 'lowpass'; f.frequency.value = o.freq || 1000; f.Q.value = o.q || 0.8;
   174→      const g = this.ctx.createGain();
   175→      g.gain.setValueAtTime(0.0001, t0);
   176→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.005));
   177→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   178→      src.connect(f); f.connect(g); g.connect(this.master);
   179→      src.start(t0); src.stop(t0 + o.dur + 0.05);
   180→    } catch (e) {}
   181→  },
   182→
   183→  /* موسیقی محیطیِ شب: درونِ بم + باد + نُت‌های پراکنده‌ی دل‌انگیز */
   184→  startAmbient() {
   185→    try {
   186→      const dg = this.ctx.createGain(); dg.gain.value = 0.045; dg.connect(this.master);
   187→      const o1 = this.ctx.createOscillator(); o1.type = 'triangle'; o1.frequency.value = 55;
   188→      const o2 = this.ctx.createOscillator(); o2.type = 'sine'; o2.frequency.value = 110;
   189→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.09;
   190→      const lg = this.ctx.createGain(); lg.gain.value = 0.02;
   191→      lfo.connect(lg); lg.connect(dg.gain);
   192→      o1.connect(dg); o2.connect(dg);
   193→      o1.start(); o2.start(); lfo.start();
   194→      /* صدای باد (نویز لوپِ بسیار بم) */
   195→      const wind = this.ctx.createBufferSource(); wind.buffer = this.noiseBuf; wind.loop = true;
   196→      const wf = this.ctx.createBiquadFilter(); wf.type = 'lowpass'; wf.frequency.value = 260;
   197→      const wg = this.ctx.createGain(); wg.gain.value = 0.02;
   198→      wind.connect(wf); wf.connect(wg); wg.connect(this.master); wind.start();
   199→      this.pluck();
   200→    } catch (e) {}
   201→  },
   202→  pluck() {
   203→    if (!this.ready) return;
   204→    const notes = [220, 246.9, 261.6, 329.6, 392];
   205→    const f = notes[Math.floor(rand(0, notes.length))];
   206→    this.tone({ type: 'sine', f0: f, dur: 2.4, vol: 0.05, att: 0.5 });
   207→    this.tone({ type: 'sine', f0: f * 1.5, dur: 2.8, vol: 0.028, att: 0.7, delay: 0.03 });
   208→    setTimeout(() => this.pluck(), rand(3500, 8000));
   209→  },

   433→    dmg: T.dmg,

   440→/* ---------------------- ۷) شلیک و سلاح‌ها ---------------------- */
   441→/* آسیب هر گلوله/ساچمه بر اساس سلاح + سطح ارتقای اسلحه
   442→   (سلاح‌های ساچمه‌ای به‌ازای هر سطح +۱ می‌گیرند و تک‌گلوله‌ای +۲) */
   443→function weaponDmg(w, extra) {
   444→  const L = state.upgrades.weapon + (extra || 0);
   445→  return w.dmg + L * (w.pellets > 1 ? 1 : 2);

   448→/* محل دهانه‌ی اسلحه (نوک لوله در بالای برج) — طول لوله به سلاح بستگی دارد */
   449→function gunMuzzle(len) {
   450→  const gx = TOWER.x, gy = TOWER_TOP_Y - 17;
   451→  return { x: gx + Math.cos(state.aimAngle) * (len + 6), y: gy + Math.sin(state.aimAngle) * (len + 6) };

   454→function tryShoot() {
   455→  if (fireTimer > 0) return;
   456→  const w = WEAPONS[state.weapon];
   457→  fireTimer = w.rate;
   458→  updateAim();                             // اطمینان از جهت صحیح گلوله
   459→  const m = gunMuzzle(w.barrel);
   460→  for (let i = 0; i < w.pellets; i++) {
   461→    const a = state.aimAngle + (Math.random() - 0.5) * 2 * w.spread;
   462→    const sp = w.spd * rand(0.92, 1.08);
   463→    state.bullets.push({
   464→      x: m.x, y: m.y,
   465→      vx: Math.cos(a) * sp, vy: Math.sin(a) * sp,
   466→      dmg: weaponDmg(w), r: w.pellets > 1 ? 3 : 4, life: w.ttl,
   467→      dead: false, pellet: w.pellets > 1,  // ساچمه‌های شاتگان ریزترند
   468→    });
   469→  }
   470→  state.stats.shots++;
   471→  state.fireFlash = 0.07;
   472→  state.recoil = w.pellets > 1 ? 1.6 : 1;  // لگد شاتکن محسوس‌تر است
   473→  SFX[w.sfx]();
   474→  spawnParticles(m.x, m.y, 3, '#ffe9a8', { speed: 90, life: 0.18, size: 2, grav: 0,
   475→    dir: state.aimAngle, spread: Math.PI / 3 });
   476→  spawnShell(state.aimAngle);              // پوکه‌ی پرتاب‌شده
   477→  if (w.pellets > 1) state.shake = Math.min(8, state.shake + 2);
   478→}

   481→function spawnShell(ang) {
   482→  const gx = TOWER.x + Math.cos(ang) * 8, gy = TOWER_TOP_Y - 17 + Math.sin(ang) * 8;
   483→  const a = ang + rand(1.6, 2.2);          // زاویه‌ی پرتاب به عقب/بالا
   484→  state.particles.push({
   485→    x: gx, y: gy,

   487→    age: 0, life: rand(0.5, 0.8), size: 2.6, color: '#d9a441', grav: 900,

   492→/* برخورد گلوله با زامبی‌ها + خروج گلوله از صحنه */
   493→function updateBullets(dt) {
   494→  for (const b of state.bullets) {
   495→    if (b.dead) continue;
   496→    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
   497→    if (b.y > GROUND_Y + 4) {              // گلوله به زمین خورد
   498→      b.dead = true;
   499→      spawnParticles(b.x, GROUND_Y, 4, '#6a5a43', { speed: 70, life: 0.3, size: 2, grav: 300, up: true });
   500→      continue;
   501→    }
   502→    if (b.x < -40 || b.x > W + 40 || b.y < -40 || b.life <= 0) { b.dead = true; continue; }
   503→    for (const z of state.zombies) {
   504→      if (z.dead) continue;
   505→      const s = z.radius / 15;             // مرکز بدن بر اساس اندازه‌ی نوع
   506→      const dx = z.x - b.x, dy = (z.y - 20 * s) - b.y;
   507→      const rr2 = z.radius + b.r + 3;
   508→      if (dx * dx + dy * dy < rr2 * rr2) {
   509→        b.dead = true;
   510→        hitZombie(z, b.dmg, b.x, b.y);
   511→        break;
   512→      }
   513→    }
   514→  }
   515→  state.bullets = state.bullets.filter(b => !b.dead);
   516→}

   518→/* ---------------------- ۸) زامبی‌ها: حرکت، حمله، تله ---------------------- */
   519→function updateZombies(dt) {
   520→  for (const z of state.zombies) {
   521→    if (z.dead) { z.deathT += dt; continue; }   // انیمیشن مرگ ادامه پیدا کند
   522→    z.hitFlash = Math.max(0, z.hitFlash - dt);
   523→    const dx = TOWER.x - z.x;
   524→    const stop = TOWER.baseHalf + z.radius; // فاصله توقف از مرکز برج (به اندازه‌ی نوع)
   525→
   526→    if (Math.abs(dx) > stop) {
   527→      /* راه رفتن به سمت برج */
   528→      z.attacking = false;
   529→      z.x += Math.sign(dx) * z.speed * dt;
   530→      z.walkPhase += dt * (4 + z.speed * 0.06);
   531→    } else {
   532→      /* رسید به برج: گاز گرفتن با فاصله زمانیِ نوعِ خودش */
   533→      z.attacking = true;
   534→      z.attackAnim += dt * 5;
   535→      z.biteTimer -= dt;
   536→      if (z.biteTimer <= 0) {
   537→        z.biteTimer = z.biteInt;
   538→        damageTower(z.dmg, z.x);
   539→      }
   540→    }
   541→  }
   542→  state.zombies = state.zombies.filter(z => !z.dead || z.deathT < 0.55);
   543→}

   546→function hitZombie(z, dmg, bx, by) {

   553→  SFX.hit();
   554→  if (z.hp <= 0) killZombie(z);
   555→}

   590→/* تله پایه برج: آسیب خودکار و پیوسته به زامبی‌های نزدیک */
   591→function updateTrap(dt) {
   592→  const lvl = state.upgrades.trap;
   593→  if (lvl <= 0) return;
   594→  const dps = UPGRADES.trap.dps + UPGRADES.trap.dpsPerLvl * (lvl - 1);
   595→  for (const z of state.zombies) {
   596→    if (z.dead) continue;
   597→    if (Math.abs(z.x - TOWER.x) < UPGRADES.trap.radius) {
   598→      z.hp -= dps * dt;
   599→      z.hitFlash = Math.max(z.hitFlash, 0.06);
   600→      if (Math.random() < dt * 6) { addZap(z.x, z.y - 18); SFX.trap(); }  // قوس برق نمایشی
   601→      if (z.hp <= 0) killZombie(z);
   602→    }
   603→  }
   604→}

   607→/* با مرگ هر زامبی چند سکه می‌پرد بیرون (تعدادش به نوع دشمن بستگی دارد) */
   608→function dropCoins(z) {
   609→  const range = ZTYPES[z.type].coins;
   610→  const n = Math.round(rand(range[0], range[1]));
   611→  for (let i = 0; i < n; i++) {
   612→    state.coinsArr.push({
   613→      x: z.x, y: z.y - 18, vx: rand(-100, 100), vy: rand(-280, -140),
   614→      phase: 'drop', t: 0, bounces: 0, flySpeed: 420, dead: false,
   615→    });
   616→  }
   617→}

   619→/* محل شمارنده سکه در HUD (تبدیل مختصات صفحه به مختصات بوم تا سکه‌ها دقیقاً به آن پرو کنند) */
   620→function coinTarget() {
   621→  try {
   622→    const r = canvas.getBoundingClientRect();
   623→    const c = $('coinItem').getBoundingClientRect();
   624→    return {
   625→      x: (c.left + c.width / 2 - r.left) / r.width * W,
   626→      y: (c.top + c.height / 2 - r.top) / r.height * H,
   627→    };
   628→  } catch (e) { return { x: W - 150, y: 26 }; }
   629→}
   630→
   631→function updateCoins(dt) {
   632→  const target = coinTarget(); // یک بار در هر فریم محاسبه می‌شود
   633→  for (const c of state.coinsArr) {
   634→    if (c.dead) continue;
   635→    c.t += dt;
   636→    if (c.phase === 'drop') {
   637→      /* پرتاب اولیه + جاذبه + وایراستن روی زمین */
   638→      c.vy += 900 * dt;
   639→      c.x += c.vx * dt; c.y += c.vy * dt;
   640→      if (c.y > GROUND_Y - 6) {
   641→        c.y = GROUND_Y - 6;
   642→        c.vy *= -0.45; c.vx *= 0.7; c.bounces++;
   643→        if (Math.abs(c.vy) < 40 || c.bounces >= 2 || c.t > 0.9) c.phase = 'fly';
   644→      }
   645→    } else {
   646→      /* پرو شتاب‌دار به سمت شمارنده سکه در HUD */
   647→      const dx = target.x - c.x, dy = target.y - c.y;
   648→      const d = Math.hypot(dx, dy) || 1;
   649→      c.flySpeed += 1400 * dt;
   650→      const step = Math.min(d, c.flySpeed * dt);
   651→      c.x += dx / d * step; c.y += dy / d * step;
   652→      if (d < 18) { collectCoin(c); continue; }
   653→    }
   654→    if (c.t > 4) collectCoin(c); // بیمه: سکه گم‌نشده جمع شود
   655→  }
   656→  state.coinsArr = state.coinsArr.filter(c => !c.dead);
   657→}

   659→function collectCoin(c) {
   660→  if (c.dead) return;
   661→  c.dead = true;
   662→  state.coins++;
   663→  state.stats.earned++;
   664→  updateHUD();
   665→  pulseCoins();
   666→}

   690→/* انتخاب سلاح (کلید ۱/۲/۳ یا کلیک روی نوار سلاح) */
   691→function selectWeapon(id) {
   692→  if (!WEAPONS[id].owned) {
   693→    SFX.deny();
   694→    const el = document.querySelector('#weaponBar .wslot[data-w="' + id + '"]');
   695→    if (el) { el.classList.remove('deny'); void el.offsetWidth; el.classList.add('deny'); }
   696→    return;
   697→  }
   698→  if (state.weapon === id) return;
   699→  state.weapon = id;
   700→  SFX.click();
   701→  refreshWeaponBar();
   702→}

   704→function attemptBuy(kind) {
   705→  const u = UPGRADES[kind];
   706→
   707→  /* --- سلاح‌های یک‌بار خرید (شاتگان/مسلسل) --- */
   708→  if (kind === 'shotgun' || kind === 'mg') {
   709→    if (WEAPONS[kind].owned) return;
   710→    if (state.coins < u.price) { denyBtn('buy' + kind[0].toUpperCase() + kind.slice(1)); return; }
   711→    state.coins -= u.price;
   712→    WEAPONS[kind].owned = true;
   713→    selectWeapon(kind);                    // بلافاصله دستِ کاربر می‌افتد
   714→    SFX.buy();
   715→    spawnParticles(TOWER.x, TOWER_TOP_Y - 20, 18, '#ffd873', { speed: 170, life: 0.6, size: 2.6, grav: 300, up: true });
   716→    updateHUD(); refreshShop();
   717→    return;
   718→  }
   719→
   720→  /* --- تعمیر سریع --- */
   721→  if (kind === 'repair') {
   722→    if (state.tower.hp >= state.tower.maxHp - 0.5) return;
   723→    if (state.coins < u.price) { denyBtn('buyRepair'); return; }
   724→    state.coins -= u.price;
   725→    const heal = Math.round(state.tower.maxHp * u.frac);
   726→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + heal);
   727→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(heal), '#6fce62', 16);
   728→    spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#6fce62', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
   729→    SFX.buy();
   730→    updateHUD(); refreshShop();
   731→    return;
   732→  }
   733→
   734→  /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله) --- */
   735→  const lvl = state.upgrades[kind];
   736→  if (lvl >= u.max) return;
   737→  const price = priceOf(kind);
   738→  if (state.coins < price) { denyBtn('buy' + kind[0].toUpperCase() + kind.slice(1)); return; }
   739→  state.coins -= price;
   740→  state.upgrades[kind] = lvl + 1;
   741→  if (kind === 'health') {
   742→    /* هم حداکثر جون زیاد می‌شود هم همین الان ترمیم می‌شود */
   743→    state.tower.maxHp += u.bonus;
   744→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + u.bonus);
   745→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(u.bonus), '#6fce62', 15);
   746→  }
   747→  if (kind === 'trap' && lvl === 0) {
   748→    /* نصب اول تله: جلوه بصری نصب */
   749→    spawnParticles(TOWER.x, GROUND_Y - 8, 16, '#ffcf6e', { speed: 150, life: 0.5, size: 2.5, grav: 300, up: true });
   750→    state.shake = Math.min(8, state.shake + 4);
   751→  }
   752→  SFX.buy();
   753→  updateHUD(); refreshShop();
   754→}
