   142→        this.speaking = true;

   324→    this.track = (this.TRACKS[name] ? name : this._want) || 'menu';
   325→    this._want = this.track;
   326→    if (!this._ensure()) return;
   327→    this.step = 0;
   328→    this.nextT = Snd.ctx.currentTime + 0.08;
   329→    const T = this.TRACKS[this.track];
   330→    try {
   331→      const t = Snd.ctx.currentTime;
   332→      this.gain.gain.cancelScheduledValues(t);
   333→      this.gain.gain.setValueAtTime(0.0001, t);
   334→      this.gain.gain.exponentialRampToValueAtTime(0.9 * (T.vol || 1), t + 1.6);
   335→    } catch (e) {}
   336→    if (!this.timer) this.timer = setInterval(() => this._tick(), 70);
   337→  },
   338→  /* محو کامل و توقف (صفحه‌ی باخت) */
   339→  fadeStop() {
   340→    if (!this.track && !this.timer) return;
   341→    this._want = null;
   342→    this.track = null;
   343→    if (this.gain && Snd.ctx) {
   344→      try {
   345→        const t = Snd.ctx.currentTime;
   346→        this.gain.gain.cancelScheduledValues(t);
   347→        this.gain.gain.setValueAtTime(Math.max(0.0001, this.gain.gain.value), t);
   348→        this.gain.gain.exponentialRampToValueAtTime(0.0001, t + 1.2);
   349→      } catch (e) {}
   350→    }
   351→    clearTimeout(this._fadeT);
   352→    this._fadeT = setTimeout(() => {
   353→      if (this.timer) { clearInterval(this.timer); this.timer = null; }
   354→    }, 1250);
   355→  },
   356→  stop() {
   357→    clearTimeout(this._fadeT); clearTimeout(this._duckT);
   358→    if (this.timer) { clearInterval(this.timer); this.timer = null; }
   359→    if (this.gain && Snd.ctx) { try { this.gain.gain.value = 0.0001; } catch (e) {} }
   360→    this.track = null; this._want = null;
   361→  },
   362→  /* موسیقی حین گفتار آرام شود */
   363→  duck(dur, level) {
   364→    if (!this.gain || !Snd.ctx) return;
   365→    clearTimeout(this._duckT);
   366→    try {
   367→      const t = Snd.ctx.currentTime;
   368→      this.duckG.gain.cancelScheduledValues(t);
   369→      this.duckG.gain.setValueAtTime(Math.max(0.0001, this.duckG.gain.value), t);
   370→      this.duckG.gain.exponentialRampToValueAtTime(level || 0.4, t + 0.25);
   371→    } catch (e) {}
   372→  },
   373→  duckClear() {
   374→    if (!this.gain || !Snd.ctx) return;
   375→    try {
   376→      const t = Snd.ctx.currentTime;
   377→      this.duckG.gain.cancelScheduledValues(t);
   378→      this.duckG.gain.setValueAtTime(Math.max(0.0001, this.duckG.gain.value), t);
   379→      this.duckG.gain.exponentialRampToValueAtTime(1, t + 0.5);
   380→    } catch (e) {}
   381→  },

   383→  /* زمان‌بند: گام‌های آینده را می‌سازد — (۱۲٫۰) ضد انباشت */
   384→  _tick() {
   385→    if (!Snd.ready || !Snd.ctx || !this.track) return;
   386→    const T = this.TRACKS[this.track];
   387→    if (!T) return;
   388→    const spb = 60 / T.bpm / 4;                       // طول یک گام ۱۶م
   389→    const now = Snd.ctx.currentTime;
   390→    /* اگر زمان‌بند جا افتاد (لگ فریم/تب مخفی)، عقب نمی‌ماند و انباشت نمی‌کند */
   391→    if (this.nextT < now - 0.12) this.nextT = now + 0.05;
   392→    let guard = 0;
   393→    while (this.nextT < now + 0.28 && guard++ < 22) {
   394→      const t = this.nextT, s = this.step % 64, bar = (s / 16) | 0, st = s % 16;
   395→      if (t >= now - 0.02) {
   396→        /* پد: ابتدای هر میزان */
   397→        if (st === 0) this.pad(t, T.chords[bar], spb * 16 * 0.98, T.padVol);
   398→        /* بیس */
   399→        for (const [gs, off, len] of T.bassPat) if (gs === st) this.bass(t, T.roots[bar] + off, spb * len * 0.95, 0.16);
   400→        /* درام */
   401→        if (T.drums.kick && T.drums.kick.includes(st)) this.kick(t, 0.5);
   402→        if (T.drums.snare && T.drums.snare.includes(st)) this.snare(t, 0.2);
   403→        if (T.drums.hat && T.drums.hat.includes(st)) this.nz(t, 0.03, 0.05 + (st % 4 ? 0 : 0.03), 8000);
   404→        /* آرپژ */
   405→        if (T.arp && T.arp.steps.includes(st)) {
   406→          const ch = T.chords[bar];
   407→          this.osc(t, this.nf(ch[(st / 2 + bar) % ch.length] + 12), spb * 1.8, T.arp.vol, 'square', 3200);
   408→        }
   409→        /* فزع‌آوات باس */
   410→        if (T.stab && st === 6) for (const m of T.stab[0]) this.osc(t, this.nf(m), spb * 3, 0.07, 'sawtooth', 2200);
   411→        /* ملودی */
   412→        for (const [mb, ms, note, len, vol] of T.mel) if (mb === bar && ms === st) {
   413→          if (T.melInst === 'pluck') this.pluck(t, note, spb * len, vol);
   414→        }
   415→      }
   416→      this.step++;
   417→      this.nextT += spb;
   418→    }
   419→  },
   420→
   421→  /* نگاشت حالت بازی → قطعه (هر فریم صدا زده می‌شود؛ ارزان است) */
   422→  update() {
   423→    if (!Snd.ready) { if (this._want) this._start(this._want); return; }
   424→    let want = 'menu';
   425→    const m = state.mode;
   426→    if (m === 'start' || m === 'intro') want = 'menu';
   427→    else if (m === 'day' || m === 'intermission') want = 'day';
   428→    else if (m === 'playing') want = state.wave >= FINAL_WAVE ? 'boss' : 'night';
   429→    else if (m === 'transition' || m === 'cutscene') want = 'night';
   430→    else if (m === 'victorycine' || m === 'victory') want = 'victory';
   431→    else if (m === 'defeatcine' || m === 'gameover') want = null;    // سکوت سینمایی (۱۲٫۰)
   432→    else if (m === 'paused') want = this.track || 'menu';
   433→    if (want === null) this.fadeStop();
   434→    else if (this.track !== want || this._want !== want) this.set(want);
   435→    /* (۱۲٫۰) توقف = موسیقی کاملاً خاموش */
   436→    if (this.gain && Snd.ctx) {
   437→      try {
   438→        const target = (m === 'paused') ? 0.0001
   439→          : (this.TRACKS[this.track] ? 0.9 * this.TRACKS[this.track].vol : 0.9);
   440→        const cur = this.gain.gain.value;
   441→        if (Math.abs(cur - target) > 0.05) this.gain.gain.setTargetAtTime(target, Snd.ctx.currentTime, 0.4);
   442→      } catch (e) {}
   443→    }
   444→    /* (۱۲٫۰) امبینتِ پد/باد هم حین توقف خاموش شود */
   445→    if (Snd.musicGain && Snd.ctx && Snd.musicOn !== false) {
   446→      const amb = (m === 'paused') ? 0.0001 : 1;
   447→      if (this._lastAmb !== amb) {
   448→        this._lastAmb = amb;
   449→        try { Snd.musicGain.gain.setTargetAtTime(amb, Snd.ctx.currentTime, 0.3); } catch (e) {}
   450→      }
   451→    }
   452→  },

   467→const DLG = {
   468→  active: false, lines: [], idx: 0, chars: 0, t: 0,
   469→  lineT: 0, voiceDur: 0, holdT: 0, onDone: null, doneFired: false,
   470→
   471→  start(lines, onDone) {
   472→    if (!lines || !lines.length) { if (onDone) onDone(); return; }
   473→    this.lines = lines; this.idx = 0; this.onDone = onDone;
   474→    this.active = true; this.doneFired = false;
   475→    $('dlgBox').classList.remove('hidden');
   476→    this._line();
   477→  },
   478→  _line() {
   479→    const L = this.lines[this.idx];
   480→    this.chars = 0; this.t = 0; this.lineT = 0; this.holdT = 0;
   481→    this.voiceDur = VOICE.speak(L.text, L.who, L.v);
   482→    const w = DLG_WHO[L.who] || DLG_WHO.n;
   483→    $('dlgWho').textContent = w.name;
   484→    $('dlgWho').style.color = w.color;
   485→    $('dlgText').textContent = '';
   486→    $('dlgText').style.color = '';              /* (۱۲٫۰) رنگ از CSS روشن — متن سیاهِ نامرئی رفع شد */
   487→    $('dlgPortrait').style.outlineColor = w.color;
   488→    $('dlgPortrait').style.boxShadow = '0 0 14px ' + w.color + '55, inset 0 0 22px rgba(0,0,0,0.4)';
   489→  },
   490→  update(dt) {
   491→    if (!this.active) return;
   492→    const L = this.lines[this.idx];
   493→    const full = L.text.length;
   494→    this.t += dt;
   495→    /* سرعت ماشین‌نویس: متن همزمان با صدا تمام شود */
   496→    const vDur = this.voiceDur || Math.max(1.2, full / 26);
   497→    const cps = full / Math.max(0.9, vDur * 0.9);
   498→    if (this.chars < full) {
   499→      this.lineT += dt * cps;
   500→      const nc = Math.min(full, Math.floor(this.lineT));
   501→      if (nc !== this.chars) {
   502→        this.chars = nc;
   503→        $('dlgText').textContent = L.text.slice(0, nc);
   504→        if (VOICE.mode === 'blip') VOICE.blip(L.who);       /* صدای شخصیت */
   505→      }
   506→    } else {
   507→      if (VOICE.mode === 'blip') VOICE.finishBlips();
   508→      this.holdT += dt;
   509→      /* بعد از پایان صدا کمی مکث، بعد خط بعد */
   510→      if (!VOICE.speaking && this.holdT > 0.12 && this.t > vDur * 0.55) this._next();
   511→      else if (this.t > vDur + 2.4) this._next();           /* مهلت ایمنی */
   512→    }
   513→    /* پرتره‌ی سخنگو (با دهانِ متحرک حین صدا) */
   514→    drawPortrait($('dlgPortrait'), L.who, VOICE.speaking || this.chars < full, this.t);
   515→  },
   516→  _next() {
   517→    this.idx++;
   518→    if (this.idx >= this.lines.length) this._finish();
   519→    else { SFX.click(); this._line(); }
   520→  },
   521→  _finish() {
   522→    VOICE.stop();
   523→    this.active = false;
   524→    $('dlgBox').classList.add('hidden');
   525→    if (!this.doneFired) { this.doneFired = true; const f = this.onDone; this.onDone = null; if (f) f(); }
   526→  },
   527→  /* کلیک/Space: متن کامل شود یا خط بعد */
   528→  advance() {
   529→    if (!this.active) return false;
   530→    const L = this.lines[this.idx];
   531→    if (this.chars < L.text.length) {
   532→      this.chars = L.text.length;
   533→      this.lineT = this.chars;
   534→      $('dlgText').textContent = L.text;
   535→      if (VOICE.mode === 'blip') VOICE.finishBlips();
   536→    } else this._next();
   537→    return true;
   538→  },
   539→  skip() { if (this.active) { this.doneFired = true; this._finish(); } },
   540→};

   543→   (۱۳٫۰) bark(v, who, text, title, sub, dur) — v = کلید کلیپ صدا */
   544→function bark(v, who, text, title, sub, dur) {
   545→  const d = (who && text && !(typeof SETT !== 'undefined' && !SETT.sfx)) ? VOICE.speak(text, who, v) : 0;
   546→  setAnnounce(title || '', sub || '', dur || Math.max(1.8, d + 0.4));
   547→  return d;
   548→}
