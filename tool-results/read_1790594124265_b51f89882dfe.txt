   204→/* ---------------------- ۳) موتور صدا (Web Audio) ---------------------- */
   205→/* همه‌ی صداها همین‌جا سینتز می‌شوند — هیچ فایل صوتی خارجی وجود ندارد.
   206→   AudioContext فقط بعد از اولین تعامل کاربر ساخته می‌شود (سیاست مرورگرها). */
   207→const Snd = {
   208→  ctx: null, master: null, noiseBuf: null, muted: false, ready: false,
   209→
   210→  init() {
   211→    if (this.ready) return;
   212→    try {
   213→      const AC = window.AudioContext || window.webkitAudioContext;
   214→      if (!AC) return;
   215→      this.ctx = new AC();
   216→      this.master = this.ctx.createGain();
   217→      this.master.gain.value = this.muted ? 0 : 0.5;
   218→      this.master.connect(this.ctx.destination);
   219→      /* بافر نویز سفید یک‌ثانیه‌ای (برای شلیک/برخورد/باد/رعد) */
   220→      this.noiseBuf = this.ctx.createBuffer(1, this.ctx.sampleRate, this.ctx.sampleRate);
   221→      const d = this.noiseBuf.getChannelData(0);
   222→      for (let j = 0; j < d.length; j++) d[j] = Math.random() * 2 - 1;
   223→      this.ready = true;
   224→      this.startAmbient();
   225→    } catch (e) { /* مرورگر بدون صدا */ }
   226→  },
   227→  resume() { if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume(); },
   228→  toggleMute() {
   229→    this.muted = !this.muted;
   230→    if (this.master) this.master.gain.value = this.muted ? 0 : 0.5;
   231→    return this.muted;
   232→  },
   233→
   234→  /* یک نُت: {type,f0,f1,dur,vol,att,delay} */
   235→  tone(o) {
   236→    if (!this.ready || this.muted) return;
   237→    try {
   238→      const t0 = this.ctx.currentTime + (o.delay || 0);
   239→      const osc = this.ctx.createOscillator(), g = this.ctx.createGain();
   240→      osc.type = o.type || 'sine';
   241→      osc.frequency.setValueAtTime(Math.max(1, o.f0), t0);
   242→      if (o.f1) osc.frequency.exponentialRampToValueAtTime(Math.max(1, o.f1), t0 + o.dur);
   243→      g.gain.setValueAtTime(0.0001, t0);
   244→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.008));
   245→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   246→      osc.connect(g); g.connect(this.master);
   247→      osc.start(t0); osc.stop(t0 + o.dur + 0.05);
   248→    } catch (e) {}
   249→  },
   250→  /* نویز فیلترشده: {dur,vol,ftype,freq,q,att,delay} */
   251→  noise(o) {
   252→    if (!this.ready || this.muted) return;
   253→    try {
   254→      const t0 = this.ctx.currentTime + (o.delay || 0);
   255→      const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf; src.loop = true;
   256→      const f = this.ctx.createBiquadFilter();
   257→      f.type = o.ftype || 'lowpass'; f.frequency.value = o.freq || 1000; f.Q.value = o.q || 0.8;
   258→      const g = this.ctx.createGain();
   259→      g.gain.setValueAtTime(0.0001, t0);
   260→      g.gain.exponentialRampToValueAtTime(o.vol, t0 + (o.att || 0.005));
   261→      g.gain.exponentialRampToValueAtTime(0.0001, t0 + o.dur);
   262→      src.connect(f); f.connect(g); g.connect(this.master);
   263→      src.start(t0); src.stop(t0 + o.dur + 0.05);
   264→    } catch (e) {}
   265→  },
   266→
   267→  /* موسیقی محیطیِ شب: درونِ بم + باد + نُت‌های پراکنده‌ی دل‌انگیز */
   268→  startAmbient() {
   269→    try {
   270→      const dg = this.ctx.createGain(); dg.gain.value = 0.045; dg.connect(this.master);
   271→      const o1 = this.ctx.createOscillator(); o1.type = 'triangle'; o1.frequency.value = 55;
   272→      const o2 = this.ctx.createOscillator(); o2.type = 'sine'; o2.frequency.value = 110;
   273→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.09;
   274→      const lg = this.ctx.createGain(); lg.gain.value = 0.02;
   275→      lfo.connect(lg); lg.connect(dg.gain);
   276→      o1.connect(dg); o2.connect(dg);

   707→function tryShoot() {
   708→  if (fireTimer > 0) return;
   709→  const w = WEAPONS[state.weapon];
   710→  /* قدرت «شلیک سریع» نرخ شلیک را تقریباً دوبرابر می‌کند +
   711→     ارتقای «سرعت شلیک» (۹٫۰) هر سطح ۸٪ فاصله‌ی شلیک را کم می‌کند */
   712→  fireTimer = w.rate * (state.powerTimers.rapid > 0 ? 0.55 : 1)
   713→            * (1 - 0.08 * (state.upgrades.speed || 0));
   714→  updateAim();                             // اطمینان از جهت صحیح گلوله
   715→  const m = gunMuzzle(w.barrel);
   716→  for (let i = 0; i < w.pellets; i++) {
   717→    const a = state.aimAngle + (Math.random() - 0.5) * 2 * w.spread;
   718→    const sp = w.spd * rand(0.92, 1.08);
   719→    state.bullets.push({
   720→      x: m.x, y: m.y,
   721→      vx: Math.cos(a) * sp, vy: Math.sin(a) * sp,
   722→      dmg: weaponDmg(w), r: w.pellets > 1 ? 3 : 4, life: w.ttl,
   723→      knock: w.knock,                      // پس‌زنی سلاح (۳٫۰)
   724→      dead: false, pellet: w.pellets > 1,  // ساچمه‌های شاتگان ریزترند
   725→    });
   726→  }
   727→  state.stats.shots++;
   728→  state.fireFlash = 0.07;
   729→  state.muzzleGlow = 0.07;                 // نور دهانه روی محیط (۳٫۰)
   730→  state.recoil = w.pellets > 1 ? 1.6 : 1;  // لگد شاتکن محسوس‌تر است
   731→  SFX[w.sfx]();
   732→  spawnParticles(m.x, m.y, 3, '#ffe9a8', { speed: 90, life: 0.18, size: 2, grav: 0,
   733→    dir: state.aimAngle, spread: Math.PI / 3 });
   734→  spawnShell(state.aimAngle);              // پوکه‌ی پرتاب‌شده
   735→  if (w.pellets > 1) state.shake = Math.min(8, state.shake + 2);
   736→}

   739→function spawnShell(ang) {

   741→  const a = ang + rand(1.6, 2.2);          // زاویه‌ی پرتاب به عقب/بالا
   742→  state.particles.push({
   743→    x: gx, y: gy,

   745→    age: 0, life: rand(0.5, 0.8), size: 2.6, color: '#d9a441', grav: 900,

   750→/* برخورد گلوله با زامبی‌ها + خروج گلوله از صحنه
   751→   (۳٫۰: اول دایره‌ی سر بررسی می‌شود — برخورد سر = ضربه‌ی بحرانی ×۲) */
   752→function updateBullets(dt) {
   753→  for (const b of state.bullets) {
   754→    if (b.dead) continue;
   755→    b.x += b.vx * dt; b.y += b.vy * dt; b.life -= dt;
   756→    /* رد برفیِ گلوله‌ی یخی برجک (۴٫۰) */
   757→    if (b.frost && Math.random() < dt * 34) {
   758→      state.particles.push({ x: b.x, y: b.y, vx: rand(-12, 12), vy: rand(-12, 12),
   759→        age: 0, life: 0.3, size: 1.3, color: 'rgba(205,240,255,0.9)', grav: 30 });
   760→    }
   761→    if (b.y > GROUND_Y + 4) {              // گلوله به زمین خورد
   762→      b.dead = true;
   763→      spawnParticles(b.x, GROUND_Y, 4, '#6a5a43', { speed: 70, life: 0.3, size: 2, grav: 300, up: true });
   764→      continue;
   765→    }
   766→    if (b.x < -40 || b.x > W + 40 || b.y < -40 || b.life <= 0) { b.dead = true; continue; }
   767→    const dir = Math.sign(b.vx) || 1;      // جهت پس‌زنی = جهت حرکت گلوله
   768→    const src = b.tur || 'player';         // منبع گلوله: بازیکن یا برجک (۴٫۰)
   769→    for (const z of state.zombies) {
   770→      if (z.dead) continue;
   771→      const s = (z.radius / 15) * (z.depth || 1);   // مرکز بدن بر اساس اندازه‌ی نوع و عمق
   772→      /* --- سر: ضربه‌ی بحرانی --- */
   773→      const hx = z.x + z.dir * 3 * s, hy = z.y - 36 * s, hr = 7 * s + b.r;
   774→      if ((b.x - hx) * (b.x - hx) + (b.y - hy) * (b.y - hy) < hr * hr) {
   775→        b.dead = true;
   776→        hitZombie(z, b.dmg, b.x, b.y, dir, true, b.knock, src);
   777→        if (b.frost) chillZombie(z);
   778→        break;
   779→      }
   780→      /* --- بدن --- */
   781→      const dx = z.x - b.x, dy = (z.y - 20 * s) - b.y;
   782→      const rr2 = z.radius * (z.depth || 1) + b.r + 3;
   783→      if (dx * dx + dy * dy < rr2 * rr2) {
   784→        b.dead = true;
   785→        hitZombie(z, b.dmg, b.x, b.y, dir, !!b.crit, b.knock, src);
   786→        if (b.frost) chillZombie(z);
   787→        break;
   788→      }
   789→    }
   790→  }
   791→  state.bullets = state.bullets.filter(b => !b.dead);
   792→}
   793→
   794→/* ---------------------- ۸) زامبی‌ها: محاصره‌ی دسته‌جمعی برج ----------------------
   795→   در ۳٫۰ زامبی‌ها دیگر در یک خط نمی‌ایستند: هر طرفِ برج تا ۵ ردیف
   796→   «دیوارِ زامبی» می‌سازند؛ نزدیک‌ترین مهاجم ردیفِ چسبیده به دیوار را می‌گیرد و
   797→   بقیه پشت سرش ردیف می‌شوند. ردیف‌های عقب بالاتر و کوچک‌تر رسم می‌شوند و
   798→   با کشته‌شدنِ ردیف جلو، بقیه فشرده شده و جلو می‌خزند. */
   799→function updateZombies(dt) {
   800→  /* رتبه‌بندی مهاجم‌های فعلی بر اساس نزدیکی به دیوار (هر طرف جداگانه):
   801→     نزدیک‌ترین = ردیف ۰؛ با مرگ ردیف جلو، رتبه‌ها فشرده می‌شوند */
   802→  const rank = { '-1': 0, '1': 0 };
   803→  const attackers = state.zombies.filter(z => !z.dead && z.attacking);
   804→  attackers.sort((a, b) => Math.abs(TOWER.x - a.x) - Math.abs(TOWER.x - b.x));
   805→  for (const z of attackers) {
   806→    z.attackSide = z.x < TOWER.x ? -1 : 1;
   807→    z.lane = Math.min(rank[z.attackSide]++, 4);
   808→  }
   809→
   810→  for (const z of state.zombies) {
   811→    if (z.dead) { z.deathT += dt; continue; }   // انیمیشن مرگ ادامه پیدا کند
   812→    z.hitFlash = Math.max(0, z.hitFlash - dt);
   813→    z.stagger = Math.max(0, z.stagger - dt);    // سرخوردن بعد از ضربه
   814→    z.chill = Math.max(0, (z.chill || 0) - dt); // یخ‌زدگی برجک یخ‌زن (۴٫۰)
   815→
   816→    const dx = TOWER.x - z.x;
   817→    const side = z.x < TOWER.x ? -1 : 1;        // کدام دیوار برج؟
   818→    const lane = z.attacking ? z.lane : Math.min(rank[side], 4);
   819→    const stop = TOWER.baseHalf + z.radius * 0.55 + lane * 15;
   820→
   821→    /* انجماد: همه کُند می‌شوند (پادشاه مقاوم‌تر است) — یخ‌زدگی برجک هم جداگانه اثر می‌کند */
   822→    const slow = state.powerTimers.freeze > 0 ? (z.type === 'boss' ? 0.6 : 0.32) : 1;
   823→    const spd = z.speed * slow * (z.chill > 0 ? 0.55 : 1) * (z.stagger > 0 ? 0.25 : 1);
   824→
   825→    if (Math.abs(dx) > stop) {
   826→      /* راه رفتن به سمت برج (مهاجمِ پس‌زده‌شده هم برمی‌گردد) */
   827→      z.attacking = false;
   828→      z.lane = Math.min(rank[side], 4);
   829→      z.x += Math.sign(dx) * spd * dt;
   830→      z.walkPhase += dt * (4 + spd * 0.06);
   831→    } else {
   832→      /* رسید به دیوار برج: کوبیدن پایه با فاصله‌ی زمانیِ نوعِ خودش */
   833→      z.attacking = true;
   834→      z.attackSide = side;
   835→      z.attackAnim += dt * 5;
   836→      z.biteTimer -= dt;
   837→      if (z.biteTimer <= 0) {
   838→        z.biteTimer = z.biteInt;
   839→        damageTower(z.dmg, z.x);
   840→      }
   841→    }
   842→
   843→    /* عمق تصویری: ردیف‌های عقب‌تر کمی بالاتر ایستاده و کوچک‌تر دیده می‌شوند */
   844→    const yTarget = GROUND_Y - z.lane * 9 + z.yJit;
   845→    z.y += (yTarget - z.y) * Math.min(1, dt * 6);
   846→    z.depth = 1 - z.lane * 0.07;
   847→
   848→    /* غبار گام‌ها (۶٫۰): ذره‌ی ریز پشت پای زامبیِ در حال راه رفتن */
   849→    if (!z.dead && Math.abs(dx) > stop && Math.random() < dt * 2.2) {
   850→      state.particles.push({
   851→        x: z.x - z.dir * 7, y: z.y + 1,
   852→        vx: -z.dir * rand(5, 15), vy: rand(-20, -7),
   853→        age: 0, life: rand(0.25, 0.42), size: rand(1.2, 1.9),
   854→        color: 'rgba(96,84,58,0.45)', grav: 60,
   855→      });
   856→    }
   857→  }
   858→  state.zombies = state.zombies.filter(z => !z.dead || z.deathT < 0.55);
   859→}

   861→/* آسیب گلوله به زامبی (۳٫۰: کریت ×۲ + پس‌زنی + سرخوردن + نشانگر برخورد) */
   862→function hitZombie(z, dmg, bx, by, dir, crit, knock, src) {
   863→  if (z.dead) return;
   864→  if (src) z.killedBy = src;                 // منبع ضربه — برای آمار کشتار برجک‌ها (۴٫۰)
   865→  const total = crit ? dmg * 2 : dmg;
   866→  z.hp -= total;
   867→  z.hitFlash = 0.15;

   971→function placementValid(x, y) {
   972→  if (y < GROUND_Y - 34 || y > GROUND_Y + 16) return false;    // فقط نوار زمین
   973→  if (x < 26 || x > W - 200) return false;                      // داخل کادر و دور از پنل ساخت‌وساز

   975→  for (const t of state.turrets) if (Math.hypot(t.x - x, t.y - y) < 36) return false;
   976→  for (const lx of LAMPS) if (Math.abs(x - lx) < 18) return false;
   977→  for (const tx of [88, 874]) if (Math.abs(x - tx) < 22) return false;   // درخت‌های خشک
   978→  for (const p of PUDDLES) if (Math.abs(x - p.x) < p.w / 2 + 8) return false;
   979→  return true;

  1042→function updateTurrets(dt) {
  1043→  for (const t of state.turrets) {
  1044→    const def = TURRETS[t.type];
  1045→    t.cd = Math.max(0, t.cd - dt);
  1046→    t.flash = Math.max(0, t.flash - dt);
  1047→    t.recoil = Math.max(0, t.recoil - dt * 5);
  1048→    t.hover = !state.placing && Math.abs(state.mouse.x - t.x) < 26 && Math.abs(state.mouse.y - (t.y - 14)) < 30;
  1049→
  1050→    /* نزدیک‌ترین زامبیِ زنده در شعاع برد */
  1051→    let best = null, bd = Infinity;
  1052→    for (const z of state.zombies) {
  1053→      if (z.dead) continue;
  1054→      const d = Math.abs(z.x - t.x);
  1055→      if (d <= def.range && d < bd) { bd = d; best = z; }
  1056→    }
  1057→    if (best) {
  1058→      const s = (best.radius / 15) * (best.depth || 1);
  1059→      const want = Math.atan2((best.y - 20 * s) - (t.y - def.h), best.x - t.x);
  1060→      let diff = want - t.aim;
  1061→      while (diff > Math.PI) diff -= Math.PI * 2;
  1062→      while (diff < -Math.PI) diff += Math.PI * 2;
  1063→      t.aim += diff * Math.min(1, dt * 8);   // چرخش نرم لوله به سمت هدف
  1064→      if (t.cd <= 0 && Math.abs(diff) < 0.35) turretFire(t);
  1065→    }
  1066→    if (t.type === 'gatling') t.spin += dt * (best ? 16 : 3);   // چرخش لوله‌های مسلسل
  1067→    /* برفک‌های محیطی برجک یخ‌زن */
  1068→    if (t.type === 'frost' && Math.random() < dt * 2.5) {
  1069→      state.particles.push({ x: t.x + rand(-13, 13), y: t.y - rand(14, 30),
  1070→        vx: rand(-6, 6), vy: rand(8, 20), age: 0, life: rand(0.6, 1.2),
  1071→        size: rand(1, 1.8), color: 'rgba(200,235,255,0.8)', grav: 10 });
  1072→    }
  1073→  }
  1074→}

  1137→function updateCoins(dt) {
  1138→  const target = coinTarget(); // یک بار در هر فریم محاسبه می‌شود
  1139→  for (const c of state.coinsArr) {
  1140→    if (c.dead) continue;
  1141→    c.t += dt;
  1142→    if (c.phase === 'drop') {
  1143→      /* پرتاب اولیه + جاذبه + وایراستن روی زمین */
  1144→      c.vy += 900 * dt;
  1145→      c.x += c.vx * dt; c.y += c.vy * dt;
  1146→      if (c.y > GROUND_Y - 6) {
  1147→        c.y = GROUND_Y - 6;
  1148→        c.vy *= -0.45; c.vx *= 0.7; c.bounces++;
  1149→        if (Math.abs(c.vy) < 40 || c.bounces >= 2 || c.t > 0.9) c.phase = 'fly';
  1150→      }
  1151→    } else {
  1152→      /* پرو شتاب‌دار به سمت شمارنده سکه در HUD */
  1153→      const dx = target.x - c.x, dy = target.y - c.y;
  1154→      const d = Math.hypot(dx, dy) || 1;
  1155→      c.flySpeed += 1400 * dt;
  1156→      const step = Math.min(d, c.flySpeed * dt);
  1157→      c.x += dx / d * step; c.y += dy / d * step;
  1158→      if (d < 18) { collectCoin(c); continue; }
  1159→    }
  1160→    /* برق کوچکی که گاهی از سکه می‌پرد (۳٫۰) */
  1161→    if (Math.random() < dt * 2.2) {
  1162→      state.particles.push({ x: c.x + rand(-5, 5), y: c.y + rand(-5, 5), vx: 0, vy: -14,
  1163→        age: 0, life: 0.35, size: 1.2, color: '#ffe98a', grav: 0 });
  1164→    }
  1165→    if (c.t > 4) collectCoin(c); // بیمه: سکه گم‌نشده جمع شود
  1166→  }
  1167→  state.coinsArr = state.coinsArr.filter(c => !c.dead);
  1168→}

  1210→  /* شانس پایه ۸٫۵٪ + «شفقت»: اگر خیلی وقت است یکی نیفتاده باشد حتماً می‌افتد */
  1211→  if (state.killsSinceDrop > 24 || Math.random() < 0.085) {
  1212→    state.killsSinceDrop = 0;
  1213→    spawnPowerupObj(pickPowerType(), z.x, z.y - 20);
  1214→  }
  1215→}

  1217→function updatePowerups(dt) {
  1218→  const tx = TOWER.x, ty = TOWER_TOP_Y - 24;   // مقصد: بازیکن روی بام
  1219→  for (const p of state.powerups) {
  1220→    if (p.dead) continue;
  1221→    p.t += dt;
  1222→    if (p.phase === 'drop') {
  1223→      p.vy += 800 * dt;
  1224→      p.x += p.vx * dt; p.y += p.vy * dt;
  1225→      if (p.y > GROUND_Y - 10) {
  1226→        p.y = GROUND_Y - 10;
  1227→        p.vy *= -0.42; p.vx *= 0.65;
  1228→        if (Math.abs(p.vy) < 30 || p.t > 0.8) p.phase = 'wait';
  1229→      }
  1230→    } else if (p.phase === 'wait') {
  1231→      if (p.t > 1.15) p.phase = 'fly';    // مکث کوتاه روی زمین، بعد پرو به بازیکن
  1232→    } else {
  1233→      const dx = tx - p.x, dy = ty - p.y, d = Math.hypot(dx, dy) || 1;
  1234→      p.fs += 900 * dt;
  1235→      const step = Math.min(d, p.fs * dt);
  1236→      p.x += dx / d * step; p.y += dy / d * step;
  1237→      if (d < 18) { applyPowerup(p); p.dead = true; continue; }
  1238→    }
  1239→    if (p.t > POWER_LIFE) p.dead = true;  // اگر کسی نگرفت محو شود
  1240→  }
  1241→  state.powerups = state.powerups.filter(p => !p.dead);
  1242→}

  1244→/* اعمال اثر جعبه‌ی قدرت */
  1245→function applyPowerup(p) {
  1246→  const def = PTYPES[p.type];
  1247→  SFX.power();
  1248→  switch (p.type) {
  1249→    case 'medkit': {
  1250→      const heal = Math.min(def.heal, state.tower.maxHp - state.tower.hp);
  1251→      state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + def.heal);
  1252→      addText(TOWER.x, TOWER_TOP_Y - 44, '+' + faNum(Math.round(heal)) + ' ❤️', '#7be389', 15);
  1253→      spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#7be389', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
  1254→      break;
  1255→    }
  1256→    case 'rapid':
  1257→      state.powerTimers.rapid = def.dur;
  1258→      addText(TOWER.x, TOWER_TOP_Y - 44, 'شلیک سریع ⚡', '#ffd873', 15);
  1259→      spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#ffd873', { speed: 160, life: 0.5, size: 2.4, grav: 250, up: true });
  1260→      break;
  1261→    case 'freeze':
  1262→      state.powerTimers.freeze = def.dur;
  1263→      SFX.freeze();
  1264→      addText(TOWER.x, TOWER_TOP_Y - 44, 'انجماد ❄', '#8fd6ff', 15);
  1265→      spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 18, '#bfe6ff', { speed: 180, life: 0.7, size: 2.4, grav: 120 });
  1266→      break;
  1267→    case 'coin': {
  1268→      state.coins += def.bonus;
  1269→      state.stats.earned += def.bonus;
  1270→      for (const c of state.coinsArr) c.phase = 'fly';   // همه‌ی سکه‌های زمین هم جمع شود
  1271→      addText(TOWER.x, TOWER_TOP_Y - 44, '+' + faNum(def.bonus) + ' 💰', '#f5c542', 15);
  1272→      pulseCoins();
  1273→      break;
  1274→    }
  1275→  }
  1276→  updateHUD();
  1277→}

  1294→function selectWeapon(id) {

  1307→/* خرید ارتقا — در ۳٫۰ وسط مبارزه هم مجاز است (پنل زنده‌ی کنار نقشه) */
  1308→function attemptBuy(kind) {
  1309→  if (!canPlay()) return;                  // فقط وقتی بازی جریان دارد
  1310→  const u = UPGRADES[kind];
  1311→
  1312→  /* --- برجک‌های دفاعی: به حالت انتخاب محل نصب می‌رود (۴٫۰) --- */
  1313→  if (kind === 'gatling' || kind === 'sniper' || kind === 'frost') {
  1314→    startPlacement(kind);
  1315→    return;
  1316→  }
  1317→
  1318→  /* --- سلاح‌های یک‌بار خرید (شاتگان/مسلسل) --- */
  1319→  if (kind === 'shotgun' || kind === 'mg') {
  1320→    if (WEAPONS[kind].owned) return;
  1321→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1322→    state.coins -= u.price;
  1323→    state.weaponSpent += u.price;               // (۸٫۰) مبنای غرامتِ «زیر آوار ماندن»
  1324→    WEAPONS[kind].owned = true;
  1325→    selectWeapon(kind);                    // بلافاصله دستِ کاربر می‌افتد
  1326→    SFX.buy();
  1327→    spawnParticles(TOWER.x, TOWER_TOP_Y - 20, 18, '#ffd873', { speed: 170, life: 0.6, size: 2.6, grav: 300, up: true });
  1328→    updateHUD(); refreshBuildPanel();
  1329→    return;
  1330→  }
  1331→
  1332→  /* --- تعمیر سریع --- */
  1333→  if (kind === 'repair') {
  1334→    if (state.tower.hp >= state.tower.maxHp - 0.5) return;
  1335→    if (state.coins < u.price) { denyBPItem(kind); return; }
  1336→    state.coins -= u.price;
  1337→    const heal = Math.round(state.tower.maxHp * u.frac);
  1338→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + heal);
  1339→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(heal), '#6fce62', 16);
  1340→    spawnParticles(TOWER.x, TOWER_TOP_Y - 30, 14, '#6fce62', { speed: 150, life: 0.55, size: 2.5, grav: 250, up: true });
  1341→    SFX.buy();
  1342→    updateHUD(); refreshBuildPanel();
  1343→    return;
  1344→  }
  1345→
  1346→  /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله) --- */
  1347→  const lvl = state.upgrades[kind];
  1348→  if (lvl >= u.max) return;
  1349→  const price = priceOf(kind);
  1350→  if (state.coins < price) { denyBPItem(kind); return; }
  1351→  state.coins -= price;
  1352→  state.upgrades[kind] = lvl + 1;
  1353→  if (kind === 'weapon' || kind === 'speed') state.weaponSpent += price;   // (۸/۹٫۰) مبنای غرامت
  1354→  if (kind === 'health') {
  1355→    /* هم حداکثر جون زیاد می‌شود هم همین الان ترمیم می‌شود */
  1356→    state.tower.maxHp += u.bonus;
  1357→    state.tower.hp = Math.min(state.tower.maxHp, state.tower.hp + u.bonus);
  1358→    addText(TOWER.x, TOWER_TOP_Y - 40, '+' + faNum(u.bonus), '#6fce62', 15);
  1359→  }
  1360→  if (kind === 'turretUp') {
  1361→    /* جلوه‌ی ارتقای برجک‌ها: جرقه‌ی طلایی روی همه‌ی برجک‌های نصب‌شده (۴٫۰) */
  1362→    for (const t of state.turrets) spawnParticles(t.x, t.y - 18, 10, '#ffd873',
  1363→      { speed: 130, life: 0.5, size: 2.2, grav: 250, up: true });
  1364→    if (state.turrets.length) addText(TOWER.x, TOWER_TOP_Y - 40, 'برجک‌ها قوی‌تر شدند! ⬆', '#ffd873', 14);
  1365→  }
  1366→  if (kind === 'trap' && lvl === 0) {
  1367→    /* نصب اول تله: جلوه بصری نصب */
  1368→    spawnParticles(TOWER.x, GROUND_Y - 8, 16, '#ffcf6e', { speed: 150, life: 0.5, size: 2.5, grav: 300, up: true });
  1369→    state.shake = Math.min(8, state.shake + 4);
  1370→  }
  1371→  SFX.buy();
  1372→  updateHUD(); refreshBuildPanel();
  1373→}
