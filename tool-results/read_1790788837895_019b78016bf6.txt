    34→    34→/* مقیاس پرسپکتیو: پیکسل به‌ازای هر واحد جهان در عمق z */
    35→    35→function drvPPU(z) { return (DRV.CAMD / Math.max(50, z)) * (W / 2); }
    36→    36→/* y صفحه برای نقطه‌ای روی سطح جاده در عمق z */
    37→    37→function drvY(z) { return DRV.HOR + (DRV.CAMD / Math.max(50, z)) * DRV.CAMH * (H / 2); }
    38→    38→/* x صفحه برای نقطه‌ی (worldX, z) — دوربین دنبال بازیکن است */
    39→    39→function drvX(worldX, z) { return W / 2 + drvPPU(z) * (worldX - DRV.px); }
    40→    40→/* نیم‌عرض جاده روی صفحه در عمق z */
    41→    41→function drvRW(z) { return drvPPU(z) * DRV.ROADW; }
    42→    42→
    43→    43→/* مرکز جاده در قطعه‌ی k — دو موج سینوسی (پیچ‌های ملایمِ بلند + کوتاه) */
    44→    44→function roadCX(k) { return 620 * Math.sin(k * 0.055) + 380 * Math.sin(k * 0.0213); }
    45→    45→/* نمونه‌ی خطیِ نرم برای عمق دلخواه */
    46→    46→function roadCXf(z) {
    47→    47→  const kk = z / DRV.SEG, k0 = Math.floor(kk), f = kk - k0;
    48→    48→  return roadCX(k0) + (roadCX(k0 + 1) - roadCX(k0)) * f;
    49→    49→}

    92→    92→  const cfg = MG.cfg, D = DRV;
    93→    93→  TR.fade = Math.max(0, TR.fade - dt * 1.9);
    94→    94→  MG.t2 += dt;

    97→    97→  /* فرمان: px = موقعیت افقی ماشین در جهان */
    98→    98→  D.px += D.steer * 1500 * dt;
    99→    99→  const rc = roadCXf(D.T);
   100→   100→  D.px = clamp(D.px, rc - 1450, rc + 1450);
   101→   101→  D.wheel += D.speed * dt * 0.004;
   102→   102→
   103→   103→  /* گاز / ترمز (لمس: گاز خودکار مگر ترمز) */
   104→   104→  let tgt = 2050;
   105→   105→  const touchAuto = (typeof IS_TOUCH !== 'undefined') && IS_TOUCH;
   106→   106→  if (D.gas || (touchAuto && !D.brake)) tgt = 3350;
   107→   107→  if (D.brake) tgt = 850;
   108→   108→  if (D.offroad > 0.2) tgt = Math.min(tgt, 1150);       // شانه‌ی جاده = شنِ سست

   110→   110→  D.T += D.speed * dt;
   111→   111→  MG.scroll = D.T * 0.12;                                // مترها برای نوار پیشرفت/subs
   112→   112→
   113→   113→  /* بیرون‌زدن از آسفالت؟ (لرزش + غبار + افت سرعت از tgt) */

   115→   115→  D.offroad = edge > (DRV.ROADW - 250)
   116→   116→    ? Math.min(1, D.offroad + dt * 6)
   117→   117→    : Math.max(0, D.offroad - dt * 4);
   118→   118→  if (D.offroad > 0.25) {
   119→   119→    state.shake = Math.max(state.shake, 3.2);

   121→   121→      spawnParticles(480 + rand(-140, 140), 540, 2, 'rgba(120,104,78,0.5)',

   124→   124→  }
   125→   125→
   126→   126→  /* تولید موانع تا جلوترِ دید مفید */

   128→   128→    const k = Math.round(D.nextZ / DRV.SEG);
   129→   129→    const r = Math.random();
   130→   130→    if (r < 0.24) {
   131→   131→      /* دسته‌ی ۲-۳ زامبی کنار هم */
   132→   132→      const base = rand(-0.55, 0.55), n = irand(2, 3);
   133→   133→      for (let i = 0; i < n; i++) {
   134→   134→        const def = DRV_OBSTACLES[irand(0, 1)];
   135→   135→        D.obs.push({ k, wx: roadCX(k) + clamp(base + i * rand(-0.24, 0.24), -0.8, 0.8) * 880,
   136→   136→          type: def.t, hw: def.hw, hit: false });
   137→   137→      }
   138→   138→    } else {
   139→   139→      const def = DRV_OBSTACLES[irand(0, DRV_OBSTACLES.length - 1)];
   140→   140→      D.obs.push({ k, wx: roadCX(k) + rand(-0.74, 0.74) * 880,
   141→   141→        type: def.t, hw: def.hw, hit: false });
   142→   142→    }
   143→   143→    D.nextZ += rand(950, 1600) + D.speed * 0.22;
   144→   144→  }
   145→   145→
   146→   146→  /* برخورد: مانع از صفحه‌ی بازیکن می‌گذرد (z ~ 740..1000) */
   147→   147→  for (const o of D.obs) {
   148→   148→    if (o.hit) continue;
   149→   149→    const z = o.k * DRV.SEG - D.T;
   150→   150→    if (z < 420) { o.hit = true; continue; }             // رد شد
   151→   151→    if (z < 1000 && D.hitCool <= 0 && Math.abs(o.wx - D.px) < 245 + o.hw) {
   152→   152→      driveHit(o);
   153→   153→    }
   154→   154→  }
   155→   155→  /* زادآوری آرایه (حذف رد‌شده‌ها) */
   156→   156→  if (D.obs.length > 90) D.obs = D.obs.filter(o => o.k * DRV.SEG - D.T > 300);
   157→   157→
   158→   158→  /* آیتم مخفی: ستون نورش از دور دیده می‌شود؛ نزدیک‌شدن → هشدار ارشان */
   159→   159→  const HI = D.hidden;
   160→   160→  if (HI && !HI.got) {
   161→   161→    const z = HI.T - D.T;
   162→   162→    const hwx = roadCXf(HI.T) + HI.lane * 880;
   163→   163→    if (!D.itemHinted && z < 6500) {
   164→   164→      D.itemHinted = true;
   165→   165→      transBark({ who: 'a', v: 'd1a2', text: 'اون طرف یه جعبه دیدم! بگیرش!' },
   166→   166→        '— [ارشان]: اون طرف یه جعبه دیدم! بگیرش!', 2.8);
   167→   167→    }
   168→   168→    if (z > 640 && z < 1120 && Math.abs(hwx - D.px) < 330) {
   169→   169→      HI.got = true;
   170→   170→      giveItem(HI.id);
   171→   171→      SFX.csGift();
   172→   172→      setTransSub('💎 آیتم مخفی پیدا شد! (' + (GEAR[HI.id] || RARE[HI.id]).name + ')', 2.6);
   173→   173→      spawnParticles(drvX(hwx, 900), 470, 24,
   174→   174→        (GEAR[HI.id] || RARE[HI.id]).color, { speed: 200, life: 0.9, size: 2.8, grav: 260 });
   175→   175→    }
   176→   176→  }
   177→   177→
   178→   178→  /* خط‌های ارشان (متن + صدای d1a1 برای اولی) */
   179→   179→  for (const s of cfg.subs) {
   180→   180→    if (!s.done && MG.scroll / PXM >= s.atM) {
   181→   181→      s.done = true;
   182→   182→      setTransSub(s.txt, s.dur || 3);
   183→   183→      SFX.peddler();
   184→   184→      if (s.barkV) {
   185→   185→        transBark({ who: 'a', v: s.barkV, text: s.barkT }, s.txt, 3.4);
   186→   186→      }
   187→   187→    }
   188→   188→  }
   189→   189→
   190→   190→  /* دود اگزوز در گاز سنگین */
   191→   191→  if ((D.gas || D.speed > 2900) && Math.random() < dt * 14) {
   192→   192→    spawnParticles(480 + rand(-40, 40), 556, 1, 'rgba(130,130,140,0.4)',
   193→   193→      { speed: 30, life: 0.8, size: 3.4, grav: -20 });
   194→   194→  }
   195→   195→
   196→   196→  /* پایان مسیر → فاز ۳ */
   197→   197→  if (D.T >= D.goalT) mgOutro();
   198→   198→}
