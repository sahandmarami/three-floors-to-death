   242→   242→function icoRR(c, x, y, w, h, r) {
   243→   243→  c.beginPath();
   244→   244→  if (c.roundRect) c.roundRect(x, y, w, h, r);
   245→   245→  else c.rect(x, y, w, h);

   247→   247→function icoStar(c, x, y, r) {
   248→   248→  c.beginPath();
   249→   249→  c.moveTo(x - r, y); c.lineTo(x, y - r * 0.35); c.lineTo(x + r, y);
   250→   250→  c.lineTo(x, y + r * 0.35); c.closePath(); c.fill();
   251→   251→  c.beginPath();
   252→   252→  c.moveTo(x, y - r); c.lineTo(x + r * 0.35, y); c.lineTo(x, y + r);
   253→   253→  c.lineTo(x - r * 0.35, y); c.closePath(); c.fill();
   254→   254→}

   663→   663→function renderPeddlerStock() {
   664→   664→  const box = $('pdStock');
   665→   665→  if (!box) return;
   666→   666→  box.innerHTML = '';
   667→   667→  if (!state.peddler.here) { hidePeddlerPanel(); return; }
   668→   668→  for (const it of state.peddler.stock) {
   669→   669→    if (it.bought) continue;                     // خریداری‌شده‌ها حذف می‌شوند
   670→   670→    const r = RARE[it.id];
   671→   671→    const el = document.createElement('div');
   672→   672→    el.className = 'pd-item' + (state.coins < r.price ? ' poor' : '');
   673→   673→    const cv = document.createElement('canvas');
   674→   674→    cv.width = 80; cv.height = 80;
   675→   675→    paintGearIcon(it.id, cv);
   676→   676→    const info = document.createElement('div');
   677→   677→    info.innerHTML = '<span class="pd-name">' + r.name + '</span><span class="pd-sub">' + r.desc + '</span>';
   678→   678→    const price = document.createElement('span');
   679→   679→    price.className = 'pd-price';
   680→   680→    price.textContent = faNum(r.price);
   681→   681→    el.appendChild(cv); el.appendChild(info); el.appendChild(price);
   682→   682→    el.addEventListener('click', () => buyRare(it));
   683→   683→    box.appendChild(el);
   684→   684→  }
   685→   685→  /* اگر هر دو خریداری شد، پنل بسته شود */
   686→   686→  if (!box.children.length) { hidePeddlerPanel(); return; }
   687→   687→  $('peddlerPanel').classList.remove('hidden');
   688→   688→}

   691→   691→function buyRare(it) {
   692→   692→  if (it.bought || !state.peddler.here || !canPlay()) return;
   693→   693→  const r = RARE[it.id];
   694→   694→  if (state.coins < r.price) { SFX.deny(); return; }
   695→   695→  state.coins -= r.price;
   696→   696→  it.bought = true;
   697→   697→  if (Math.random() < 0.3) {
   698→   698→    /* کلک خوردی! کالای تقلبی — افکت + پیام */
   699→   699→    SFX.scam();

   703→   703→    spawnParticles(150, 180, 18, '#9f8fd0', { speed: 170, life: 0.75, size: 2.6, grav: 280 });
   704→   704→    renderPeddlerStock(); updateHUD(); saveGame();
   705→   705→    return;
   706→   706→  }
   707→   707→  if (!giveItem(it.id)) {
   708→   708→    state.coins += r.price;                      // کوله پر بود → پول برگشت
   709→   709→    it.bought = false;
   710→   710→    return;
   711→   711→  }
   712→   712→  SFX.buy();
   713→   713→  setAnnounce('💎 ' + r.name + ' گرفتی!', 'از کوله‌پشتی مصرفش کن — ' + r.desc, 2.6);
   714→   714→  renderPeddlerStock(); updateHUD(); saveGame();
   715→   715→}
