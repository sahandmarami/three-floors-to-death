    18→/* ---------------------- ۵-۱) ثابت‌های فصل‌بندی ---------------------- */
    19→const WAVES_CH = [4, 6, 8, 10];              // (۱۸٫۰) موج‌های هر فصل — که جلو می‌ریم بیشتر
    20→const WAVES_PER_CH = 8;                    // مبنای مقیاس‌گذاری سختی (نسخه‌های ۱۵-۱۷)
    21→const FINAL_WAVE = 28;                     // ۴+۶+۸+۱۰ = آخرین شبِ کل بازی
    22→
    23→/* شماره‌ی فصل/موجِ درون‌فصلی از شماره‌ی موجِ سراسری (۱۸٫۰: هر فصل طول متفاوت) */
    24→function wavesInChapter(ch) { return WAVES_CH[clamp(ch, 1, 4) - 1]; }
    25→function waveChapter(w) {
    26→  w = clamp(w, 1, FINAL_WAVE);
    27→  let acc = 0;
    28→  for (let c = 1; c <= 4; c++) { acc += WAVES_CH[c - 1]; if (w <= acc) return c; }
    29→  return 4;
    30→}
    31→function waveInChapter(w) {
    32→  w = clamp(w, 1, FINAL_WAVE);
    33→  const ch = waveChapter(w);
    34→  let acc = 0;
    35→  for (let c = 1; c < ch; c++) acc += WAVES_CH[c - 1];
    36→  return w - acc;
    37→}
    38→/* (۱۸٫۰) موجِ «معادل» روی مقیاس ۱..۸ برای سختی/ترکیب/فاصله‌ی تولد —
    39→   اولین موج هر فصل = ۱ و آخرین موج هر فصل = ۸؛ یعنی انتهای هر فصل
    40→   دقیقاً به سختیِ نسخه‌های ۱۵-۱۷ می‌رسد، فقط راهِ رسیدن کوتاهتر است */
    41→function wvEff(w) {
    42→  const ch = waveChapter(w), wv = waveInChapter(w), n = wavesInChapter(ch);
    43→  return n <= 1 ? 1 : 1 + (wv - 1) * 7 / (n - 1);
    44→}
    45→/* سیوهای نسخه‌های ۳۳موجی به‌تناسب به بازه‌ی ۱..۲۸ منتقل می‌شوند */
    46→function compatWave(w) {
    47→  w = Math.round(Number(w) || 1);
    48→  if (w > FINAL_WAVE) w = Math.max(1, Math.round(w * FINAL_WAVE / 33));
    49→  return clamp(w, 1, FINAL_WAVE);
    50→}

   150→/* برچسب موج برای HUD: «فصل ۲ • موج ۳/۸» یا «آخرین شب» */
   151→function waveLabel() {
   152→  if (state.wave >= FINAL_WAVE) return '🌑 آخرین شب';
   153→  const w = clamp(state.wave, 1, FINAL_WAVE);
   154→  return 'فصل ' + faNum(waveChapter(w)) + ' • موج ' + faNum(waveInChapter(w)) + '/' + faNum(wavesInChapter(waveChapter(w)));
   155→}

   157→/* ---------------------- ۵-۲) تولید ترکیب موج‌ها ---------------------- */
   158→/* ترکیب هر موج به‌صورت رویه‌ای از (فصل، شماره‌ی درون‌فصلی) ساخته می‌شود؛
   159→   موج آخرِ کل بازی فقط نگهبان است (زامبی‌هایش را خودش فراخوان می‌کند) */
   160→function buildWaveQueue(w) {
   161→  if (w >= FINAL_WAVE) return ['boss'];
   162→  const ch = waveChapter(w), e = wvEff(w), C = CHAPTERS[ch - 1];
   163→  let n = Math.round((5 + e * 1.85) * C.mulCount);         // (۱۱٫۰) موج‌ها پرتر
   164→  const f = e >= C.unlock.fast ? Math.round(n * 0.32) : 0;
   165→  const t = e >= C.unlock.tank ? Math.max(1, Math.round(n * 0.16)) : 0;
   166→  n = Math.max(3, n - Math.round((f + t) * 0.55));
   167→  const q = [];
   168→  for (let i = 0; i < n; i++) q.push('normal');
   169→  for (let i = 0; i < f; i++) q.push('fast');
   170→  for (let i = 0; i < t; i++) q.push('tank');
   171→  for (let i = q.length - 1; i > 0; i--) {          // شافل فیشر–ییتس
   172→    const j = Math.floor(rand(0, i + 1));
   173→    const tmp = q[i]; q[i] = q[j]; q[j] = tmp;
   174→  }
   175→  return q;
   176→}

   179→   (۱۵٫۰) کمی بازتر — فشار هجوم روی گوشی قابل‌تحمل شد */
   180→function spawnInterval(w) {
   181→  const ch = waveChapter(w), e = wvEff(w);
   182→  return clamp(2.15 - e * 0.12 - (ch - 1) * 0.16, 0.6, 2.15);
   183→}

   246→/* آغاز موج سراسری n — صف ترکیبِ فصل‌محور + زمان‌بندی قطع برق */
   247→function startWave(n) {
   248→  state.wave = n;
   249→  state.mode = 'playing';
   250→  const isFinal = n >= FINAL_WAVE;
   251→  state.spawn = {
   252→    queue: buildWaveQueue(n),
   253→    timer: 1.2,
   254→    interval: spawnInterval(n),
   255→    bossDelay: false,
   256→  };
   257→  /* (۱۰٫۰) زمان‌بندی قطع‌برق‌های تصادفیِ این موج (فصل‌های ۲ به بعد) */
   258→  state.blackout.waveT = 0;
   259→  state.blackout.on = false;
   260→  state.blackout.schedule = [];
   261→  const C = curCh();
   262→  if (!isFinal && C.blackout) {
   263→    const k = irand(C.blackout.min, C.blackout.max);
   264→    for (let i = 0; i < k; i++) {
   265→      state.blackout.schedule.push({ at: rand(5, 24), dur: rand(8, 13), fired: false });
   266→    }
   267→  }
   268→  if (isFinal) {
   269→    $('lockBadge').classList.remove('hidden');    // فروشگاه قفل
   270→    setAnnounce('🌑 آخرین شب', 'فروشگاه قفل است — فقط کوله‌پشتی و برجت', 2.6);
   271→    SFX.roar();
   272→  } else {
   273→    setAnnounce(waveLabel(), 'دفاع کن! 🧟', 2);
   274→    SFX.wave();
   275→  }
   276→  SFX.dusk();
   277→  setBuildPanel(true, true);             // شب: پنل جمع
   278→  if (typeof PED !== 'undefined') PED.leave(false);   // (۱۴٫۰) شب = دستفروع باید رفته باشد
   279→  hidePeddlerPanel();
   280→  $('bpHead').classList.remove('attention');
   281→  saveGame();                             // (۱۵٫۰) سیوِ شروع موج: با باخت، خریدهای همین روز هم می‌ماند
   282→  updateHUD();
   283→}

   285→/* پایان موج: جایزه + مسیر جدید (انتقال فصل / روز / پیروزی) */
   286→function waveComplete() {
   287→  if (state.wave >= FINAL_WAVE) { startVictoryCine(); return; }   // برد واقعی (۱۰٫۰)
   288→  const bonus = 14 + state.wave * 4;      // (۱۵٫۰) جایزه‌ی کمی بیشتر — اقتصاد نفس‌گیرتر
   289→  state.coins += bonus;
   290→  state.stats.earned += bonus;
   291→  addText(TOWER.x, TOWER_TOP_Y - 56, 'جایزه‌ی موج: +' + faNum(bonus) + ' 💰', '#ffd873', 16);
   292→  pulseCoins();
   293→  state.zombies = []; state.bullets = []; state.projectiles = [];
   294→  state.blackout.on = false;              // برق برمی‌گردد
   295→  $('lockBadge').classList.add('hidden');
   296→  /* آخرین موجِ فصل → انتقال سینمایی-قابل‌بازی به فصل بعد (۱۰٫۰) */
   297→  const ch = waveChapter(state.wave), wv = waveInChapter(state.wave);
   298→  if (wv === wavesInChapter(ch) && ch < 4) {
   299→    startTransition(ch);
   300→    return;
   301→  }
   302→  enterDay();
   303→}

   429→/* (۱۳٫۰) لایه‌ی ذخیره‌سازی مقاوم: اگر localStorage در دسترس نباشد
   430→   (مرورگر خصوصی / وب‌ویو / مرورگر درون‌اپ)، بازی نه قفل می‌شود نه
   431→   بی‌صدا سیو را می‌بلعد — حافظه‌ی موقت جایگزین می‌شود + هشدار منو
   432→   + «کد سیو» برای انتقال دستی پیشرفت. */
   433→const STORE = {
   434→  ok: true, mem: {},
   435→  _check() {
   436→    try {
   437→      localStorage.setItem('__ttb3_probe', '1');
   438→      localStorage.removeItem('__ttb3_probe');
   439→      this.ok = true;
   440→    } catch (e) { this.ok = false; }
   441→    return this.ok;
   442→  },
   443→  get(k) {
   444→    if (this.ok) { try { return localStorage.getItem(k); } catch (e) { return (k in this.mem) ? this.mem[k] : null; } }
   445→    return (k in this.mem) ? this.mem[k] : null;
   446→  },
   447→  set(k, v) {
   448→    if (this.ok) {
   449→      try { localStorage.setItem(k, v); return true; } catch (e) { this.ok = false; }
   450→    }
   451→    this.mem[k] = v;
   452→    return this.mem[k] === v;
   453→  },
   454→  del(k) {
   455→    if (this.ok) { try { localStorage.removeItem(k); } catch (e) {} }
   456→    delete this.mem[k];
   457→  },
   458→};

   520→function loadSaveData() {

   524→    if (!CHAPTERS[(d.chapter | 0) - 1]) return null;

   529→  } catch (e) { return null; }

   532→/* اعمال سیو روی state (بعد از ریست کامل) — مقاوم در برابر داده‌های ناقص
   533→   (۱۲٫۰) در پایان، سیو دوباره نوشته می‌شود تا اگر ریستِ داخلی چیزی پاک
   534→   کرد، «ادامه‌ی بازی» هرگز بی‌سیو نماند (رفع باگ دکمه‌ی خاموش‌مانده)
   535→   (۱۳٫۰) کل بدنه در try/finally — حتی اگر داده‌ی سیو ناقص باشد،
   536→   بازنویسی سیو هرگز از دست نمی‌رود */
   537→function applySave(d) {
   538→  try {
   539→    startGame();
   540→    state.chapter = waveChapter(clamp(d.wave | 0, 1, FINAL_WAVE));   // (۱۸٫۰) فصل از خودِ موج
   541→    state.towerVariant = state.chapter - 1;
   542→    state.wave = clamp(d.wave | 0, 1, FINAL_WAVE) - 1;   // enterDay با +1 ادامه می‌دهد
   543→    state.coins = Math.max(0, d.coins | 0);
   544→    state.tower.maxHp = Math.max(100, d.maxHp | 0);
   545→    state.tower.hp = clamp(d.hp | 0, 1, state.tower.maxHp);
   546→    state.upgrades = Object.assign({ weapon: 0, health: 0, trap: 0, turretUp: 0, speed: 0 }, d.upgrades);
   547→    WEAPONS.shotgun.owned = !!(d.owned && d.owned.shotgun);
   548→    WEAPONS.mg.owned = !!(d.owned && d.owned.mg);
   549→    state.weapon = (d.weapon && WEAPONS[d.weapon]) ? d.weapon : 'pistol';
   550→    state.weaponSpent = d.weaponSpent | 0;
   551→    /* (۱۵٫۰) برجک‌های نصب‌شده برمی‌گردند — فقط نوع/جای معتبر، حداکثر MAX_TURRETS */
   552→    state.turrets = (Array.isArray(d.turrets) ? d.turrets : [])
   553→      .filter(tr => tr && TURRETS[tr.t] && isFinite(tr.x) && isFinite(tr.y))
   554→      .slice(0, MAX_TURRETS)
   555→      .map(tr => ({
   556→        type: tr.t, x: clamp(tr.x, 26, W - 200), y: clamp(tr.y, GROUND_Y - 34, GROUND_Y + 16),
   557→        cd: 0.45, aim: tr.x < TOWER.x ? 0 : Math.PI,
   558→        spin: 0, flash: 0, recoil: 0, hover: false, bob: rand(0, 6.28),
   559→      }));
   560→    state.backpack = (d.backpack || []).slice(0, 6)
   561→      .map(id => ((id && GEAR[id]) || (id && RARE[id])) ? { id } : null);
   562→    while (state.backpack.length < 6) state.backpack.push(null);
   563→    state.torch = {
   564→      owned: !!(d.flash && d.flash.owned),
   565→      on: false,
   566→      battery: clamp((d.flash && d.flash.battery) | 0, 0, 100),
   567→    };
   568→    state.buffs = Object.assign({ vest: false, goldAmmo: false, helmet: false }, d.buffs);
   569→    state.peddler = { cd: d.peddlerCd | 0, here: false, stock: [] };
   570→    state.stats = Object.assign({ kills: 0, earned: 0, shots: 0, maxCombo: 0, turKills: 0 }, d.stats);
   571→    state.intermission = 0;
   572→    state.dayness = 1; state.dayTarget = 1;
   573→    updateHUD(); refreshBackpack(); refreshGearHUD(); refreshBuildPanel();
   574→  } finally {
   575→    saveGame();                      // (۱۳٫۰) حتی در صورت خطا هم سیو بازنویسی می‌شود
   576→  }
   577→}
