    98→/* رسم آیکون آیتم روی بوم کوچک: گوشه‌های گرد + اسپرایت نرم */
    99→function paintIcon(kind, cv) {
   100→  cv.width = 88; cv.height = 88;                 // ۲ برابرِ نمایش ۴۴px برای تیزی رتینا
   101→  const c = cv.getContext('2d');
   102→  const im = CUBE_IMGS[kind];
   103→  c.clearRect(0, 0, 88, 88);
   104→  if (im && im.complete && im.naturalWidth) {
   105→    c.imageSmoothingEnabled = true;              // کاشی‌های کاریکاتوریِ نرم
   106→    c.imageSmoothingQuality = 'high';
   107→    /* گوشه‌های گرد مثل کاشی‌های فروشگاه مرجع */
   108→    if (c.roundRect) { c.beginPath(); c.roundRect(2, 2, 84, 84, 10); c.clip(); }
   109→    c.drawImage(im, 2, 2, 84, 84);
   110→    /* خط لبه‌ی ظریف برای جدا شدن از پس‌زمینه‌ی پنل */
   111→    c.strokeStyle = 'rgba(120,170,230,.28)';
   112→    c.lineWidth = 2;
   113→    c.strokeRect(2, 2, 84, 84);
   114→  } else {
   115→    /* جای‌نگهدار تا بارگذاری تصویر */
   116→    c.fillStyle = 'rgba(42,58,94,.85)';
   117→    c.fillRect(18, 18, 52, 52);
   118→    c.strokeStyle = 'rgba(255,216,115,.4)'; c.lineWidth = 2;
   119→    c.strokeRect(18, 18, 52, 52);
   120→  }
   121→}

   131→/* وضعیت لحظه‌ای آیتم برای تولتیپ (قیمت/سطح/مالکیت...) */
   132→function tileStatus(kind) {
   133→  /* وسایل بقی */
   134→  if (kind === 'flashlight') {
   135→    return state.torch.owned
   136→      ? 'خریده شد ✅ (کلید L روشن/خاموش) — باتری: ' + faNum(Math.round(state.torch.battery)) + '٪'
   137→      : 'برای مه و قطع برق لازمش داری';
   138→  }
   139→  if (kind === 'battery' || kind === 'canned' || kind === 'medkit') {
   140→    const g = GEAR[kind];
   141→    return 'داخل کوله: ' + faNum(state.backpack.filter(s => s && s.id === kind).length)
   142→      + ' • قیمت: ' + faNum(g.price) + ' سکه';
   143→  }
   144→  if (kind === 'shotgun' || kind === 'mg') {
   145→    if (!WEAPONS[kind].owned) return 'قفل — بعد از خرید، خودش در دستت می‌آید';
   146→    return state.weapon === kind ? 'همین الان در دستته ✅' : 'خریده شد — کلیک کن تا در دستت بیاید';
   147→  }
   148→  if (kind === 'repair') {
   149→    return state.tower.hp >= state.tower.maxHp - 0.5
   150→      ? 'برج سرِ سلامتی است'
   151→      : 'جون برج: ' + faNum(Math.ceil(state.tower.hp)) + '/' + faNum(state.tower.maxHp);
   152→  }
   153→  if (kind === 'gatling' || kind === 'sniper' || kind === 'frost') {
   154→    const cnt = state.turrets.filter(t => t.type === kind).length;
   155→    return (cnt ? faNum(cnt) + ' عدد فعال • ' : '') + 'قیمت: ' + faNum(TURRETS[kind].price) + ' سکه';
   156→  }
   157→  const u = UPGRADES[kind], lvl = state.upgrades[kind];
   158→  if (lvl >= u.max) return 'حداکثر ✅';
   159→  return 'سطح ' + faNum(lvl) + '/' + faNum(u.max) + ' • قیمت بعدی: ' + faNum(priceOf(kind)) + ' سکه';
   160→}

   170→function positionTip(el) {
   171→  const tip = $('bpTip');
   172→  const wrapR = $('wrap').getBoundingClientRect();
   173→  const r = el.getBoundingClientRect();
   174→  const left = r.left - wrapR.left - tip.offsetWidth - 10;
   175→  const top = clamp(r.top - wrapR.top + r.height / 2 - tip.offsetHeight / 2, 6, H - tip.offsetHeight - 6);
   176→  tip.style.left = Math.max(4, left) + 'px';

   245→/* به‌روزرسانی وضعیت کاشی‌های پنل (۱۲٫۰: تب‌دار + سلاحِ در دست) */
   246→function refreshBuildPanel() {
   247→  for (const d of BUILD_DEFS) {
   248→    const item = $('bp-' + d.kind);
   249→    if (!item) continue;
   250→    const cost = $('bp-' + d.kind + '-cost');
   251→    item.classList.remove('deny');
   252→    /* فقط کاشی‌های تبِ فعال دیده شوند */
   253→    item.style.display = (d.tab === BP_TAB) ? '' : 'none';
   254→
   255→    /* --- سلاح‌های یک‌بار خرید (۱۲٫۰: بعد از خرید = انتخاب/در دست) --- */
   256→    if (d.kind === 'shotgun' || d.kind === 'mg') {
   257→      const owned = WEAPONS[d.kind].owned;
   258→      const armed = owned && state.weapon === d.kind;
   259→      item.classList.toggle('armed', armed);
   260→      item.classList.toggle('max', owned && !armed);
   261→      cost.textContent = armed ? 'در دست' : (owned ? 'انتخاب' : faNum(UPGRADES[d.kind].price));
   262→      item.classList.toggle('poor', !owned && state.coins < UPGRADES[d.kind].price);
   263→      continue;
   264→    }
   265→
   266→    /* --- تعمیر سریع --- */
   267→    if (d.kind === 'repair') {
   268→      const full = state.tower.hp >= state.tower.maxHp - 0.5;
   269→      item.classList.toggle('max', full);
   270→      cost.textContent = full ? '✅' : faNum(UPGRADES.repair.price);
   271→      item.classList.toggle('poor', !full && state.coins < UPGRADES.repair.price);
   272→      continue;
   273→    }
   274→
   275→    /* --- برجک‌های دفاعی: خرید → حالت نصب --- */
   276→    if (d.kind === 'gatling' || d.kind === 'sniper' || d.kind === 'frost') {
   277→      const def = TURRETS[d.kind];
   278→      const maxed = state.turrets.length >= MAX_TURRETS;
   279→      const isPlacing = state.placing && state.placing.type === d.kind;
   280→      item.classList.toggle('placing', !!isPlacing);
   281→      item.classList.toggle('max', maxed && !isPlacing);
   282→      cost.textContent = maxed && !isPlacing ? '✅' : faNum(def.price);
   283→      item.classList.toggle('poor', !maxed && !isPlacing && state.coins < def.price);
   284→      continue;
   285→    }
   286→
   287→    /* --- وسایل بقی: چراغ‌قوه یک‌بار خرید؛ بقیه تکراری‌اند --- */
   288→    if (d.kind === 'flashlight' || d.kind === 'battery' || d.kind === 'canned' || d.kind === 'medkit') {
   289→      const g = GEAR[d.kind];
   290→      const owned = d.kind === 'flashlight' && state.torch.owned;
   291→      item.classList.toggle('max', owned);
   292→      cost.textContent = owned ? '✅' : faNum(g.price);
   293→      item.classList.toggle('poor', !owned && state.coins < g.price);
   294→      continue;
   295→    }
   296→
   297→    /* --- ارتقاهای سطح‌دار (اسلحه/جون/تله/برجک‌ها) --- */
   298→    const u = UPGRADES[d.kind], lvl = state.upgrades[d.kind];
   299→    const maxed = lvl >= u.max;
   300→    item.classList.toggle('max', maxed);
   301→    cost.textContent = maxed ? '✅' : faNum(priceOf(d.kind));
   302→    item.classList.toggle('poor', !maxed && state.coins < priceOf(d.kind));
   303→    /* پین‌های سطح روشن/خاموش */
   304→    const lv = $('bp-' + d.kind + '-lv');
   305→    if (lv) Array.from(lv.children).forEach((pip, i) => pip.classList.toggle('on', i < lvl));
   306→  }
   307→}

   350→/* ---------------------- ۱۶) حلقه اصلی و راه‌اندازی ---------------------- */
   351→let last = performance.now();
   352→function frame(now) {
   353→  const dt = Math.min((now - last) / 1000, 0.05); // سقف dt برای جلوگیری از پرش فیزیک
   354→  last = now;
   355→  animTime += dt;                                // زمان نمایشی همیشه جلو می‌رود
   356→
   357→  /* (۱۲٫۰) حلقه‌ی ضدّهنگ: هر خطای لحظه‌ای (مثلاً وسط رگبار شلیک)
   358→     دیگر کل بازی را فریز نمی‌کند — فریم رد می‌شود و بازی ادامه می‌یابد */
   359→  try {
   360→    /* اسلوموشن سینمایی (۳٫۰): زمان بازی کند می‌شود ولی رندر عادی می‌ماند */
   361→    let dtG = dt;
   362→    if (state.slowmo > 0) {
   363→      dtG = dt * 0.3;
   364→      state.slowmo = Math.max(0, state.slowmo - dt);
   365→    }
   366→
   367→    if (state.mode === 'playing' || state.mode === 'intermission' || state.mode === 'day') update(dtG);
   368→    else if (state.mode === 'cutscene') updateCutscene(dtG);   // رویداد داستانی (۷٫۰)
   369→    else if (state.mode === 'defeatcine') updateDefeatCine(dtG);  // (۱۲٫۰) سینماییِ سقوط برج
   370→    else if (state.mode === 'transition') updateTransition(dtG);   // (۱۰٫۰) انتقال بین فصل‌ها
   371→    else if (state.mode === 'intro') updateIntro(dtG);             // (۱۰٫۰) کات‌سین افتتاحیه
   372→    else if (state.mode === 'victorycine') updateVictoryCine(dtG); // (۱۰٫۰) صحنه‌ی پیروزی
   373→    else if (state.mode === 'gameover' || state.mode === 'victory') updateFx(dtG);
   374→    else if (state.mode === 'start') updateMenuScene(dtG);
   375→    /* حالت paused: هیچ آپدیتی نمی‌شود (انجماد کامل) */
   376→
   377→    /* جلوه‌های محیطی (شهاب و رعدوبرق) در همه‌ی حالت‌ها جریان دارند */
   378→    updateShootingStar(dt);
   379→    updateLightning(dt);
   380→    render();
   381→  } catch (e) {
   382→    /* فریمِ خطادار رد می‌شود — بازی زنده می‌ماند */
   383→  }
   384→  /* (۱۲٫۰) موسیقی در همه‌ی حالت‌ها به‌روز می‌شود: منو/توقف (سکوت)/سینمایی/باخت */
   385→  try { MUSIC.update(); } catch (e) {}
   386→  requestAnimationFrame(frame);
   387→}
