   330→        for (const [gs, off, len] of T.bassPat) if (gs === st) this.bass(t, T.roots[bar] + off, spb * len * 0.95, 0.16);
   331→        /* درام */
   332→        if (T.drums.kick && T.drums.kick.includes(st)) this.kick(t, 0.5);
   333→        if (T.drums.snare && T.drums.snare.includes(st)) this.snare(t, 0.2);
   334→        if (T.drums.hat && T.drums.hat.includes(st)) this.nz(t, 0.03, 0.05 + (st % 4 ? 0 : 0.03), 8000);
   335→        /* آرپژ */
   336→        if (T.arp && T.arp.steps.includes(st)) {
   337→          const ch = T.chords[bar];
   338→          this.osc(t, this.nf(ch[(st / 2 + bar) % ch.length] + 12), spb * 1.8, T.arp.vol, 'square', 3200);
   339→        }
   340→        /* فزع‌آوات باس */
   341→        if (T.stab && st === 6) for (const m of T.stab[0]) this.osc(t, this.nf(m), spb * 3, 0.07, 'sawtooth', 2200);
   342→        /* ملودی */
   343→        for (const [mb, ms, note, len, vol] of T.mel) if (mb === bar && ms === st) {
   344→          if (T.melInst === 'pluck') this.pluck(t, note, spb * len, vol);
   345→        }
   346→      }
   347→      this.step++;
   348→      this.nextT += spb;
   349→    }
   350→  },

   352→  /* نگاشت حالت بازی → قطعه (هر فریم صدا زده می‌شود؛ ارزان است) */
   353→  update() {
   354→    if (!Snd.ready) { if (this._want) this._start(this._want); return; }
   355→    let want = 'menu';
   356→    const m = state.mode;
   357→    if (m === 'start' || m === 'intro') want = 'menu';
   358→    else if (m === 'day' || m === 'intermission') want = 'day';
   359→    else if (m === 'playing') want = state.wave >= FINAL_WAVE ? 'boss' : 'night';
   360→    else if (m === 'transition' || m === 'cutscene') want = 'night';
   361→    else if (m === 'victorycine' || m === 'victory') want = 'victory';
   362→    else if (m === 'defeatcine' || m === 'gameover') want = null;    // سکوت سینمایی (۱۲٫۰)
   363→    else if (m === 'paused') want = this.track || 'menu';
   364→    if (want === null) this.fadeStop();
   365→    else if (this.track !== want || this._want !== want) this.set(want);
   366→    /* (۱۲٫۰) توقف = موسیقی کاملاً خاموش */
   367→    if (this.gain && Snd.ctx) {
   368→      try {
   369→        const target = (m === 'paused') ? 0.0001
   370→          : (this.TRACKS[this.track] ? 0.9 * this.TRACKS[this.track].vol : 0.9);
   371→        const cur = this.gain.gain.value;
   372→        if (Math.abs(cur - target) > 0.05) this.gain.gain.setTargetAtTime(target, Snd.ctx.currentTime, 0.4);
   373→      } catch (e) {}
   374→    }
   375→    /* (۱۲٫۰) امبینتِ پد/باد هم حین توقف خاموش شود */
   376→    if (Snd.musicGain && Snd.ctx && Snd.musicOn !== false) {
   377→      const amb = (m === 'paused') ? 0.0001 : 1;
   378→      if (this._lastAmb !== amb) {
   379→        this._lastAmb = amb;
   380→        try { Snd.musicGain.gain.setTargetAtTime(amb, Snd.ctx.currentTime, 0.3); } catch (e) {}
   381→      }
   382→    }
   383→  },

   398→const DLG = {
   399→  active: false, lines: [], idx: 0, chars: 0, t: 0,
   400→  lineT: 0, voiceDur: 0, holdT: 0, onDone: null, doneFired: false,
   401→
   402→  start(lines, onDone) {
   403→    if (!lines || !lines.length) { if (onDone) onDone(); return; }
   404→    this.lines = lines; this.idx = 0; this.onDone = onDone;
   405→    this.active = true; this.doneFired = false;
   406→    $('dlgBox').classList.remove('hidden');
   407→    this._line();
   408→  },
   409→  _line() {
   410→    const L = this.lines[this.idx];
   411→    this.chars = 0; this.t = 0; this.lineT = 0; this.holdT = 0;
   412→    this.voiceDur = VOICE.speak(L.text, L.who);
   413→    const w = DLG_WHO[L.who] || DLG_WHO.n;
   414→    $('dlgWho').textContent = w.name;
   415→    $('dlgWho').style.color = w.color;
   416→    $('dlgText').textContent = '';
   417→    $('dlgText').style.color = '';              /* (۱۲٫۰) رنگ از CSS روشن — متن سیاهِ نامرئی رفع شد */
   418→    $('dlgPortrait').style.outlineColor = w.color;
   419→    $('dlgPortrait').style.boxShadow = '0 0 14px ' + w.color + '55, inset 0 0 22px rgba(0,0,0,0.4)';
   420→  },
   421→  update(dt) {
   422→    if (!this.active) return;
   423→    const L = this.lines[this.idx];
   424→    const full = L.text.length;
   425→    this.t += dt;
   426→    /* سرعت ماشین‌نویس: متن همزمان با صدا تمام شود */
   427→    const vDur = this.voiceDur || Math.max(1.2, full / 26);
   428→    const cps = full / Math.max(0.9, vDur * 0.9);
   429→    if (this.chars < full) {
   430→      this.lineT += dt * cps;
   431→      const nc = Math.min(full, Math.floor(this.lineT));
   432→      if (nc !== this.chars) {
   433→        this.chars = nc;
   434→        $('dlgText').textContent = L.text.slice(0, nc);
   435→        if (VOICE.mode === 'blip') VOICE.blip(L.who);       /* صدای شخصیت */
   436→      }
   437→    } else {
   438→      if (VOICE.mode === 'blip') VOICE.finishBlips();
   439→      this.holdT += dt;
   440→      /* بعد از پایان صدا کمی مکث، بعد خط بعد */
   441→      if (!VOICE.speaking && this.holdT > 0.12 && this.t > vDur * 0.55) this._next();
   442→      else if (this.t > vDur + 2.4) this._next();           /* مهلت ایمنی */
   443→    }
   444→    /* پرتره‌ی سخنگو (با دهانِ متحرک حین صدا) */
   445→    drawPortrait($('dlgPortrait'), L.who, VOICE.speaking || this.chars < full, this.t);
   446→  },
   447→  _next() {
   448→    this.idx++;
   449→    if (this.idx >= this.lines.length) this._finish();
   450→    else { SFX.click(); this._line(); }
   451→  },
   452→  _finish() {
   453→    VOICE.stop();
   454→    this.active = false;
   455→    $('dlgBox').classList.add('hidden');
   456→    if (!this.doneFired) { this.doneFired = true; const f = this.onDone; this.onDone = null; if (f) f(); }
   457→  },
   458→  /* کلیک/Space: متن کامل شود یا خط بعد */
   459→  advance() {
   460→    if (!this.active) return false;
   461→    const L = this.lines[this.idx];
   462→    if (this.chars < L.text.length) {
   463→      this.chars = L.text.length;
   464→      this.lineT = this.chars;
   465→      $('dlgText').textContent = L.text;
   466→      if (VOICE.mode === 'blip') VOICE.finishBlips();
   467→    } else this._next();
   468→    return true;
   469→  },
   470→  skip() { if (this.active) { this.doneFired = true; this._finish(); } },
   471→};

   474→   (۱۲٫۰) bark(who, text, title, sub, dur) — متن با صدای خود شخصیت خوانده می‌شود */
   475→function bark(who, text, title, sub, dur) {

   477→  setAnnounce(title || '', sub || '', dur || Math.max(1.8, d + 0.4));
   478→  return d;
   479→}
