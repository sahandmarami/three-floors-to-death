    92→function updateShelterCine(dt) {
    93→  if (!SHEL) return;
    94→  SHEL.t += dt;
    95→  DLG.update(dt);
    96→  state.shake = Math.max(0, state.shake - dt * 22);
    97→  state.flash = Math.max(0, state.flash - dt * 2.2);
    98→  updateParticles(dt);
    99→  updateTexts(dt);
   100→  if (SHEL.subT > 0) SHEL.subT -= dt;
   101→  const ph = SHEL.phase;
   102→
   103→  if (ph === 'walk') {
   104→    setTimedShelterSubs(SHEL_SUBS.walk);
   105→    /* صدای ورود (اولین بار) — صدا + زیرنویسِ خودِ سینمایی (بدون اعلان) */
   106→    if (!SHEL.hello && SHEL.t > 0.4) {
   107→      SHEL.hello = true;
   108→      const d1 = VOICE.speak('اونجاست! درِ پناهگاه رو می‌بینم! زود باش!', 'a', 'sh1a');
   109→      SHEL.sub = '— [ارشان]: اونجاست! درِ پناهگاه رو می‌بینم! زود باش!';
   110→      SHEL.subT = Math.max(2.2, d1 + 0.3);
   111→    }
   112→    if (!SHEL.narr && SHEL.t > 3.4) {
   113→      SHEL.narr = true;
   114→      const d2 = VOICE.speak('دری فولادی در دلِ تپه... و صدایی که می‌گه هنوز جایی برای موندن هست.', 'n', 'sh2n');
   115→      SHEL.sub = 'دری فولادی در دلِ تپه... و صدایی که می‌گه هنوز جایی برای موندن هست.';
   116→      SHEL.subT = Math.max(2.6, d2 + 0.3);
   117→    }
   118→    if (SHEL.t >= 6.2) { SHEL.phase = 'gate'; SHEL.t = 0; SFX.csRumble(); }
   119→  } else if (ph === 'gate') {
   120→    setTimedShelterSubs(SHEL_SUBS.gate);
   121→    if (!SHEL.doorSfx && SHEL.t > 0.3) {
   122→      SHEL.doorSfx = true;
   123→      SFX.csEngine(); state.shake = 5;
   124→    }
   125→    /* نورِ ریخته از در وقتی باز می‌شود */
   126→    if (SHEL.t > 1.0 && Math.random() < dt * 14) {
   127→      spawnParticles(700 + rand(-60, 60), 330 + rand(-30, 30), 1, 'rgba(255,226,168,0.5)',
   128→        { speed: 16, life: 1.6, size: 2.4, grav: -8 });
   129→    }
   130→    if (SHEL.t >= 4.0) { SHEL.phase = 'inside'; SHEL.t = 0; }
   131→  } else if (ph === 'inside') {
   132→    /* گفتگوی پناهگاه (کمی بعد از ورود) */
   133→    if (!SHEL.dlgStarted && SHEL.t > 0.8) {
   134→      SHEL.dlgStarted = true;
   135→      DLG.start(SHEL_DLGS, () => { if (SHEL) SHEL.dlgDone = true; });
   136→    }
   137→    /* چراغ‌های سقفی سوسوزن + بخارِ تنفس سرد */
   138→    if (Math.random() < dt * 3) {
   139→      spawnParticles(rand(80, 880), 300, 1, 'rgba(255,226,168,0.22)',
   140→        { speed: 6, life: 2.2, size: 2, grav: 6 });
   141→    }
   142→    if (SHEL.dlgDone && SHEL.t > 1.0 && !SHEL.alarmStarted) {
   143→      SHEL.alarmStarted = true;
   144→      SHEL.phase = 'alarm'; SHEL.t = 0;
   145→      SHEL.dlgStarted = false; SHEL.dlgDone = false;
   146→      SFX.alarm();
   147→      state.shake = 8;
   148→    }
   149→  } else if (ph === 'alarm') {
   150→    /* آژیر: چراغ قرمز چرخان + لرزش + ریزش غبار */
   151→    state.shake = Math.max(state.shake, 2.5);
   152→    if (Math.random() < dt * 16) {
   153→      spawnParticles(rand(60, 900), 240, 1, 'rgba(160,140,110,0.4)',
   154→        { speed: 20, life: 1.4, size: 2, grav: 60 });
   155→    }
   156→    if (Math.sin(SHEL.t * 4.4) > 0.92 && !SHEL._al2) { SHEL._al2 = true; SFX.alarm(); }
   157→    if (Math.sin(SHEL.t * 4.4) < -0.92) SHEL._al2 = false;
   158→    if (!SHEL.dlgStarted && SHEL.t > 0.7) {
   159→      SHEL.dlgStarted = true;
   160→      DLG.start(SHEL_DLGS2, () => { if (SHEL) SHEL.dlgDone = true; });
   161→    }
   162→    /* جمله‌ی آخرِ راوی بعد از گفتگو → محو به «ادامه دارد» */
   163→    if (SHEL.dlgDone && SHEL.t > 1.2) {
   164→      if (!SHEL.lastN) {
   165→        SHEL.lastN = true;
   166→        try { VOICE.speak('و شب، وقتی همه خواب بودند... داستان هنوز تمام نشده بود.', 'n', 'sh11n'); } catch (e) {}
   167→      }
   168→      SHEL.fade = clamp((SHEL.t - 3.6) / 1.6, 0, 1);
   169→      if (SHEL.t >= 5.4) startTbcScreen();
   170→    }
   171→  } else if (ph === 'tbc') {
   172→    /* تایپ تدریجی «این داستان ادامه دارد...» */
   173→    const full = 'این داستان ادامه دارد...';
   174→    if (SHEL.typed < full.length) {
   175→      SHEL.typeAcc = (SHEL.typeAcc || 0) + dt * 7.5;
   176→      while (SHEL.typeAcc >= 1 && SHEL.typed < full.length) {
   177→        SHEL.typeAcc -= 1;
   178→        SHEL.typed++;
   179→        if (SHEL.typed % 3 === 0) SFX.click();
   180→      }
   181→      const el = $('tbcTitle');
   182→      if (el) el.innerHTML = full.slice(0, SHEL.typed) + '<span class="tbccaret">▌</span>';
   183→    } else {
   184→      if (!SHEL.tbcDone) {
   185→        SHEL.tbcDone = true;
   186→        const el = $('tbcTitle');
   187→        if (el) el.textContent = full;
   188→        $('tbcSub').classList.add('on');
   189→        $('tbcTease').classList.add('on');
   190→        $('tbcRow').classList.add('on');
   191→        try { VOICE.speak(full, 'n', 'tbc1n'); } catch (e) {}
   192→        SFX.heartbeat();
   193→      }
   194→      SHEL.teaseT = (SHEL.teaseT || 0) + dt;
   195→      if (SHEL.teaseT > 2.2) { SHEL.teaseT = 0; SFX.heartbeat(); }
   196→    }
   197→  }
   198→}

   205→    }
