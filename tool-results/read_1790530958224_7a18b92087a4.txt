   257→/* هدیه‌ی راننده: سکه‌ی اضافه + آنلاک زودهنگام سلاح (idempotent) */
   258→function cutsceneApplyGift() {
   259→  if (!CS || CS.giftApplied) return;
   260→  CS.giftApplied = true;
   261→  const g = CS.gift;
   262→  state.coins += g.coins;
   263→  state.stats.earned += g.coins;
   264→  let extra = '';
   265→  if (g.weapon && !WEAPONS[g.weapon].owned) {
   266→    WEAPONS[g.weapon].owned = true;
   267→    extra = ' + ' + WEAPONS[g.weapon].name;
   268→  } else if (!WEAPONS.shotgun.owned) {
   269→    WEAPONS.shotgun.owned = true;
   270→    extra = ' + ' + WEAPONS.shotgun.name;
   271→  } else if (!WEAPONS.mg.owned) {
   272→    WEAPONS.mg.owned = true;
   273→    extra = ' + ' + WEAPONS.mg.name;
   274→  }
   275→  CS.giftText = 'هدیه‌ی راننده: +' + faNum(g.coins) + ' سکه' + extra;
   276→  SFX.csGift();
   277→  pulseCoins();
   278→  refreshWeaponBar();
   279→  refreshBuildPanel();
   280→  updateHUD();
   281→}

   283→/* پایان کات‌سین: ورود به «روزِ برج جدید» — همه‌ی پیشرفت‌ها حفظ شده‌اند */
   284→function cutsceneFinish() {
   285→  if (!CS) return;
   286→  cutsceneApplyGift();                       // اطمینان: هدیه حتماً داده شود
   287→  const giftSub = CS.giftText;
   288→  const variant = CS.variant;
   289→  CS = null;
   290→  state.mode = 'day';
   291→  state.towerVariant = variant;              // برج دوم + شهر سوخته
   292→  state.cutsceneDone = true;
   293→  state.dayTarget = 1;
   294→  state.dayness = Math.max(state.dayness, 0.85);
   295→  $('wrap').classList.remove('cs-mode');
   296→  $('skipCs').classList.add('hidden');
   297→  hideAnnounce();
   298→  showDayPanel();                            // «شب شود — موج ۶»
   299→  setBuildPanel(false, true);
   300→  $('bpHead').classList.add('attention');
   301→  setAnnounce('🌅 برج بعدی!',
   302→    giftSub + ' — موج ' + faNum(state.wave + 1) + ' در راه است', 3.4);
   303→  SFX.morning();
   304→  updateHUD();
   305→}

   321→/* ------------------ به‌روزرسانی کات‌سین (هر فریم) ------------------ */
   322→function updateCutscene(dt) {
   323→  if (!CS || CS.paused) return;
   324→  const cs = CS;
   325→  const t = (cs.t += dt);
   326→  const D = cs.draw;
   327→
   328→  /* زوال جلوه‌های عمومی */
   329→  state.shake = Math.max(0, state.shake - dt * 22);
   330→  state.flash = Math.max(0, state.flash - dt * 2.2);
   331→
   332→  /* ---- مرحله ۱: لرزش شدید + ترک‌خوردن پیش‌رونده ---- */
   333→  if (t < CS_T.shakeEnd) {
   334→    D.cracks = t / CS_T.shakeEnd;
   335→    state.shake = 7 + 6 * Math.abs(Math.sin(t * 18));
   336→    if (Math.random() < dt * 20) {           // گردوغاک چکه‌کن از ترک‌ها
   337→      spawnParticles(TOWER.x + rand(-52, 52), GROUND_Y - rand(20, 120), 1, '#9c8a6d',
   338→        { speed: 40, life: 0.7, size: 1.8, grav: 300 });
   339→    }
   340→    SFX.csRumble();
   341→    if (Math.random() < dt * 1.5) SFX.csCrack();
   342→  }
   343→
   344→  /* ---- مرحله ۲: فروریختن طبقه‌ی اول و دوم (آجر/چوب) ---- */
   345→  else if (t < CS_T.colEnd) {
   346→    const p = clamp((t - CS_T.shakeEnd) / (CS_T.colEnd - CS_T.shakeEnd), 0, 1);
   347→    if (!cs.boomed) { cs.boomed = true; SFX.csBoom(); state.shake = 14; }
   348→    D.cracks = 1;
   349→    D.floors[0].a = clamp(1 - p * 1.5, 0, 1);      // طبقه‌ی اول: فرورفتن و محو
   350→    D.floors[0].dy = p * p * 46;
   351→    D.floors[0].rot = p * 0.05;
   352→    const p2 = clamp((p - 0.12) / 0.88, 0, 1);     // طبقه‌ی دوم: کمی دیرتر
   353→    D.floors[1].a = clamp(1 - p2 * 1.5, 0, 1);
   354→    D.floors[1].dy = p2 * p2 * 56;
   355→    D.floors[1].rot = -p2 * 0.06;
   356→    D.rubble = clamp(p * 0.55, 0, 0.55);           // توده‌ی آوار رشد می‌کند
   357→    state.shake = Math.max(state.shake, 5 + 7 * (1 - p));
   358→    /* موج‌های انفجار آجر در چهار آستانه‌ی پیشرفت */
   359→    const bn = [0.04, 0.3, 0.6, 0.9].filter(b => p >= b).length;
   360→    while (cs.bursts < bn) { csBrickBurst(cs.bursts); cs.bursts++; }
   361→  }
   362→
   363→  /* ---- مرحله ۳: سقوط تخته‌بام (طبقه‌ی سوم) و خرد شدن روی زمین ---- */
   364→  const cf = clamp((t - CS_T.chunkDrop) / CS_T.chunkDur, 0, 1);
   365→  if (cf > 0 && t < CS_T.fadeOut) {
   366→    if (cf < 1) {
   367→      D.floors[2].a = 1;
   368→      D.floors[2].dy = cf * cf * (GROUND_Y - TOWER_TOP_Y - 22);   // سقوط شتاب‌دار
   369→      D.floors[2].rot = -cf * 0.16;                               // کمی کج می‌شود
   370→    } else if (!cs.chunkHit) {
   371→      cs.chunkHit = true;
   372→      D.floors[2].a = 0;                 // تخته‌بام خورد و در آوار رفت
   373→      D.rubble = 1;
   374→      state.shake = 12;
   375→      SFX.csLand();
   376→      spawnParticles(TOWER.x, GROUND_Y - 10, 26, '#8a6f52', { speed: 240, life: 0.8, size: 3.4, grav: 560, up: true });
   377→      spawnParticles(TOWER.x, GROUND_Y - 4, 18, '#6e5a44', { speed: 170, life: 1.2, size: 4.2, grav: 260, up: true });
   378→    }
   379→  }
   380→
   381→  /* ---- ماشین نجات: ورود سریع از چپ، عبور از میان زامبی‌ها ---- */
   382→  if (t >= CS_T.carEnter && t < CS_T.fadeOut + 0.6) {
   383→    if (!cs.car) {
   384→      cs.car = { x: -160, y: GROUND_Y + 14, spin: 0, kick: 0, hand: 1.4 };
   385→      SFX.csEngine();
   386→    }
   387→    /* حرکت قطعی از زمان (قابل تست و پرش زمانی) */
   388→    cs.car.x = -160 + CS_T.carSpd * (t - CS_T.carEnter);
   389→    cs.car.spin += (CS_T.carSpd / 13) * dt;
   390→    cs.car.kick = Math.max(0, cs.car.kick - dt * 4);
   391→    cs.car.hand = Math.max(0, cs.car.hand - dt);   // اول کار دست می‌دهد
   392→    SFX.csEngine();
   393→    if (Math.random() < dt * 26) {                 // دم اگزوز + غبار چرخ‌ها
   394→      spawnParticles(cs.car.x - 80, cs.car.y - 8, 1, 'rgba(190,180,168,0.5)',
   395→        { speed: 34, life: 0.55, size: 2.6, grav: -26, dir: Math.PI, spread: 0.7 });
   396→    }
   397→    /* تصادف ساده: هر زامبیِ جلوی مسیر کنار پرتاب می‌شود */
   398→    for (const z of cs.zombies) {
   399→      if (!z.hit && Math.abs(z.x - cs.car.x) < 76) csKnockZombie(z);
   400→    }
   401→  } else if (t >= CS_T.fadeOut + 0.6 && cs.car) {
   402→    cs.car = null;                                 // ماشین صحنه را ترک کرد
   403→  }
   404→
   405→  /* ---- سقوط سینمایی بازیکن و نشستن روی ماشین ---- */
   406→  const pf = clamp((t - CS_T.playerDrop) / CS_T.fallDur, 0, 1);
   407→  if (pf > 0 && !cs.caught) {
   408→    const y0 = TOWER_TOP_Y + 30;                   // لبه‌ی بامِ در حال سقوط
   409→    const y1 = GROUND_Y + 14 - 54;                 // سقف ماشین
   410→    D.player = {
   411→      x: TOWER.x + 8 + pf * (CS_T.catchDX - 8),    // کمی به سمت ماشین می‌لغزد
   412→      y: y0 + (y1 - y0) * pf * pf,                 // شتاب‌دار (جاذبه)
   413→      rot: pf * 0.62,                              // چرخش تدریجی حین سقوط
   414→      aim: -2.35,                                  // اسلحه بالا (چنگ زدن به هوا)
   415→    };
   416→    if (!cs.whistled) { cs.whistled = true; SFX.csWhistle(); }
   417→    if (pf >= 1) {                                 // رسید به ماشین — سوار شد
   418→      cs.caught = true;
   419→      state.shake = 4;
   420→      SFX.csThud();
   421→      spawnParticles(D.player.x, D.player.y, 9, '#ffd873',
   422→        { speed: 120, life: 0.5, size: 2.4, grav: 320, up: true });
   423→    }
   424→  }
   425→  if (cs.caught && cs.car) {
   426→    /* سوار شدن: بازیکن روی سقف ماشین، کمی به عقب خم (باد سرعت) */
   427→    D.player = { x: cs.car.x + CS_T.catchDX, y: cs.car.y - 54, rot: -0.07, aim: -0.5 };
   428→  }
   429→
   430→  /* ---- هدیه‌ی راننده وسط راه ---- */
   431→  if (t >= CS_T.giftT) {
   432→    cutsceneApplyGift();
   433→    if (CS && CS.car && CS.giftApplied && !cs.giftTextShown) {
   434→      cs.giftTextShown = true;
   435→      addText(cs.car.x, cs.car.y - 92, '🎁 +' + faNum(cs.gift.coins), '#ffd873', 17);
   436→    }
   437→  }
   438→
   439→  /* ---- زامبی‌های صحنه: خزش به سمت برج یا فیزیک پرتاب ---- */
   440→  for (const z of cs.zombies) {
   441→    if (z.hit) {
   442→      z.vy += 900 * dt;
   443→      z.x += z.vx * dt; z.y += z.vy * dt; z.rot += z.vr * dt;
   444→      if (z.y > z.groundY) {                       // فرود روی زمین
   445→        z.y = z.groundY;
   446→        if (!z.landed) {
   447→          z.landed = true;
   448→          addDecal(z.x, true);
   449→          spawnParticles(z.x, z.y, 6, '#7a2f2f', { speed: 110, life: 0.4, size: 2.2, grav: 400 });
   450→        }
   451→        z.vx *= 0.8; z.vr = 0;
   452→        z.rot += (Math.sign(z.rot || 1) * 1.45 - z.rot) * Math.min(1, dt * 9);   // خوابیدن صاف
   453→      }
   454→    } else {
   455→      const dir = Math.sign(TOWER.x - z.x) || 1;
   456→      if (Math.abs(TOWER.x - z.x) > 155) { z.x += dir * z.speed * dt; z.walkPhase += dt * 5; }
   457→      else z.walkPhase += dt * 2.2;                // دور آوار جمع می‌شوند
   458→    }
   459→  }
   460→
   461→  /* ---- محو به سیاهی / سیاهی / بازگشت صحنه ---- */
   462→  cs.fade = t < CS_T.fadeOut ? 0
   463→    : t < CS_T.black ? clamp((t - CS_T.fadeOut) / CS_T.fadeDur, 0, 1)
   464→    : t < CS_T.fadeIn ? 1
   465→    : clamp(1 - (t - CS_T.fadeIn) / CS_T.fadeInDur, 0, 1);
   466→
   467→  /* تعویض صحنه در دلِ سیاهی: برج دوم + شهر سوخته + صحنه‌ی تمیز */
   468→  if (t >= CS_T.switchT && !cs.switched) {
   469→    cs.switched = true;
   470→    state.towerVariant = cs.variant;
   471→    state.cutsceneDone = true;
   472→    state.dayness = 0.3;
   473→    state.dayTarget = 1;                    // صبحِ شهر سوخته
   474→    cs.zombies = [];
   475→    state.particles = []; state.texts = []; state.zaps = []; state.decals = [];
   476→    cs.car = null;
   477→    D.floors = [{ a: 1, dy: 0, rot: 0 }, { a: 1, dy: 0, rot: 0 }, { a: 1, dy: 0, rot: 0 }];
   478→    D.cracks = 0; D.rubble = 0; D.player = null;   // بازیکن روی برج جدید
   479→  }
   480→
   481→  /* ---- پایان کات‌سین → روزِ برج جدید ---- */
   482→  if (t >= CS_T.end) { cutsceneFinish(); return; }
   483→
   484→  /* ذرات/متن‌های کات‌سین همیشه جریان دارند */
   485→  updateParticles(dt); updateTexts(dt); updateZaps(dt);
   486→}

   647→    ctx.fillRect(w2.x, w2.y, 3.4, 5);
   648→  }
   649→  /* کانون‌های آتش + دودِ بالا رونده — حتی روز هم به‌روشنی دیده می‌شوند
   650→     (شهر هنوز می‌سوزد؛ آتشِ روز کمی کمرنگ‌تر از شب است) */
   651→  const fireA = 0.78 + 0.22 * (1 - d);
   652→  for (const f of RUIN_FIRES) {
   653→    const fl = 0.55 + 0.45 * Math.sin(animTime * 5 + f.ph);
   654→    const g2 = ctx.createRadialGradient(f.x, f.y, 1, f.x, f.y, 17);
   655→    g2.addColorStop(0, 'rgba(255,150,66,' + (0.8 * fireA * fl).toFixed(2) + ')');
   656→    g2.addColorStop(0.5, 'rgba(255,110,40,' + (0.4 * fireA * fl).toFixed(2) + ')');
   657→    g2.addColorStop(1, 'rgba(255,110,40,0)');
   658→    ctx.fillStyle = g2; circle(f.x, f.y, 17);
   659→    ctx.fillStyle = 'rgba(255,200,120,' + (0.85 * fireA * fl).toFixed(2) + ')';
   660→    circle(f.x, f.y, 2.4);                        // هسته‌ی روشن شعله
   661→    const sy = f.y - 13 - (animTime * 7 + f.ph * 10) % 26;
   662→    ctx.fillStyle = 'rgba(64,50,52,' + (0.45 * fireA * fl).toFixed(2) + ')';
   663→    ctx.beginPath(); ctx.ellipse(f.x + Math.sin(animTime + f.ph) * 4, sy, 7.5, 13, 0, 0, Math.PI * 2); ctx.fill();
   664→    ctx.beginPath(); ctx.ellipse(f.x + Math.sin(animTime * 0.7 + f.ph) * 7, sy - 17, 11, 16, 0, 0, Math.PI * 2); ctx.fill();
   665→  }
   666→  /* مهِ پایه‌ی ساختمان‌ها: خاکستریِ دودی */
   667→  const fog = ctx.createLinearGradient(0, GROUND_Y - 46, 0, GROUND_Y);
   668→  fog.addColorStop(0, 'rgba(120,95,80,0)');
   669→  fog.addColorStop(1, d > 0.5 ? 'rgba(160,130,110,' + (0.24 * d).toFixed(2) + ')'
   670→                              : 'rgba(80,60,55,' + (0.2 * (1 - d)).toFixed(2) + ')');
   671→  ctx.fillStyle = fog;
   672→  ctx.fillRect(-20, GROUND_Y - 46, W + 40, 46);
   673→}
   674→
   675→/* ------------------ لایه‌ی رویی کات‌سین ------------------
   676→   نوارهای سینمایی، زیرنویس مراحل، تپش قرمزِ لرزش، محو/سیاهی
   677→   و متن‌های صفحه‌ی سیاه — بعد از همه‌ی لایه‌ها رسم می‌شود */
   678→function drawCutscene() {
   679→  if (!CS) return;
   680→  const t = CS.t;
   681→  /* نوارهای سینمایی (ورود/خروج نرم) */
   682→  const lIn = clamp(t / 0.45, 0, 1);
   683→  const lOut = clamp((CS_T.end - t) / 0.55, 0, 1);
   684→  const L = 46 * Math.min(lIn, lOut);
   685→  if (L > 0.5) {
   686→    ctx.fillStyle = '#000';
   687→    ctx.fillRect(0, 0, W, L);
   688→    ctx.fillRect(0, H - L, W, L);
   689→  }
   690→  /* زیرنویس مراحل (کپسول پایین صحنه) */
   691→  ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   692→  ctx.font = 'bold 15px Tahoma, sans-serif';
   693→  const caps = CS_CAPS.slice();
   694→  if (CS.giftApplied && CS.giftText) caps.push([CS_T.giftT, CS_T.giftT + 1.35, '🎁 ' + CS.giftText]);
   695→  for (const c of caps) {
   696→    if (t < c[0] || t > c[1]) continue;
   697→    const a = Math.min(1, (t - c[0]) / 0.25, (c[1] - t) / 0.25);
   698→    ctx.globalAlpha = a;
   699→    const tw = ctx.measureText(c[2]).width + 26;
   700→    ctx.fillStyle = 'rgba(0,0,0,0.66)';
   701→    rr(W / 2 - tw / 2, H - L - 42, tw, 26, 13); ctx.fill();
   702→    ctx.fillStyle = '#f3e9cf';
   703→    ctx.fillText(c[2], W / 2, H - L - 29);
   704→    ctx.globalAlpha = 1;
   705→  }
   706→  /* تپش قرمز لبه‌ها حین لرزش زمین */
   707→  if (t < CS_T.shakeEnd) {
   708→    const a = 0.1 + 0.08 * Math.sin(t * 26);
   709→    const g = ctx.createRadialGradient(W / 2, H / 2, H * 0.3, W / 2, H / 2, H * 0.8);
   710→    g.addColorStop(0, 'rgba(180,20,20,0)');
   711→    g.addColorStop(1, 'rgba(180,20,20,' + a.toFixed(2) + ')');
   712→    ctx.fillStyle = g;
   713→    ctx.fillRect(0, 0, W, H);
   714→  }
   715→  /* محو به سیاهی */
   716→  if (CS.fade > 0.003) {
   717→    ctx.fillStyle = 'rgba(0,0,0,' + CS.fade.toFixed(3) + ')';
   718→    ctx.fillRect(0, 0, W, H);
   719→  }
   720→  /* متن‌های صفحه‌ی سیاه: «به موقعیت جدید رسیدید...» / «برج بعدی» */
   721→  if (CS.fade > 0.965) {
   722→    const aM = clamp((t - CS_T.black - 0.2) / 0.5, 0, 1);
   723→    const aS = clamp((t - CS_T.black - 0.8) / 0.5, 0, 1);
   724→    if (aM > 0) {
   725→      ctx.globalAlpha = aM;
   726→      ctx.font = 'bold 30px Tahoma, sans-serif';
   727→      ctx.fillStyle = '#e8ecf5';
   728→      ctx.shadowColor = 'rgba(0,0,0,0.9)'; ctx.shadowBlur = 12;
   729→      ctx.fillText('به موقعیت جدید رسیدید...', W / 2, H / 2 - 16);
   730→      ctx.shadowBlur = 0;
   731→    }
   732→    if (aS > 0) {
   733→      ctx.globalAlpha = aS;
   734→      ctx.font = 'bold 19px Tahoma, sans-serif';
   735→      ctx.fillStyle = '#ffd873';
   736→      ctx.fillText('🏗 برج بعدی', W / 2, H / 2 + 26);
   737→    }
   738→    ctx.globalAlpha = 1;
   739→  }
   740→}
