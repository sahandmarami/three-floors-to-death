   123→/* برچسب موج برای HUD: «فصل ۲ • موج ۳/۸» یا «آخرین شب» */
   124→function waveLabel() {
   125→  if (state.wave >= FINAL_WAVE) return '🌑 آخرین شب';
   126→  const w = clamp(state.wave, 1, FINAL_WAVE);
   127→  return 'فصل ' + faNum(waveChapter(w)) + ' • موج ' + faNum(waveInChapter(w)) + '/' + faNum(WAVES_PER_CH);
   128→}

   130→/* ---------------------- ۵-۲) تولید ترکیب موج‌ها ---------------------- */
   131→/* ترکیب هر موج به‌صورت رویه‌ای از (فصل، شماره‌ی درون‌فصلی) ساخته می‌شود؛
   132→   موج آخرِ کل بازی فقط نگهبان است (زامبی‌هایش را خودش فراخوان می‌کند) */
   133→function buildWaveQueue(w) {
   134→  if (w >= FINAL_WAVE) return ['boss'];
   135→  const ch = waveChapter(w), wv = waveInChapter(w), C = CHAPTERS[ch - 1];
   136→  let n = Math.round((5 + wv * 1.85) * C.mulCount);          // (۱۱٫۰) موج‌ها پرتر
   137→  const f = wv >= C.unlock.fast ? Math.round(n * 0.32) : 0;
   138→  const t = wv >= C.unlock.tank ? Math.max(1, Math.round(n * 0.16)) : 0;
   139→  n = Math.max(3, n - Math.round((f + t) * 0.55));
   140→  const q = [];
   141→  for (let i = 0; i < n; i++) q.push('normal');
   142→  for (let i = 0; i < f; i++) q.push('fast');
   143→  for (let i = 0; i < t; i++) q.push('tank');
   144→  for (let i = q.length - 1; i > 0; i--) {          // شافل فیشر–ییتس
   145→    const j = Math.floor(rand(0, i + 1));
   146→    const tmp = q[i]; q[i] = q[j]; q[j] = tmp;
   147→  }
   148→  return q;
   149→}

   151→/* فاصله‌ی تولد زامبی‌ها: هر موج و هر فصل فشرده‌تر */
   152→function spawnInterval(w) {
   153→  const ch = waveChapter(w), wv = waveInChapter(w);
   154→  return clamp(2.0 - wv * 0.13 - (ch - 1) * 0.2, 0.5, 2.0);   // (۱۱٫۰) فشرده‌تر
   155→}

   215→/* آغاز موج سراسری n — صف ترکیبِ فصل‌محور + زمان‌بندی قطع برق */
   216→function startWave(n) {
   217→  state.wave = n;
   218→  state.mode = 'playing';
   219→  const isFinal = n >= FINAL_WAVE;
   220→  state.spawn = {
   221→    queue: buildWaveQueue(n),
   222→    timer: 1.2,
   223→    interval: spawnInterval(n),
   224→    bossDelay: false,
   225→  };
   226→  /* (۱۰٫۰) زمان‌بندی قطع‌برق‌های تصادفیِ این موج (فصل‌های ۲ به بعد) */
   227→  state.blackout.waveT = 0;
   228→  state.blackout.on = false;
   229→  state.blackout.schedule = [];
   230→  const C = curCh();
   231→  if (!isFinal && C.blackout) {
   232→    const k = irand(C.blackout.min, C.blackout.max);
   233→    for (let i = 0; i < k; i++) {
   234→      state.blackout.schedule.push({ at: rand(5, 24), dur: rand(8, 13), fired: false });
   235→    }
   236→  }
   237→  if (isFinal) {
   238→    $('lockBadge').classList.remove('hidden');    // فروشگاه قفل
   239→    setAnnounce('🌑 آخرین شب', 'فروشگاه قفل است — فقط کوله‌پشتی و برجت', 2.6);
   240→    SFX.roar();
   241→  } else {
   242→    setAnnounce(waveLabel(), 'دفاع کن! 🧟', 2);
   243→    SFX.wave();
   244→  }
   245→  SFX.dusk();
   246→  setBuildPanel(true, true);             // شب: پنل جمع
   247→  $('bpHead').classList.remove('attention');
   248→  updateHUD();
   249→}

   251→/* پایان موج: جایزه + مسیر جدید (انتقال فصل / روز / پیروزی) */
   252→function waveComplete() {
   253→  if (state.wave >= FINAL_WAVE) { startVictoryCine(); return; }   // برد واقعی (۱۰٫۰)
   254→  const bonus = 12 + state.wave * 4;
   255→  state.coins += bonus;
   256→  state.stats.earned += bonus;
   257→  addText(TOWER.x, TOWER_TOP_Y - 56, 'جایزه‌ی موج: +' + faNum(bonus) + ' 💰', '#ffd873', 16);
   258→  pulseCoins();
   259→  state.zombies = []; state.bullets = []; state.projectiles = [];
   260→  state.blackout.on = false;              // برق برمی‌گردد
   261→  $('lockBadge').classList.add('hidden');
   262→  /* آخرین موجِ فصل → انتقال سینمایی-قابل‌بازی به فصل بعد (۱۰٫۰) */
   263→  const ch = waveChapter(state.wave), wv = waveInChapter(state.wave);
   264→  if (wv === WAVES_PER_CH && ch < 4) {
   265→    startTransition(ch);
   266→    return;
   267→  }
   268→  enterDay();
   269→}

   272→function enterDay() {
   273→  state.mode = 'day';
   274→  state.dayTarget = 1;
   275→  state.goingNight = false;
   276→  state.slowmo = Math.max(state.slowmo, 0.45);

   280→    isFinalNext ? '⚠️ شب بعدی «آخرین شب» است — آخرین فرصت خرید!'
   281→                : 'موج ' + faNum(state.wave) + ' پاک شد — ارتقا بخر و شب را شروع کن', 3);

   283→  SFX.morning();
   284→  showDayPanel();
   285→  $('dpHint').textContent = isFinalNext
   286→    ? '⚠️ شب بعدی «آخرین شب» است — کوله‌پشتیت را پر کن!'
   287→    : 'زامبی نمی‌آید — ارتقا بخر و برجک بگذار';

   320→function spawnZombie(type) {
   321→  const T = ZTYPES[type || 'normal'];
   322→  const C = curCh();
   323→  const wv = waveInChapter(clamp(state.wave, 1, FINAL_WAVE));
   324→  const side = Math.random() < 0.5 ? -1 : 1;
   325→  /* (۱۱٫۰) نگهبانِ آخرین شب جونِ ثابت دارد — ضریب فصل روی او اعمال نمی‌شود
   326→     (زرهاش خودش سختی را می‌سازند: ×۰٫۵ معمولی، ×۲٫۶ کریتِ سر) */
   327→  const hp = Math.max(1, Math.round((T.hpBase + T.hpWave * wv) * ((type || 'normal') === 'boss' ? 1 : C.mulHp)));
   328→  state.zombies.push({

   425→function loadSaveData() {
   426→  try {
   427→    const d = JSON.parse(localStorage.getItem(SAVE_KEY));
   428→    if (!d || d.v !== 1) return null;
